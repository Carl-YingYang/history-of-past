    14→// ═══════════════════════════════════════════════════════════════════
    15→// CONSTANTS
    16→// ═══════════════════════════════════════════════════════════════════
    17→
    18→const INTERNAL_WIDTH = 480;
    19→const INTERNAL_HEIGHT = 320;
    20→const TILE_SIZE = 24;            // Internal pixels per tile (48/2 for 2× scale)
    21→const DISPLAY_TILE_SIZE = 48;    // Display pixels per tile
    22→const SLIDE_DURATION = 150;      // ms to slide between tiles
    23→const SLIDE_PAUSE = 50;          // ms pause between consecutive tiles when key held
    24→const IDLE_ANIM_RATE = 300;      // ms between idle animation frames
    25→const WALK_ANIM_RATE = 100;      // ms between walk animation frames
    26→const CAMERA_LERP = 0.12;        // Camera smoothing factor (0 = no follow, 1 = instant)
    27→const NPC_INTERACT_DIST = 1.5;   // Tile distance to interact with NPC
    28→const CHAR_RENDER_SIZE = 32;     // Character render size in internal pixels
    29→
    30→// 4-direction movement
    31→type Direction4 = 'north' | 'south' | 'east' | 'west';
    32→
    33→const DIR_OFFSET: Record<Direction4, { dr: number; dc: number }> = {
    34→  north: { dr: -1, dc: 0 },
    35→  south: { dr: 1, dc: 0 },
    36→  east:  { dr: 0, dc: 1 },
    37→  west:  { dr: 0, dc: -1 },
    38→};
    39→
    40→// Direction to facing string for sprite lookup
    41→const DIR_TO_FACING: Record<Direction4, string> = {
    42→  north: 'north',
    43→  south: 'south',
    44→  east: 'east',
    45→  west: 'west',
    46→};
    47→
    48→// Key to direction mapping
    49→const KEY_TO_DIR: Record<string, Direction4> = {
    50→  'w': 'north', 'arrowup': 'north',
    51→  's': 'south', 'arrowdown': 'south',
    52→  'd': 'east',  'arrowright': 'east',
    53→  'a': 'west',  'arrowleft': 'west',
    54→};

    60→interface PlayerState {
    61→  row: number;           // Current tile row
    62→  col: number;           // Current tile col
    63→  facing: Direction4;    // Current facing direction
    64→  isSliding: boolean;    // Whether currently sliding between tiles
    65→  slideStart: number;    // Timestamp when slide started
    66→  slideFromRow: number;  // Row we're sliding from
    67→  slideFromCol: number;  // Col we're sliding from
    68→  slideToRow: number;    // Row we're sliding to
    69→  slideToCol: number;    // Col we're sliding to
    70→  pixelX: number;        // Current render pixel X (internal coords)
    71→  pixelY: number;        // Current render pixel Y (internal coords)
    72→  animFrame: number;     // Current animation frame index
    73→  animTimer: number;     // Time accumulator for animation
    74→  bufferedDir: Direction4 | null;  // Buffered next direction during slide
    75→  lastMoveTime: number;  // Timestamp of last completed move
    76→}
    77→
    78→interface NPCState {
    79→  id: string;
    80→  characterKey: string;
    81→  row: number;

    85→  appearsAfterObjective?: string;
    86→  spriteSheet?: string;
    87→  animFrame: number;
    88→  animTimer: number;
    89→  displayName: string;
    90→  // Pixel position for rendering (computed from tile position)
    91→  pixelX: number;
    92→  pixelY: number;
    93→}

   166→

   184→

   187→  private buildingsLayer: number[][] = [];

   212→      triggered: false,

   214→  }

   217→    const spawn = mapData.playerSpawn as { row: number; col: number };
   218→    return {
   219→      row: spawn.row,
   220→      col: spawn.col,
   221→      facing: 'north',
   222→      isSliding: false,
   223→      slideStart: 0,
   224→      slideFromRow: spawn.row,
   225→      slideFromCol: spawn.col,
   226→      slideToRow: spawn.row,
   227→      slideToCol: spawn.col,
   228→      pixelX: spawn.col * TILE_SIZE,
   229→      pixelY: spawn.row * TILE_SIZE,
   230→      animFrame: 0,
   231→      animTimer: 0,

   238→  // INITIALIZATION
   239→  // ═══════════════════════════════════════════════════════════════════
   240→
   241→  async init(canvas: HTMLCanvasElement): Promise<void> {
   242→    this.canvas = canvas;
   243→
   244→    // Set canvas to internal resolution
   245→    canvas.width = INTERNAL_WIDTH;
   246→    canvas.height = INTERNAL_HEIGHT;
   247→    this.ctx = canvas.getContext('2d')!;
   248→    this.ctx.imageSmoothingEnabled = false;
   249→
   250→    // Create offscreen buffer (same internal resolution)
   251→    this.offscreen = document.createElement('canvas');
   252→    this.offscreen.width = INTERNAL_WIDTH;
   253→    this.offscreen.height = INTERNAL_HEIGHT;
   254→    this.offCtx = this.offscreen.getContext('2d')!;
   255→    this.offCtx.imageSmoothingEnabled = false;
   256→
   257→    // Load all assets via assetManager
   258→    await assetManager.loadAll();
   259→
   260→    // Load character sprites via spriteLoader
   261→    await spriteLoader.loadCharacter('student');
   262→    await spriteLoader.loadCharacter('ibara');
   263→
   264→    // Load save data
   265→    await saveManager.loadProgress();
   266→    saveManager.startAutoSave();
   267→
   268→    // Parse map data
   269→    this._parseMapData();
   270→
   271→    // Parse NPC positions from mapData
   272→    this._parseNPCData();
   273→
   274→    // Parse buildings and props
   275→    this._parseBuildingsAndProps();
   276→
   277→    // Parse trigger zones
   278→    this._parseTriggerZones();
   279→
   280→    // Load NPC interaction counts from localStorage
   281→    this._loadNpcInteractions();
   282→
   283→    // Initialize player position from save data if available
   284→    this._initPlayerFromSave();
   285→
   286→    // Set up input listeners
   287→    this._setupInputListeners();
   288→
   289→    // Initialize camera to center on player
   290→    this._snapCameraToPlayer();
   291→
   292→    // Start game loop
   293→    this.running = true;
   294→    this.lastTime = performance.now();
   295→    this._gameLoop(this.lastTime);
   296→
   297→    this.initialized = true;
   298→    gameEvents.emit('game:ready');
   299→
   300→    // Start ambient sounds
   301→    soundManager.initOnUserGesture();
   302→  }
   303→
   304→  destroy(): void {
   305→    this.running = false;
   306→    if (this.animFrameId) {
   307→      cancelAnimationFrame(this.animFrameId);
   308→      this.animFrameId = null;
   309→    }
   310→    this._removeInputListeners();
   311→    saveManager.stopAutoSave();
   312→  }

   318→  private _parseMapData(): void {
   319→    const layers = mapData.layers as any;
   320→    this.groundLayer = layers.ground;
   321→    this.buildingsLayer = layers.buildings;
   322→    this.collisionLayer = layers.collision;
   323→    this.overlayLayer = layers.overlay || Array.from({ length: mapData.height }, () => Array(mapData.width).fill(0));
   324→    this.mapWidth = mapData.width;
   325→    this.mapHeight = mapData.height;
   326→  }
   327→
   328→  private _parseNPCData(): void {
   329→    this.npcs = [];
   330→    const npcPositions = mapData.npcPositions as Record<string, any>;
   331→    for (const [id, pos] of Object.entries(npcPositions)) {
   332→      const charKey = id;
   333→      const charConfig = (characterData.characters as any)[charKey] || (characterData.characters as any)[id];
   334→      const displayName = charConfig?.displayName || id;
   335→
   336→      // Check visibility condition
   337→      let visible = true;
   338→      if (pos.appearsAfter) {
   339→        visible = saveManager.isObjectiveCompleted(pos.appearsAfter);
   340→      }
   341→
   342→      const row = pos.row as number;
   343→      const col = pos.col as number;
   344→
   345→      this.npcs.push({
   346→        id,
   347→        characterKey: charKey,
   348→        row,
   349→        col,
   350→        facing: this._toDirection4(pos.direction as string),
   351→        visible,
   352→        appearsAfterObjective: pos.appearsAfter,
   353→        spriteSheet: pos.spriteSheet || charConfig?.spriteSheet,
   354→        animFrame: 0,
   355→        animTimer: 0,
   356→        displayName,
   357→        pixelX: col * TILE_SIZE,
   358→        pixelY: row * TILE_SIZE,
   359→      });
   360→    }
   361→  }
   362→
   363→  private _parseBuildingsAndProps(): void {
   364→    // Parse buildings from mapData
   365→    this.buildings = [];
   366→    const buildingsData = mapData.buildings as any[];
   367→    if (buildingsData) {
   368→      for (const b of buildingsData) {
   369→        this.buildings.push({
   370→          key: b.key,
   371→          row: b.row,
   372→          col: b.col,
   373→          width: b.width,
   374→          height: b.height,
   375→          spriteWidth: b.spriteWidth || b.width * DISPLAY_TILE_SIZE,
   376→          spriteHeight: b.spriteHeight || b.height * DISPLAY_TILE_SIZE,
   377→        });
   378→      }
   379→    }
   380→
   381→    // Parse props from mapData
   382→    this.props = [];
   383→    const propsData = mapData.props as any[];
   384→    if (propsData) {
   385→      for (const p of propsData) {
   386→        this.props.push({
   387→          key: p.key,
   388→          row: p.row,
   389→          col: p.col,
   390→          width: p.width,
   391→          height: p.height,
   392→          spriteWidth: p.spriteWidth || p.width * DISPLAY_TILE_SIZE,
   393→          spriteHeight: p.spriteHeight || p.height * DISPLAY_TILE_SIZE,
   394→          solid: p.solid !== undefined ? p.solid : true,
   395→        });
   396→      }
   397→    }
   398→  }
   399→
   400→  private _parseTriggerZones(): void {
   401→    this.triggerZones = [];
   402→    const zones = mapData.triggerZones as any[];
   403→    if (zones) {
   404→      for (const z of zones) {
   405→        this.triggerZones.push({
   406→          id: z.id,
   407→          type: z.type,
   408→          npcId: z.npcId,
   409→          row: z.row,
   410→          col: z.col,
   411→          width: z.width,
   412→          height: z.height,
   413→          radius: z.radius,
   414→          requiresObjective: z.requiresObjective,
   415→          triggersTimeTransition: z.triggersTimeTransition,
   416→          dialogueId: z.dialogueId,
   417→          triggered: false,
   418→        });
   419→      }
   420→    }
   421→  }
   422→
   423→  private _initPlayerFromSave(): void {
   424→    const saveData = saveManager.getSaveData();
   425→    if (saveData.gameState.playerPosition) {
   426→      const pos = saveData.gameState.playerPosition;
   427→      // Validate position is within map and walkable
   428→      if (pos.row >= 0 && pos.row < this.mapHeight &&
   429→          pos.col >= 0 && pos.col < this.mapWidth &&
   430→          !this._isCollision(pos.row, pos.col)) {
   431→        this.player.row = pos.row;
   432→        this.player.col = pos.col;
   433→      }
   434→    }
   435→    // Recalculate pixel position
   436→    this.player.pixelX = this.player.col * TILE_SIZE;
   437→    this.player.pixelY = this.player.row * TILE_SIZE;
   438→    this.player.slideFromRow = this.player.row;
   439→    this.player.slideFromCol = this.player.col;
   440→    this.player.slideToRow = this.player.row;
   441→    this.player.slideToCol = this.player.col;
   442→  }

   445→  // INPUT HANDLING
   446→  // ═══════════════════════════════════════════════════════════════════
   447→
   448→  private _keydownHandler: ((e: KeyboardEvent) => void) | null = null;
   449→  private _keyupHandler: ((e: KeyboardEvent) => void) | null = null;
   450→
   451→  private _setupInputListeners(): void {
   452→    this._keydownHandler = (e: KeyboardEvent) => this._onKeyDown(e);
   453→    this._keyupHandler = (e: KeyboardEvent) => this._onKeyUp(e);
   454→    window.addEventListener('keydown', this._keydownHandler);
   455→    window.addEventListener('keyup', this._keyupHandler);
   456→  }
   457→
   458→  private _removeInputListeners(): void {
   459→    if (this._keydownHandler) window.removeEventListener('keydown', this._keydownHandler);
   460→    if (this._keyupHandler) window.removeEventListener('keyup', this._keyupHandler);
   461→  }
   462→
   463→  private _onKeyDown(e: KeyboardEvent): void {
   464→    // Ignore input when focus is in text fields
   465→    const target = e.target as HTMLElement;
   466→    if (target && (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable)) return;
   467→
   468→    const key = e.key.toLowerCase();
   469→
   470→    // Prevent default for game keys (stop page scrolling)
   471→    if (KEY_TO_DIR[key] || key === ' ' || key === 'enter') {
   472→      e.preventDefault();
   473→    }
   474→
   475→    this.keysDown.add(key);
   476→
   477→    // Interaction key (Space/Enter)
   478→    if (key === ' ' || key === 'enter') {
   479→      if (!this.interactPressed) {
   480→        this.interactPressed = true;
   481→        this._handleInteract();
   482→      }
   483→    }
   484→  }
   485→
   486→  private _onKeyUp(e: KeyboardEvent): void {
   487→    const key = e.key.toLowerCase();
   488→    this.keysDown.delete(key);
   489→
   490→    if (key === ' ' || key === 'enter') {
   491→      this.interactPressed = false;
   492→    }
   493→  }
   494→
   495→  // Touch / external controls
   496→  setMoveDirection(dir: 'up' | 'down' | 'left' | 'right' | null): void {
   497→    if (!dir) {
   498→      this.touchDir = null;
   499→      return;
   500→    }
   501→    const mapping: Record<string, Direction4> = {
   502→      up: 'north', down: 'south', left: 'west', right: 'east'
   503→    };
   504→    this.touchDir = mapping[dir] || null;
   505→  }
   506→
   507→  triggerInteract(): void {
   508→    this._handleInteract();
   509→  }

   511→  // ═══════════════════════════════════════════════════════════════════
   512→  // MOVEMENT SYSTEM (Pokémon Grid Movement)
   513→  // ═══════════════════════════════════════════════════════════════════
   514→
   515→  private _getDesiredDirection(): Direction4 | null {
   516→    // Check keyboard keys
   517→    for (const key of this.keysDown) {
   518→      const dir = KEY_TO_DIR[key];
   519→      if (dir) return dir;
   520→    }
   521→    // Check touch direction
   522→    return this.touchDir;
   523→  }
   524→
   525→  private _tryMove(dir: Direction4, now: number): void {
   526→    // Update facing direction immediately
   527→    this.player.facing = dir;
   528→
   529→    if (this.player.isSliding) {
   530→      // Buffer the direction for when slide completes
   531→      this.player.bufferedDir = dir;
   532→      return;
   533→    }
   534→
   535→    // Check pause between moves
   536→    if (now - this.player.lastMoveTime < SLIDE_PAUSE) return;
   537→
   538→    // Calculate target tile
   539→    const offset = DIR_OFFSET[dir];
   540→    const targetRow = this.player.row + offset.dr;
   541→    const targetCol = this.player.col + offset.dc;
   542→
   543→    // Check bounds
   544→    if (targetRow < 0 || targetRow >= this.mapHeight || targetCol < 0 || targetCol >= this.mapWidth) return;
   545→
   546→    // Check collision
   547→    if (this._isCollision(targetRow, targetCol)) return;
   548→
   549→    // Check if NPC occupies target tile
   550→    if (this._isNpcAtTile(targetRow, targetCol)) return;
   551→
   552→    // Check if prop occupies target tile
   553→    if (this._isPropAtTile(targetRow, targetCol)) return;
   554→
   555→    // Start sliding
   556→    this.player.isSliding = true;
   557→    this.player.slideStart = now;
   558→    this.player.slideFromRow = this.player.row;
   559→    this.player.slideFromCol = this.player.col;
   560→    this.player.slideToRow = targetRow;
   561→    this.player.slideToCol = targetCol;
   562→    this.player.bufferedDir = null;
   563→
   564→    // Play footstep sound
   565→    if (now - this.lastFootstepTime > this.FOOTSTEP_INTERVAL) {
   566→      soundManager.playFootstep();
   567→      this.lastFootstepTime = now;
   568→    }
   569→  }
   570→
   571→  private _updateMovement(now: number, dt: number): void {
   572→    if (this.dialogue.active) return; // No movement during dialogue
   573→
   574→    // Process desired direction
   575→    const desired = this._getDesiredDirection();
   576→    if (desired) {
   577→      this._tryMove(desired, now);
   578→    }
   579→
   580→    // Update slide interpolation
   581→    if (this.player.isSliding) {
   582→      const elapsed = now - this.player.slideStart;
   583→      const t = Math.min(1, elapsed / SLIDE_DURATION);
   584→
   585→      // Interpolate pixel position
   586→      const fromX = this.player.slideFromCol * TILE_SIZE;
   587→      const fromY = this.player.slideFromRow * TILE_SIZE;
   588→      const toX = this.player.slideToCol * TILE_SIZE;
   589→      const toY = this.player.slideToRow * TILE_SIZE;
   590→
   591→      this.player.pixelX = fromX + (toX - fromX) * t;
   592→      this.player.pixelY = fromY + (toY - fromY) * t;
   593→
   594→      // Update walk animation
   595→      this.player.animTimer += dt;
   596→      if (this.player.animTimer >= WALK_ANIM_RATE) {
   597→        this.player.animTimer -= WALK_ANIM_RATE;
   598→        this.player.animFrame++;
   599→      }
   600→
   601→      // Check if slide is complete
   602→      if (t >= 1) {
   603→        this.player.isSliding = false;
   604→        this.player.row = this.player.slideToRow;
   605→        this.player.col = this.player.slideToCol;
   606→        this.player.pixelX = this.player.col * TILE_SIZE;
   607→        this.player.pixelY = this.player.row * TILE_SIZE;
   608→        this.player.lastMoveTime = now;
   609→
   610→        // Emit player:move event
   611→        gameEvents.emit('player:move', { row: this.player.row, col: this.player.col });
   612→
   613→        // Update save state
   614→        saveManager.updateGameState({
   615→          playerPosition: { row: this.player.row, col: this.player.col },
   616→          playerDirection: this.player.facing,
   617→        });
   618→
   619→        // Check trigger zones
   620→        this._checkTriggerZones(now);
   621→
   622→        // Process buffered direction
   623→        if (this.player.bufferedDir) {
   624→          const bufDir = this.player.bufferedDir;
   625→          this.player.bufferedDir = null;
   626→          this._tryMove(bufDir, now);
   627→        } else {
   628→          // Switch to idle
   629→          this.player.animFrame = 0;
   630→          this.player.animTimer = 0;
   631→        }
   632→      }
   633→    } else {
   634→      // Idle animation
   635→      this.player.animTimer += dt;
   636→      if (this.player.animTimer >= IDLE_ANIM_RATE) {
   637→        this.player.animTimer -= IDLE_ANIM_RATE;
   638→        this.player.animFrame++;
   639→      }
   640→
   641→      // Snap pixel position to tile (safety)
   642→      this.player.pixelX = this.player.col * TILE_SIZE;
   643→      this.player.pixelY = this.player.row * TILE_SIZE;
   644→    }
   645→  }

   651→  private _isCollision(row: number, col: number): boolean {
   652→    if (row < 0 || row >= this.mapHeight || col < 0 || col >= this.mapWidth) return true;
   653→    return this.collisionLayer[row]?.[col] === 1;
   654→  }

   656→  private _isNpcAtTile(row: number, col: number): boolean {
   657→    return this.npcs.some(npc => npc.visible && npc.row === row && npc.col === col);
   658→  }

   660→  private _isPropAtTile(row: number, col: number): boolean {
   661→    for (const prop of this.props) {
   662→      if (!prop.solid) continue; // Skip non-solid (decorative) props
   663→      if (row >= prop.row && row < prop.row + prop.height &&
   664→          col >= prop.col && col < prop.col + prop.width) {
   665→        return true;
   666→      }
   667→    }
   668→    return false;
   669→  }

   675→  private _updateCamera(): void {
   676→    // Camera target: center on player
   677→    const playerCenterX = this.player.pixelX + TILE_SIZE / 2;
   678→    const playerCenterY = this.player.pixelY + TILE_SIZE / 2;
   679→
   680→    this.camera.targetX = playerCenterX - INTERNAL_WIDTH / 2;
   681→    this.camera.targetY = playerCenterY - INTERNAL_HEIGHT / 2;
   682→
   683→    // Clamp to map edges
   684→    const maxCamX = this.mapWidth * TILE_SIZE - INTERNAL_WIDTH;
   685→    const maxCamY = this.mapHeight * TILE_SIZE - INTERNAL_HEIGHT;
   686→    this.camera.targetX = Math.max(0, Math.min(maxCamX, this.camera.targetX));
   687→    this.camera.targetY = Math.max(0, Math.min(maxCamY, this.camera.targetY));
   688→
   689→    // Smooth lerp
   690→    this.camera.x += (this.camera.targetX - this.camera.x) * CAMERA_LERP;
   691→    this.camera.y += (this.camera.targetY - this.camera.y) * CAMERA_LERP;
   692→
   693→    // Clamp again (for precision)
   694→    this.camera.x = Math.max(0, Math.min(maxCamX, this.camera.x));
   695→    this.camera.y = Math.max(0, Math.min(maxCamY, this.camera.y));
   696→  }

   698→  private _snapCameraToPlayer(): void {
   699→    const playerCenterX = this.player.pixelX + TILE_SIZE / 2;
   700→    const playerCenterY = this.player.pixelY + TILE_SIZE / 2;
   701→    this.camera.x = playerCenterX - INTERNAL_WIDTH / 2;
   702→    this.camera.y = playerCenterY - INTERNAL_HEIGHT / 2;
   703→    const maxCamX = this.mapWidth * TILE_SIZE - INTERNAL_WIDTH;
   704→    const maxCamY = this.mapHeight * TILE_SIZE - INTERNAL_HEIGHT;
   705→    this.camera.x = Math.max(0, Math.min(maxCamX, this.camera.x));
   706→    this.camera.y = Math.max(0, Math.min(maxCamY, this.camera.y));
   707→    this.camera.targetX = this.camera.x;
   708→    this.camera.targetY = this.camera.y;
   709→  }

   715→  private _updateNPCs(dt: number): void {
   716→    for (const npc of this.npcs) {
   717→      // Check appearance condition
   718→      if (npc.appearsAfterObjective) {
   719→        const wasVisible = npc.visible;
   720→        npc.visible = saveManager.isObjectiveCompleted(npc.appearsAfterObjective);
   721→        if (npc.visible && !wasVisible) {
   722→          gameEvents.emit('ibarra:appear');
   723→        }
   724→      }
   725→
   726→      // Update idle animation
   727→      npc.animTimer += dt;
   728→      if (npc.animTimer >= IDLE_ANIM_RATE) {
   729→        npc.animTimer -= IDLE_ANIM_RATE;
   730→        npc.animFrame++;
   731→      }
   732→
   733→      // Update pixel position (NPCs don't move, but snap to tile)
   734→      npc.pixelX = npc.col * TILE_SIZE;
   735→      npc.pixelY = npc.row * TILE_SIZE;
   736→    }

   740→  // INTERACTION & TRIGGER SYSTEM
   741→  // ═══════════════════════════════════════════════════════════════════

   743→  private _handleInteract(): void {
   744→    if (this.dialogue.active) {
   745→      // Advance dialogue
   746→      this.advanceDialogue();
   747→      return;
   748→    }
   749→
   750→    // Try to talk to an adjacent NPC
   751→    const facing = this.player.facing;
   752→    const offset = DIR_OFFSET[facing];
   753→    const targetRow = this.player.row + offset.dr;
   754→    const targetCol = this.player.col + offset.dc;
   755→
   756→    // Check NPCs at target tile or current tile
   757→    let closestNpc: NPCState | null = null;
   758→    let closestDist = Infinity;
   759→
   760→    for (const npc of this.npcs) {
   761→      if (!npc.visible) continue;
   762→      const dist = Math.abs(npc.row - targetRow) + Math.abs(npc.col - targetCol);
   763→      if (dist === 0 && dist < closestDist) {
   764→        closestNpc = npc;
   765→        closestDist = dist;
   766→      }
   767→    }
   768→
   769→    // Also check NPCs on player's tile (standing on top of them)
   770→    if (!closestNpc) {
   771→      for (const npc of this.npcs) {
   772→        if (!npc.visible) continue;
   773→        const dist = Math.abs(npc.row - this.player.row) + Math.abs(npc.col - this.player.col);
   774→        if (dist <= 1 && dist < closestDist) {
   775→          closestNpc = npc;
   776→          closestDist = dist;
   777→        }
   778→      }
   779→    }
   780→
   781→    if (closestNpc) {
   782→      // Face NPC toward player
   783→      closestNpc.facing = this._oppositeDirection(this.player.facing);
   784→      this._startNpcDialogue(closestNpc);
   785→    }
   786→  }

   788→  private _startNpcDialogue(npc: NPCState): void {
   789→    // Increment interaction count
   790→    this.npcInteractionCounts[npc.id] = (this.npcInteractionCounts[npc.id] || 0) + 1;
   791→    this._saveNpcInteractions();
   792→
   793→    // Determine which dialogue to use
   794→    let dialogueId: string | null = null;
   795→    const npcId = npc.id;
   796→
   797→    // Check for first-time dialogues based on NPC ID
   798→    if (npcId === 'mang-tenyo') {
   799→      if (!saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   800→        dialogueId = 'mang-tenyo-first';
   801→      } else if (!saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   802→        dialogueId = 'mang-tenyo-repeat';
   803→      } else {
   804→        dialogueId = 'mang-tenyo-after-gossip';
   805→      }
   806→    } else if (npcId === 'vendor-1' || npcId === 'vendor-2' || npcId === 'kitchen-staff-1' || npcId === 'kitchen-staff-2') {
   807→      // Vendors share market-gossip dialogue
   808→      if (saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo') && !saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   809→        dialogueId = 'market-gossip';
   810→      } else {
   811→        // Generic vendor repeat
   812→        dialogueId = 'mang-tenyo-repeat'; // Fallback
   813→      }
   814→    } else if (npcId === 'ibara') {
   815→      // Ibarra sighting
   816→      dialogueId = 'ibarra-sighting';
   817→    }
   818→
   819→    if (dialogueId && (dialogueData.dialogues as any)[dialogueId]) {
   820→      this._startDialogue(dialogueId, npcId);
   821→    }
   822→  }
   823→
   824→  private _startDialogue(dialogueId: string, npcId: string | null): void {
   825→    const dialogueObj = (dialogueData.dialogues as any)[dialogueId];
   826→    if (!dialogueObj) return;
   827→
   828→    const lines: DialogueLine[] = dialogueObj.lines.map((l: any) => ({
   829→      speaker: l.speaker,
   830→      text: l.text,
   831→      translation: l.translation,
   832→      isTimeTransition: l.isTimeTransition,
   833→      isIbarraSighting: l.isIbarraSighting,
   834→    }));
   835→
   836→    this.dialogue = {
   837→      active: true,
   838→      dialogueId,
   839→      lines,
   840→      currentLineIndex: 0,
   841→      npcId,
   842→      triggered: true,
   843→    };
   844→
   845→    // Emit dialogue:start event
   846→    const firstLine = lines[0];
   847→    gameEvents.emit('dialogue:start', {
   848→      dialogueId,
   849→      line: firstLine,
   850→      lineIndex: 0,
   851→      totalLines: lines.length,
   852→    });
   853→
   854→    // Complete any objectives this dialogue unlocks
   855→    if (dialogueObj.unlocksObjective) {
   856→      saveManager.completeObjective(dialogueObj.unlocksObjective);
   857→      saveManager.saveProgress();
   858→    }
   859→  }
   860→
   861→  advanceDialogue(): void {
   862→    if (!this.dialogue.active) return;
   863→
   864→    const nextIndex = this.dialogue.currentLineIndex + 1;
   865→
   866→    if (nextIndex >= this.dialogue.lines.length) {
   867→      // End dialogue
   868→      const wasTimeTransition = this.dialogue.lines.some(l => l.isTimeTransition);
   869→
   870→      this.dialogue = {
   871→        active: false,
   872→        dialogueId: null,
   873→        lines: [],
   874→        currentLineIndex: 0,
   875→        npcId: null,
   876→        triggered: false,
   877→      };
   878→
   879→      gameEvents.emit('dialogue:end');
   880→
   881→      // Handle time transition
   882→      if (wasTimeTransition) {
   883→        gameEvents.emit('time:transition', 'morning');
   884→        saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   885→        soundManager.play('time-transition');
   886→      }
   887→
   888→      return;
   889→    }
   890→
   891→    // Advance to next line
   892→    this.dialogue.currentLineIndex = nextIndex;
   893→    const line = this.dialogue.lines[nextIndex];
   894→
   895→    gameEvents.emit('dialogue:line', {
   896→      dialogueId: this.dialogue.dialogueId,
   897→      line,
   898→      lineIndex: nextIndex,
   899→      totalLines: this.dialogue.lines.length,
   900→    });
   901→
   902→    // Handle time transition in dialogue line
   903→    if (line.isTimeTransition) {
   904→      gameEvents.emit('time:transition', 'morning');
   905→      saveManager.updateGameState({ timeOfDay: 'morning' });
   906→    }
   907→  }

   909→  private _checkTriggerZones(now: number): void {
   910→    for (const zone of this.triggerZones) {
   911→      if (zone.triggered) continue;
   912→
   913→      // Check objective requirement
   914→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   915→
   916→      // Check if player is in the zone
   917→      const inZone = this.player.row >= zone.row && this.player.row < zone.row + zone.height &&
   918→                     this.player.col >= zone.col && this.player.col < zone.col + zone.width;
   919→
   920→      // Or within radius
   921→      const centerRow = zone.row + (zone.height - 1) / 2;
   922→      const centerCol = zone.col + (zone.width - 1) / 2;
   923→      const dist = Math.sqrt(
   924→        Math.pow(this.player.row - centerRow, 2) +
   925→        Math.pow(this.player.col - centerCol, 2)
   926→      );
   927→      const inRadius = zone.radius > 0 && dist <= zone.radius;
   928→
   929→      if (inZone || inRadius) {
   930→        zone.triggered = true;
   931→
   932→        if (zone.dialogueId) {
   933→          this._startDialogue(zone.dialogueId, zone.npcId || null);
   934→        }
   935→
   936→        gameEvents.emit('interaction:trigger', { zoneId: zone.id, type: zone.type });
   937→
   938→        // If this triggers a time transition, handle it
   939→        if (zone.triggersTimeTransition) {
   940→          // Time transition will be handled by the dialogue lines
   941→        }
   942→      }
   943→    }
   944→  }

   968→  private _render(): void {
   969→    if (!this.offCtx || !this.ctx) return;
   970→
   971→    const ctx = this.offCtx;
   972→
   973→    // Clear
   974→    ctx.fillStyle = '#1a1a2e';
   975→    ctx.fillRect(0, 0, INTERNAL_WIDTH, INTERNAL_HEIGHT);
   976→
   977→    const camX = Math.round(this.camera.x);
   978→    const camY = Math.round(this.camera.y);
   979→
   980→    // Calculate visible tile range
   981→    const startCol = Math.floor(camX / TILE_SIZE);
   982→    const startRow = Math.floor(camY / TILE_SIZE);
   983→    const endCol = Math.ceil((camX + INTERNAL_WIDTH) / TILE_SIZE);
   984→    const endRow = Math.ceil((camY + INTERNAL_HEIGHT) / TILE_SIZE);
   985→
   986→    // ── LAYER 1: Ground tiles ──
   987→    this._renderGroundTiles(ctx, camX, camY, startRow, endRow, endCol);
   988→
   989→    // ── LAYER 2: Building sprites ──
   990→    this._renderBuildings(ctx, camX, camY);
   991→
   992→    // ── LAYER 3: Props (below characters) ──
   993→    this._renderProps(ctx, camX, camY);
   994→
   995→    // ── LAYER 4: Y-sorted characters (NPCs + Player) ──
   996→    this._renderCharacters(ctx, camX, camY);
   997→
   998→    // ── LAYER 5: Overlay (shadows) ──
   999→    this._renderOverlay(ctx, camX, camY, startRow, endRow, endCol);
  1000→
  1001→    // ── Copy to display canvas ──
  1002→    this.ctx.imageSmoothingEnabled = false;
  1003→    this.ctx.drawImage(this.offscreen!, 0, 0);
  1004→  }

  1006→  private _renderGroundTiles(
  1007→    ctx: CanvasRenderingContext2D, camX: number, camY: number,
  1008→    startRow: number, endRow: number, endCol: number
  1009→  ): void {
  1010→    // Tile types that are buildings (rendered as sprites, not tiles)
  1011→    const BUILDING_TILE_TYPES = new Set([5, 6]);
  1012→
  1013→    // Fallback colors for tile types without images
  1014→    const fallbackColors: Record<number, string> = {
  1015→      1: '#C4A76C', 2: '#8FBC8F', 3: '#D2B48C', 4: '#A0522D',
  1016→      5: '#696969', 6: '#8B4513', 7: '#6B8E6B', 8: '#555555',
  1017→      9: '#B8A070', 10: '#4A90A4', 11: '#7CB342', 12: '#00000033',
  1018→    };
  1019→
  1020→    for (let row = startRow; row <= endRow && row < this.mapHeight; row++) {
  1021→      if (row < 0) continue;
  1022→      for (let col = 0; col <= endCol && col < this.mapWidth; col++) {
  1023→        if (col < 0) continue;
  1024→
  1025→        const tileType = this.groundLayer[row]?.[col] || 0;
  1026→        if (tileType === 0) continue; // Empty tile — should not exist but safety check
  1027→
  1028→        // Skip building tile types (church=5, mansion=6) — they are rendered as sprites
  1029→        if (BUILDING_TILE_TYPES.has(tileType)) continue;
  1030→
  1031→        const screenX = col * TILE_SIZE - camX;
  1032→        const screenY = row * TILE_SIZE - camY;
  1033→
  1034→        // Render tile image from assetManager
  1035→        const tileAsset = assetManager.getTile(tileType);
  1036→        if (tileAsset) {
  1037→          ctx.drawImage(tileAsset.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1038→        } else {
  1039→          // Fallback: colored rectangle for tiles without images
  1040→          ctx.fillStyle = fallbackColors[tileType] || '#FF00FF';
  1041→          ctx.fillRect(screenX, screenY, TILE_SIZE, TILE_SIZE);
  1042→        }
  1043→      }
  1044→    }
  1045→  }
  1046→
  1047→  private _renderBuildings(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {

  1057→      const drawWidth = building.width * TILE_SIZE;
  1058→      const drawHeight = building.height * TILE_SIZE;
  1059→
  1060→      ctx.drawImage(
  1061→        buildingAsset.image,
  1062→        0, 0, buildingAsset.image.width, buildingAsset.image.height,
  1063→        bx, by, drawWidth, drawHeight
  1064→      );
  1065→    }
  1066→  }

  1068→  private _renderProps(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {

  1070→      const propAsset = assetManager.getProp(prop.key);

  1073→      const px = prop.col * TILE_SIZE - camX;
  1074→      const py = prop.row * TILE_SIZE - camY;
  1075→      const drawWidth = prop.width * TILE_SIZE;
  1076→      const drawHeight = prop.height * TILE_SIZE;

  1086→  private _renderCharacters(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {
  1087→    // Collect all renderable characters with their Y position for sorting
  1088→    interface Renderable {
  1089→      type: 'player' | 'npc';
  1090→      y: number;
  1091→      data: any;
  1092→    }
  1093→
  1094→    const renderables: Renderable[] = [];
  1095→
  1096→    // Add player
  1097→    renderables.push({
  1098→      type: 'player',
  1099→      y: this.player.pixelY + TILE_SIZE, // Bottom of sprite for Y-sorting
  1100→      data: this.player,
  1101→    });
  1102→
  1103→    // Add visible NPCs
  1104→    for (const npc of this.npcs) {
  1105→      if (!npc.visible) continue;
  1106→      renderables.push({
  1107→        type: 'npc',
  1108→        y: npc.pixelY + TILE_SIZE,
  1109→        data: npc,
  1110→      });
  1111→    }
  1112→
  1113→    // Sort by Y position (top to bottom)
  1114→    renderables.sort((a, b) => a.y - b.y);
  1115→
  1116→    // Render in sorted order
  1117→    for (const r of renderables) {
  1118→      if (r.type === 'player') {
  1119→        this._renderPlayer(ctx, camX, camY);
  1120→      } else {
  1121→        this._renderNPC(ctx, r.data as NPCState, camX, camY);
  1122→      }
  1123→    }
  1124→  }
  1125→
  1126→  private _renderPlayer(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {
  1127→    const screenX = this.player.pixelX - camX;
  1128→    const screenY = this.player.pixelY - camY;
  1129→
  1130→    // Try to get player sprite from spriteLoader
  1131→    const facing = DIR_TO_FACING[this.player.facing];
  1132→    const isMoving = this.player.isSliding;

  1157→  private _renderNPC(ctx: CanvasRenderingContext2D, npc: NPCState, camX: number, camY: number): void {
  1158→    const screenX = npc.pixelX - camX;
  1159→    const screenY = npc.pixelY - camY;
  1160→
  1161→    // Try to render using sprite sheet from assetManager
  1162→    if (npc.spriteSheet) {
  1163→      const spriteSet = assetManager.getNpcSheet(npc.spriteSheet);
  1164→      if (spriteSet) {
  1165→        const facing = DIR_TO_FACING[npc.facing];
  1166→        const frames = spriteSet.idle; // NPCs use idle animation
  1167→        const dirFrames = frames[facing] || frames['south'];
  1168→
  1169→        if (dirFrames && dirFrames.length > 0) {
  1170→          const frameIndex = npc.animFrame % dirFrames.length;
  1171→          const frame = dirFrames[frameIndex];
  1172→
  1173→          // Scale NPC sprite to fit character render area
  1174→          const drawWidth = CHAR_RENDER_SIZE;
  1175→          const drawHeight = CHAR_RENDER_SIZE + 8; // NPCs can be slightly taller
  1176→          const offsetX = (TILE_SIZE - drawWidth) / 2;
  1177→          const offsetY = TILE_SIZE - drawHeight;
  1178→
  1179→          ctx.drawImage(
  1180→            frame.canvas,
  1181→            0, 0, frame.width, frame.height,
  1182→            screenX + offsetX, screenY + offsetY, drawWidth, drawHeight
  1183→          );
  1184→          return;
  1185→        }
  1186→      }
  1187→    }
  1188→
  1189→    // Try spriteLoader as fallback (for ibara)
  1190→    const charKey = npc.characterKey;
  1191→    const facing = DIR_TO_FACING[npc.facing];
  1192→    const frame = spriteLoader.getFrame(charKey, 'Breathing_Idle', facing, npc.animFrame % 4);
  1193→
  1194→    if (frame) {
  1195→      const drawSize = CHAR_RENDER_SIZE;
  1196→      const offsetX = (TILE_SIZE - drawSize) / 2;
  1197→      const offsetY = TILE_SIZE - drawSize;
  1198→
  1199→      ctx.drawImage(
  1200→        frame,
  1201→        0, 0, frame.width, frame.height,
  1202→        screenX + offsetX, screenY + offsetY, drawSize, drawSize
  1203→      );
  1204→    } else {
  1205→      // Final fallback: colored placeholder
  1206→      const npcColors: Record<string, string> = {
  1207→        'mang-tenyo': '#8B4513',
  1208→        'vendor-1': '#D2691E',
  1209→        'vendor-2': '#CD853F',
  1210→        'ibara': '#1E3A5F',
  1211→        'kitchen-staff-1': '#D2691E',
  1212→        'kitchen-staff-2': '#CD853F',
  1213→      };

  1216→    }
  1217→  }

  1247→  private _renderOverlay(
  1248→    ctx: CanvasRenderingContext2D, camX: number, camY: number,
  1249→    startRow: number, endRow: number, endCol: number
  1250→  ): void {
  1251→    for (let row = startRow; row <= endRow && row < this.mapHeight; row++) {
  1252→      if (row < 0) continue;
  1253→      for (let col = 0; col <= endCol && col < this.mapWidth; col++) {
  1254→        if (col < 0) continue;
  1255→
  1256→        const overlayType = this.overlayLayer[row]?.[col] || 0;
  1257→        if (overlayType === 0) continue;
  1258→
  1259→        const screenX = col * TILE_SIZE - camX;
  1260→        const screenY = row * TILE_SIZE - camY;
  1261→
  1262→        const tileAsset = assetManager.getTile(overlayType);
  1263→        if (tileAsset) {
  1264→          ctx.globalAlpha = 0.5;
  1265→          ctx.drawImage(tileAsset.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1266→          ctx.globalAlpha = 1.0;
  1267→        }
  1268→      }
  1269→    }
  1270→  }

  1276→  private _gameLoop = (now: number): void => {
  1277→    if (!this.running) return;
  1278→
  1279→    const dt = Math.min(now - this.lastTime, 50); // Cap delta to avoid huge jumps
  1280→    this.lastTime = now;
  1281→
  1282→    // Update systems
  1283→    this._updateMovement(now, dt);
  1284→    this._updateNPCs(dt);
  1285→    this._updateCamera();
  1286→    this._updateAmbient();
  1287→
  1288→    // Render
  1289→    this._render();
  1290→
  1291→    // Schedule next frame
  1292→    this.animFrameId = requestAnimationFrame(this._gameLoop);
  1293→  };
