   102→// Deterministic hash for tile decoration patterns
   103→function tileHash(row: number, col: number, seed: number = 0): number {
   104→  let h = seed + row * 374761393 + col * 668265263;
   105→  h = (h ^ (h >> 13)) * 1274126177;
   106→  h = h ^ (h >> 16);
   107→  return (h & 0x7FFFFFFF) / 0x7FFFFFFF; // Returns 0..1
   108→}

   163→  private _computeBuildingGroups(): void {
   164→    // For each building tile type, find contiguous rectangular groups
   165→    const buildingTypes = [5, 6]; // Church and Market building types
   166→    for (const tileType of buildingTypes) {
   167→      const visited: Set<string> = new Set();
   168→      const groups: { minRow: number; minCol: number; maxRow: number; maxCol: number }[] = [];
   169→      
   170→      for (let row = 0; row < mapData.height; row++) {
   171→        for (let col = 0; col < mapData.width; col++) {
   172→          const key = `${row},${col}`;
   173→          if (visited.has(key)) continue;
   174→          if (mapData.layers.buildings[row]?.[col] !== tileType) continue;
   175→          
   176→          // Find bounding box of contiguous group
   177→          let minRow = row, maxRow = row, minCol = col, maxCol = col;
   178→          const toVisit: [number, number][] = [[row, col]];
   179→          
   180→          while (toVisit.length > 0) {
   181→            const [r, c] = toVisit.pop()!;
   182→            const k = `${r},${c}`;
   183→            if (visited.has(k)) continue;
   184→            if (mapData.layers.buildings[r]?.[c] !== tileType) continue;
   185→            visited.add(k);
   186→            minRow = Math.min(minRow, r);
   187→            maxRow = Math.max(maxRow, r);
   188→            minCol = Math.min(minCol, c);
   189→            maxCol = Math.max(maxCol, c);
   190→            toVisit.push([r - 1, c], [r + 1, c], [r, c - 1], [r, c + 1]);
   191→          }
   192→          
   193→          groups.push({ minRow, minCol, maxRow, maxCol });
   194→        }
   195→      }
   196→      
   197→      this.buildingGroups.set(tileType, groups);
   198→    }
   199→  }

   201→  async init(canvas: HTMLCanvasElement): Promise<void> {

   213→    }
   214→
   215→    // Create player entity
   216→    const spawn = mapData.playerSpawn;
   217→    this.player = {
   218→      id: 'player',
   219→      x: spawn.col * this.tileSize + this.tileSize / 2,
   220→      y: spawn.row * this.tileSize + this.tileSize / 2,
   221→      direction: 'north',
   222→      characterKey: 'student',
   223→      isMoving: false,
   224→      animationFrame: 0,
   225→      animationTimer: 0,
   226→      speed: 120,
   227→      visible: true,
   228→      isPlayer: true,
   229→      dialogueTriggered: false,
   230→    };
   231→    this.entities.set('player', this.player);
   232→
   233→    // Create NPC entities
   234→    const npcPositions = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   235→    for (const [npcId, pos] of Object.entries(npcPositions)) {
   236→      const entity: Entity = {
   237→        id: npcId,
   238→        x: pos.col * this.tileSize + this.tileSize / 2,
   239→        y: pos.row * this.tileSize + this.tileSize / 2,
   240→        direction: pos.direction as Direction,
   241→        characterKey: npcId,
   242→        isMoving: false,
   243→        animationFrame: 0,
   244→        animationTimer: 0,
   245→        speed: 0,
   246→        visible: !pos.appearsAfter || saveManager.isObjectiveCompleted(pos.appearsAfter),
   247→        isPlayer: false,
   248→        dialogueTriggered: false,
   249→        appearsAfterObjective: pos.appearsAfter,
   250→      };
   251→      this.entities.set(npcId, entity);
   252→    }
   253→
   254→    // Load saved progress
   255→    await saveManager.loadProgress();
   256→    const gameState = saveManager.getGameState();
   257→    this.chapterPhase = gameState.chapterPhase;
   258→    this.timeOfDay = gameState.timeOfDay;
   259→    
   260→    if (this.chapterPhase !== 'intro') {
   261→      this.introShown = true;
   262→    }
   263→
   264→    // Update entity visibility based on completed objectives
   265→    this._updateEntityVisibility();
   266→
   267→    // Mark already-triggered zones
   268→    this.triggerZones.forEach(zone => {
   269→      if (zone.requiresObjective && saveManager.isObjectiveCompleted(zone.requiresObjective)) {
   270→        // The prerequisite is done, but this zone itself may not have been triggered yet
   271→        // Only mark as triggered if the objective THIS zone completes is also done
   272→      }
   273→    });
   274→
   275→    // Bind input handlers
   276→    this._bindInput();
   277→
   278→    // Listen for objective completion events to spawn celebration bursts
   279→    gameEvents.on('quest:objectiveComplete', (objectiveId: unknown) => {
   280→      const id = objectiveId as string;
   281→      const loc = OBJECTIVE_LOCATIONS[id];
   282→      if (loc) {
   283→        // Spawn burst at the objective's location in world coords
   284→        this._spawnCelebrationBurst(
   285→          loc.col * this.tileSize + this.tileSize / 2,
   286→          loc.row * this.tileSize + this.tileSize / 2
   287→        );
   288→      }
   289→    });
   290→
   291→    // Start auto-save
   292→    saveManager.startAutoSave();
   293→
   294→    // Emit game ready
   295→    gameEvents.emit('game:ready');
   296→
   297→    // Initialize some particles
   298→    this._initParticles();
   299→
   300→    // Start game loop
   301→    this.lastTimestamp = performance.now();
   302→    this._gameLoop(this.lastTimestamp);
   303→  }

   305→  /**
   306→   * Spawn a golden celebration burst at the given world position.
   307→   * Used when an objective is completed.
   308→   */
   309→  private _spawnCelebrationBurst(worldX: number, worldY: number): void {
   310→    const colors = ['#FFD700', '#FFA500', '#FFEC8B', '#FFEE00', '#FFC125'];
   311→    const count = 28;
   312→    for (let i = 0; i < count; i++) {
   313→      const angle = (i / count) * Math.PI * 2 + Math.random() * 0.3;
   314→      const speed = 60 + Math.random() * 80;
   315→      this.burstParticles.push({
   316→        x: worldX,
   317→        y: worldY,
   318→        vx: Math.cos(angle) * speed,
   319→        vy: Math.sin(angle) * speed - 30, // Slight upward bias
   320→        life: 0,
   321→        maxLife: 1.0 + Math.random() * 0.6,
   322→        color: colors[Math.floor(Math.random() * colors.length)],
   323→        size: 2 + Math.random() * 3,
   324→        gravity: 80,
   325→      });
   326→    }
   327→  }
   328→
   329→  private _updateBurstParticles(dt: number): void {
   330→    for (let i = this.burstParticles.length - 1; i >= 0; i--) {
   331→      const p = this.burstParticles[i];
   332→      p.life += dt;
   333→      p.vy += p.gravity * dt;     // gravity pulls down
   334→      p.x += p.vx * dt;
   335→      p.y += p.vy * dt;
   336→      // Air resistance
   337→      p.vx *= 0.96;
   338→      p.vy *= 0.96;
   339→      if (p.life >= p.maxLife) {
   340→        this.burstParticles.splice(i, 1);
   341→      }
   342→    }
   343→  }
   344→
   345→  private _renderBurstParticles(ctx: CanvasRenderingContext2D): void {
   346→    for (const p of this.burstParticles) {
   347→      const screenX = p.x - this.cameraX;
   348→      const screenY = p.y - this.cameraY;
   349→      if (screenX < -20 || screenX > this.viewWidth + 20 || screenY < -20 || screenY > this.viewHeight + 20) continue;
   350→      const lifeRatio = p.life / p.maxLife;
   351→      const alpha = Math.max(0, 1 - lifeRatio);
   352→      // Glow
   353→      ctx.globalAlpha = alpha * 0.4;
   354→      ctx.fillStyle = p.color;
   355→      ctx.beginPath();
   356→      ctx.arc(screenX, screenY, p.size * 2.5, 0, Math.PI * 2);
   357→      ctx.fill();
   358→      // Core
   359→      ctx.globalAlpha = alpha;
   360→      ctx.fillStyle = p.color;
   361→      ctx.beginPath();
   362→      ctx.arc(screenX, screenY, p.size, 0, Math.PI * 2);
   363→      ctx.fill();
   364→    }
   365→    ctx.globalAlpha = 1;
   366→  }

   417→  private _handleInteract(): void {
   418→    if (this.dialogueActive) return;
   419→    if (this.interactCooldown > 0) return;
   420→    
   421→    // Check if player is near an NPC they can talk to
   422→    if (!this.player) return;
   423→
   424→    for (const [npcId, npc] of this.entities) {
   425→      if (npc.isPlayer || !npc.visible) continue;
   426→      
   427→      const dist = this._distanceBetween(this.player, npc);
   428→      if (dist < this.tileSize * 2) {
   429→        this.interactCooldown = 500;
   430→        
   431→        // Determine which dialogue to trigger
   432→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   433→          this._startDialogue('mang-tenyo-first');
   434→          return;
   435→        }
   436→        // Mang Tenyo can be talked to again — different dialogue based on progress
   437→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   438→          // If gossip objective completed, give more insightful dialogue
   439→          if (saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   440→            this._startDialogue('mang-tenyo-after-gossip');
   441→          } else {
   442→            this._startDialogue('mang-tenyo-repeat');
   443→          }
   444→          return;
   445→        }
   446→      }
   447→    }
   448→  }
   449→
   450→  private _distanceBetween(a: Entity, b: Entity): number {
   451→    const dx = a.x - b.x;
   452→    const dy = a.y - b.y;
   453→    return Math.sqrt(dx * dx + dy * dy);
   454→  }

   456→  private _startDialogue(dialogueId: string): void {
   457→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   458→    if (!dialogue) {
   459→      console.warn(`Dialogue not found: ${dialogueId}`);
   460→      return;
   461→    }
   462→
   463→    this.dialogueActive = true;
   464→    this.currentDialogueId = dialogueId;
   465→    this.currentDialogueLineIndex = 0;
   466→
   467→    gameEvents.emit('dialogue:start', {
   468→      dialogueId,
   469→      line: dialogue.lines[0],
   470→      lineIndex: 0,
   471→      totalLines: dialogue.lines.length,
   472→    });
   473→  }
   474→
   475→  advanceDialogue(): void {
   476→    if (!this.currentDialogueId) return;
   477→    
   478→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];
   479→    if (!dialogue) return;
   480→
   481→    this.currentDialogueLineIndex++;
   482→
   483→    if (this.currentDialogueLineIndex >= dialogue.lines.length) {
   484→      // Dialogue complete
   485→      this._endDialogue(dialogue);
   486→      return;
   487→    }
   488→
   489→    const line = dialogue.lines[this.currentDialogueLineIndex];
   490→    
   491→    // Check for time transition
   492→    if (line.isTimeTransition) {
   493→      this.timeOfDay = 'morning';
   494→      this.chapterPhase = 'ibarra-sighting';
   495→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   496→      gameEvents.emit('time:transition', 'morning');
   497→    }
   498→
   499→    // Check for Ibarra sighting
   500→    if (line.isIbarraSighting) {
   501→      this.ibarraAppeared = true;
   502→      // Show Ibarra NPC
   503→      const ibaraEntity = this.entities.get('ibara');
   504→      if (ibaraEntity) {
   505→        ibaraEntity.visible = true;
   506→      }
   507→    }
   508→
   509→    gameEvents.emit('dialogue:line', {
   510→      dialogueId: this.currentDialogueId,
   511→      line,
   512→      lineIndex: this.currentDialogueLineIndex,
   513→      totalLines: dialogue.lines.length,
   514→    });
   515→  }

   517→  private _endDialogue(dialogue: { unlocksObjective?: string | null; id: string }): void {
   518→    this.dialogueActive = false;
   519→    this.currentDialogueId = null;
   520→    this.currentDialogueLineIndex = 0;
   521→
   522→    // Update chapter phase based on dialogue completion
   523→    if (dialogue.id === 'intro-text') {
   524→      this.chapterPhase = 'explore';
   525→      saveManager.updateGameState({ chapterPhase: 'explore' });
   526→    }
   527→
   528→    // Complete objective if dialogue unlocks one
   529→    if (dialogue.unlocksObjective) {
   530→      saveManager.completeObjective(dialogue.unlocksObjective);
   531→      
   532→      // Update entity visibility after objectives
   533→      this._updateEntityVisibility();
   534→
   535→      // Update chapter phase based on quest progress
   536→      if (dialogue.id === 'mang-tenyo-first') {
   537→        this.chapterPhase = 'explore';
   538→        saveManager.updateGameState({ chapterPhase: 'explore' });
   539→      }
   540→      if (dialogue.id === 'market-gossip') {
   541→        this.chapterPhase = 'gossip';
   542→        saveManager.updateGameState({ chapterPhase: 'gossip' });
   543→      }
   544→    }
   545→
   546→    // ===== Progressive Codex unlocks =====
   547→    // Codex entries unlock as the player meets characters and overhears gossip,
   548→    // so the Codex fills up DURING the chapter (not all at once at the end).
   549→    // This gives the player a sense of progression and discovery.
   550→    if (dialogue.id === 'mang-tenyo-first') {
   551→      // After meeting Mang Tenyo: unlock himself + he mentions Ibarra
   552→      saveManager.unlockCodexEntry('char.mang-tenyo');
   553→      saveManager.unlockCodexEntry('char.ibarra');
   554→      saveManager.addJournalEntry(
   555→        'I arrived in San Diego. A cart driver named Mang Tenyo warned me not to mention the name "Ibarra" too loudly — something happened at last night\'s reception.',
   556→        'ch1'
   557→      );
   558→    } else if (dialogue.id === 'market-gossip') {
   559→      // After overhearing gossip: unlock the people mentioned
   560→      saveManager.unlockCodexEntry('char.tiago');
   561→      saveManager.unlockCodexEntry('char.damaso');
   562→      saveManager.unlockCodexEntry('place.san-diego');
   563→      saveManager.addJournalEntry(

   640→  private _gameLoop(timestamp: number): void {
   641→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   642→    this.lastTimestamp = timestamp;
   643→    this.gameTime += deltaTime;

   651→  private _update(dt: number): void {
   652→    if (!this.player) return;
   653→    
   654→    // Update interact cooldown
   655→    if (this.interactCooldown > 0) {
   656→      this.interactCooldown -= dt * 1000;
   657→    }
   658→
   659→    // Don't allow movement during dialogue
   660→    if (this.dialogueActive) return;
   661→
   662→    // Handle intro state
   663→    if (this.chapterPhase === 'intro' && !this.introShown) {
   664→      this.introShown = true;
   665→      this._startDialogue('intro-text');
   666→      return;
   667→    }
   668→
   669→    // Player movement
   670→    let moveX = 0;
   671→    let moveY = 0;
   672→
   673→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   674→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   675→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   676→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   677→
   678→    // Normalize diagonal movement
   679→    if (moveX !== 0 && moveY !== 0) {
   680→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   681→      moveX /= len;
   682→      moveY /= len;
   683→    }
   684→
   685→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   686→
   687→    if (this.player.isMoving) {
   688→      // Determine direction from movement vector
   689→      this.player.direction = this._vectorToDirection(moveX, moveY);
   690→      
   691→      const newX = this.player.x + moveX * this.player.speed * dt;
   692→      const newY = this.player.y + moveY * this.player.speed * dt;
   693→
   694→      // Collision check
   695→      if (!this._isBlocked(newX, newY)) {
   696→        this.player.x = newX;
   697→        this.player.y = newY;
   698→      }
   699→
   700→      // Update animation
   701→      this.player.animationTimer += dt;
   702→      const animConfig = characterData.characters.student.animations.Walk;
   703→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   704→        this.player.animationTimer = 0;
   705→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   706→      }
   707→    } else {
   708→      // Idle animation
   709→      this.player.animationTimer += dt;
   710→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   711→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   712→        this.player.animationTimer = 0;
   713→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   714→      }
   715→    }
   716→
   717→    // Update camera to follow player
   718→    this.cameraX = this.player.x - this.viewWidth / 2;
   719→    this.cameraY = this.player.y - this.viewHeight / 2;
   720→
   721→    // Clamp camera to map bounds
   722→    const mapPixelWidth = mapData.width * this.tileSize;
   723→    const mapPixelHeight = mapData.height * this.tileSize;
   724→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   725→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   726→
   727→    // NPC idle animations
   728→    for (const [npcId, npc] of this.entities) {
   729→      if (npc.isPlayer || !npc.visible) continue;
   730→      npc.animationTimer += dt;
   731→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   732→      if (animConfig) {
   733→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   734→          npc.animationTimer = 0;
   735→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   736→        }
   737→      }
   738→    }
   739→
   740→    // Check trigger zones
   741→    this._checkTriggerZones();
   742→
   743→    // Update particles
   744→    this._updateParticles(dt);
   745→
   746→    // Update burst particles (celebration effect)
   747→    this._updateBurstParticles(dt);
   748→
   749→    // Update game state in SaveManager
   750→    saveManager.updateGameState({
   751→      playerPosition: {
   752→        row: Math.floor(this.player.y / this.tileSize),
   753→        col: Math.floor(this.player.x / this.tileSize),
   754→      },
   755→      playerDirection: this.player.direction,
   756→      currentDialogue: this.currentDialogueId,
   757→      timeOfDay: this.timeOfDay,
   758→      chapterPhase: this.chapterPhase,
   759→    });
   760→  }

   762→  private _vectorToDirection(x: number, y: number): Direction {
   763→    // Map movement vector to 8-direction name
   764→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   765→    
   766→    if (angle >= -22.5 && angle < 22.5) return 'east';
   767→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   768→    if (angle >= 67.5 && angle < 112.5) return 'south';
   769→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   770→    if (angle >= 157.5 || angle < -157.5) return 'west';
   771→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   772→    if (angle >= -112.5 && angle < -67.5) return 'north';
   773→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   774→    return 'south';
   775→  }

   777→  private _isBlocked(x: number, y: number): boolean {
   778→    // Check collision map
   779→    const col = Math.floor(x / this.tileSize);
   780→    const row = Math.floor(y / this.tileSize);
   781→    
   782→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   783→    
   784→    // Check player sprite footprint (check a small area around center)
   785→    const checks = [
   786→      { x: x - 16, y: y - 16 },
   787→      { x: x + 16, y: y - 16 },
   788→      { x: x - 16, y: y + 16 },
   789→      { x: x + 16, y: y + 16 },
   790→    ];
   791→    
   792→    for (const check of checks) {
   793→      const cCol = Math.floor(check.x / this.tileSize);
   794→      const cRow = Math.floor(check.y / this.tileSize);
   795→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   796→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   797→    }
   798→    
   799→    return false;
   800→  }

   802→  private _checkTriggerZones(): void {
   803→    if (!this.player || this.dialogueActive) return;
   804→    
   805→    const playerRow = Math.floor(this.player.y / this.tileSize);
   806→    const playerCol = Math.floor(this.player.x / this.tileSize);
   807→
   808→    for (const zone of this.triggerZones) {
   809→      if (zone.triggered) continue;
   810→
   811→      // Check prerequisite objective
   812→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   813→
   814→      // Check if player is in the zone
   815→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   816→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   817→
   818→      // Or within radius of NPC
   819→      let nearNpc = false;
   820→      if (zone.radius > 0 && zone.npcId) {
   821→        const npc = this.entities.get(zone.npcId);
   822→        if (npc && npc.visible) {
   823→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   824→        }
   825→      }
   826→
   827→      if (inZone || nearNpc) {
   828→        zone.triggered = true;
   829→
   830→        switch (zone.type) {
   831→          case 'npc-interact':
   832→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   833→            break;
   834→          case 'auto-trigger':
   835→            // Generic auto-trigger: if zone has a dialogueId, start that dialogue
   836→            if (zone.dialogueId) {
   837→              // Special handling for Ibarra sighting (time transition)
   838→              if (zone.id === 'ibarra-sighting') {
   839→                this.timeOfDay = 'morning';
   840→                this.chapterPhase = 'ibarra-sighting';
   841→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   842→                this._updateEntityVisibility();
   843→              }
   844→              this._startDialogue(zone.dialogueId);
   845→            } else {
   846→              // Legacy hardcoded triggers as fallback
   847→              if (zone.id === 'market-gossip') {
   848→                this._startDialogue('market-gossip');
   849→              } else if (zone.id === 'ibarra-sighting') {
   850→                this.timeOfDay = 'morning';
   851→                this.chapterPhase = 'ibarra-sighting';
   852→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   853→                this._updateEntityVisibility();
   854→                this._startDialogue('ibarra-sighting');
   855→              }
   856→            }
   857→            break;
   858→        }
   859→      }
   860→    }
   861→  }

   865→  private _initParticles(): void {

   867→    for (let i = 0; i < 15; i++) {

   869→    }

   872→  private _spawnParticle(): void {
   873→    if (!this.player) return;
   874→    // Spawn particle near the camera view area
   875→    const cx = this.cameraX + this.viewWidth / 2;
   876→    const cy = this.cameraY + this.viewHeight / 2;
   877→    this.particles.push({
   878→      x: cx + (Math.random() - 0.5) * this.viewWidth * 0.8,
   879→      y: cy + (Math.random() - 0.5) * this.viewHeight * 0.8,
   880→      vx: (Math.random() - 0.5) * 8, // Slow drift
   881→      vy: -Math.random() * 4 - 1, // Slight upward drift
   882→      alpha: 0,
   883→      size: Math.random() * 2.5 + 1,
   884→      life: 0,
   885→      maxLife: Math.random() * 6 + 4, // 4-10 seconds
   886→    });
   887→  }
   888→
   889→  private _updateParticles(dt: number): void {
   890→    // Spawn new particles periodically
   891→    this.particleSpawnTimer += dt;
   892→    if (this.particleSpawnTimer > 0.3 && this.particles.length < 25) {
   893→      this.particleSpawnTimer = 0;
   894→      this._spawnParticle();
   895→    }
   896→
   897→    // Update existing particles
   898→    for (let i = this.particles.length - 1; i >= 0; i--) {
   899→      const p = this.particles[i];
   900→      p.life += dt;
   901→      p.x += p.vx * dt;
   902→      p.y += p.vy * dt;
   903→      // Gentle oscillation
   904→      p.x += Math.sin(this.gameTime * 0.5 + i) * 0.3;
   905→
   906→      // Fade in then fade out
   907→      const lifeRatio = p.life / p.maxLife;
   908→      if (lifeRatio < 0.2) {
   909→        p.alpha = lifeRatio / 0.2 * 0.4; // Fade in
   910→      } else if (lifeRatio > 0.7) {
   911→        p.alpha = (1 - lifeRatio) / 0.3 * 0.4; // Fade out
   912→      } else {
   913→        p.alpha = 0.4; // Visible
   914→      }
   915→
   916→      // Remove dead particles
   917→      if (p.life >= p.maxLife) {
   918→        this.particles.splice(i, 1);
   919→      }
   920→    }
   921→  }

   988→  // ==================== WAYPOINT ARROW ====================

   990→  /**
   991→   * Render a golden arrow at the screen edge pointing toward the next incomplete objective.
   992→   * Only shown when no dialogue is active and the player isn't standing in the objective zone.
   993→   */
   994→  private _renderWaypointArrow(ctx: CanvasRenderingContext2D): void {
   995→    if (!this.player || this.dialogueActive) return;
   996→    if (this.chapterPhase === 'intro' || this.chapterPhase === 'complete') return;
   997→
   998→    // Find the next incomplete objective
   999→    const objectivesOrder = ['obj.ch1.follow_tenyo', 'obj.ch1.overhear_gossip', 'obj.ch1.see_ibarra'];
  1000→    const nextObjId = objectivesOrder.find(id => !saveManager.isObjectiveCompleted(id));
  1001→    if (!nextObjId) return;
  1002→
  1003→    const loc = OBJECTIVE_LOCATIONS[nextObjId];
  1004→    if (!loc) return;
  1005→
  1006→    const targetX = loc.col * this.tileSize + this.tileSize / 2;
  1007→    const targetY = loc.row * this.tileSize + this.tileSize / 2;
  1008→
  1009→    // Vector from player to target
  1010→    const dx = targetX - this.player.x;
  1011→    const dy = targetY - this.player.y;
  1012→    const dist = Math.sqrt(dx * dx + dy * dy);
  1013→
  1014→    // Don't show arrow if very close
  1015→    if (dist < this.tileSize * 2.5) return;
  1016→
  1017→    // Compute target position on screen (may be off-screen)
  1018→    const screenTargetX = targetX - this.cameraX;
  1019→    const screenTargetY = targetY - this.cameraY;
  1020→
  1021→    const cx = this.viewWidth / 2;
  1022→    const cy = this.viewHeight / 2;
  1023→
  1024→    // If the target is on screen, don't show the edge arrow (the indicator is enough)
  1025→    const onScreen = screenTargetX > 40 && screenTargetX < this.viewWidth - 40 &&
  1026→                     screenTargetY > 80 && screenTargetY < this.viewHeight - 80;
  1027→    if (onScreen) return;
  1028→
  1029→    // Direction from center of screen to target
  1030→    const angle = Math.atan2(dy, dx);
  1031→
  1032→    // Place arrow at a fixed radius from screen center, clamped to a margin inside the screen
  1033→    const margin = 80;
  1034→    const maxRadiusX = this.viewWidth / 2 - margin;
  1035→    const maxRadiusY = this.viewHeight / 2 - margin;
  1036→    // Project the direction onto the screen-edge ellipse
  1037→    // Solve for t such that (t*cos(angle), t*sin(angle)) lies on the ellipse (maxRX, maxRY)
  1038→    const cosA = Math.cos(angle);
  1039→    const sinA = Math.sin(angle);
  1040→    const t = 1 / Math.sqrt((cosA * cosA) / (maxRadiusX * maxRadiusX) + (sinA * sinA) / (maxRadiusY * maxRadiusY));
  1041→    const arrowX = cx + cosA * t;
  1042→    const arrowY = cy + sinA * t;
  1043→
  1044→    // Pulsing effect
  1045→    const pulse = Math.sin(this.gameTime * 4) * 0.15 + 0.85;
  1046→
  1047→    ctx.save();
  1048→    ctx.translate(arrowX, arrowY);
  1049→    ctx.rotate(angle);
  1050→
  1051→    // Outer glow
  1052→    ctx.fillStyle = `rgba(255,215,0,${0.15 * pulse})`;
  1053→    ctx.beginPath();
  1054→    ctx.arc(0, 0, 28, 0, Math.PI * 2);
  1055→    ctx.fill();
  1056→
  1057→    // Arrow shadow
  1058→    ctx.fillStyle = 'rgba(0,0,0,0.5)';
  1059→    ctx.beginPath();
  1060→    ctx.moveTo(18, 0);
  1061→    ctx.lineTo(-8, -12);
  1062→    ctx.lineTo(-4, 0);
  1063→    ctx.lineTo(-8, 12);
  1064→    ctx.closePath();
  1065→    ctx.fill();
  1066→
  1067→    // Arrow body
  1068→    ctx.fillStyle = `rgba(255,215,0,${pulse})`;
  1069→    ctx.strokeStyle = 'rgba(180,120,0,1)';
  1070→    ctx.lineWidth = 1.5;
  1071→    ctx.beginPath();
  1072→    ctx.moveTo(15, 0);
  1073→    ctx.lineTo(-10, -10);
  1074→    ctx.lineTo(-6, 0);
  1075→    ctx.lineTo(-10, 10);
  1076→    ctx.closePath();
  1077→    ctx.fill();
  1078→    ctx.stroke();
  1079→
  1080→    ctx.restore();
  1081→
  1082→    // Distance label below arrow
  1083→    const distTiles = Math.round(dist / this.tileSize);
  1084→    ctx.font = 'bold 10px "Geist", sans-serif';
  1085→    ctx.fillStyle = 'rgba(0,0,0,0.6)';
  1086→    ctx.fillRect(arrowX - 22, arrowY + 14, 44, 14);
  1087→    ctx.fillStyle = '#FFD700';
  1088→    ctx.textAlign = 'center';
  1089→    ctx.fillText(`${distTiles}m →`, arrowX, arrowY + 24);
  1090→  }

  1118→  private _renderMap(ctx: CanvasRenderingContext2D): void {
  1119→    // Only render visible tiles (camera culling)
  1120→    const startCol = Math.floor(this.cameraX / this.tileSize);
  1121→    const startRow = Math.floor(this.cameraY / this.tileSize);
  1122→    const endCol = Math.min(mapData.width, startCol + Math.ceil(this.viewWidth / this.tileSize) + 1);
  1123→    const endRow = Math.min(mapData.height, startRow + Math.ceil(this.viewHeight / this.tileSize) + 1);
  1124→
  1125→    // Draw ground layer with detailed patterns
  1126→    for (let row = startRow; row < endRow; row++) {
  1127→      for (let col = startCol; col < endCol; col++) {
  1128→        if (row < 0 || col < 0) continue;
  1129→        const tileType = mapData.layers.ground[row]?.[col];
  1130→        if (tileType && tileType !== 0) {
  1131→          const x = col * this.tileSize - this.cameraX;
  1132→          const y = row * this.tileSize - this.cameraY;
  1133→          
  1134→          switch (tileType) {
  1135→            case 1: // Dirt/stone (ochre/tan) - building interior floor
  1136→              this._renderDirtFloor(ctx, x, y, row, col);
  1137→              break;
  1138→            case 2: // Grass (green) - outskirts/wild areas
  1139→              this._renderGrass(ctx, x, y, row, col);
  1140→              break;
  1141→            case 3: // Plaza (tan/dirt) - walkable area
  1142→              this._renderPlazaCobblestone(ctx, x, y, row, col);
  1143→              break;
  1144→            case 4: // Market stalls (brown)
  1145→              this._renderMarketStalls(ctx, x, y, row, col);
  1146→              break;
  1147→            default:
  1148→              const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
  1149→              ctx.fillStyle = color;
  1150→              ctx.fillRect(x, y, this.tileSize, this.tileSize);
  1151→          }
  1152→        }
  1153→      }
  1154→    }
  1155→
  1156→    // Draw buildings layer (flat base color - details rendered separately)
  1157→    for (let row = startRow; row < endRow; row++) {
  1158→      for (let col = startCol; col < endCol; col++) {
  1159→        if (row < 0 || col < 0) continue;
  1160→        const tileType = mapData.layers.buildings[row]?.[col];
  1161→        if (tileType && tileType !== 0) {
  1162→          const x = col * this.tileSize - this.cameraX;
  1163→          const y = row * this.tileSize - this.cameraY;
  1164→
  1165→          // Base building fill
  1166→          const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
  1167→          ctx.fillStyle = color;
  1168→          ctx.fillRect(x, y, this.tileSize, this.tileSize);
  1169→        }
  1170→      }
  1171→    }
  1172→  }
