    80→// Deterministic hash for tile decoration patterns
    81→function tileHash(row: number, col: number, seed: number = 0): number {
    82→  let h = seed + row * 374761393 + col * 668265263;
    83→  h = (h ^ (h >> 13)) * 1274126177;
    84→  h = h ^ (h >> 16);
    85→  return (h & 0x7FFFFFFF) / 0x7FFFFFFF; // Returns 0..1
    86→}

   138→  private _computeBuildingGroups(): void {
   139→    // For each building tile type, find contiguous rectangular groups
   140→    const buildingTypes = [5, 6]; // Church and Market building types
   141→    for (const tileType of buildingTypes) {
   142→      const visited: Set<string> = new Set();
   143→      const groups: { minRow: number; minCol: number; maxRow: number; maxCol: number }[] = [];
   144→      
   145→      for (let row = 0; row < mapData.height; row++) {
   146→        for (let col = 0; col < mapData.width; col++) {
   147→          const key = `${row},${col}`;
   148→          if (visited.has(key)) continue;
   149→          if (mapData.layers.buildings[row]?.[col] !== tileType) continue;
   150→          
   151→          // Find bounding box of contiguous group
   152→          let minRow = row, maxRow = row, minCol = col, maxCol = col;
   153→          const toVisit: [number, number][] = [[row, col]];
   154→          
   155→          while (toVisit.length > 0) {
   156→            const [r, c] = toVisit.pop()!;
   157→            const k = `${r},${c}`;
   158→            if (visited.has(k)) continue;
   159→            if (mapData.layers.buildings[r]?.[c] !== tileType) continue;
   160→            visited.add(k);
   161→            minRow = Math.min(minRow, r);
   162→            maxRow = Math.max(maxRow, r);
   163→            minCol = Math.min(minCol, c);
   164→            maxCol = Math.max(maxCol, c);
   165→            toVisit.push([r - 1, c], [r + 1, c], [r, c - 1], [r, c + 1]);
   166→          }
   167→          
   168→          groups.push({ minRow, minCol, maxRow, maxCol });
   169→        }
   170→      }
   171→      
   172→      this.buildingGroups.set(tileType, groups);
   173→    }
   174→  }

   176→  async init(canvas: HTMLCanvasElement): Promise<void> {

   189→
   190→    // Create player entity
   191→    const spawn = mapData.playerSpawn;
   192→    this.player = {
   193→      id: 'player',
   194→      x: spawn.col * this.tileSize + this.tileSize / 2,
   195→      y: spawn.row * this.tileSize + this.tileSize / 2,

   204→      dialogueTriggered: false,

   207→
   208→    // Create NPC entities
   209→    const npcPositions = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   210→    for (const [npcId, pos] of Object.entries(npcPositions)) {
   211→      const entity: Entity = {
   212→        id: npcId,
   213→        x: pos.col * this.tileSize + this.tileSize / 2,
   214→        y: pos.row * this.tileSize + this.tileSize / 2,
   215→        direction: pos.direction as Direction,
   216→        characterKey: npcId,
   217→        isMoving: false,
   218→        animationFrame: 0,
   219→        animationTimer: 0,
   220→        speed: 0,
   221→        visible: !pos.appearsAfter || saveManager.isObjectiveCompleted(pos.appearsAfter),
   222→        isPlayer: false,
   223→        dialogueTriggered: false,
   224→        appearsAfterObjective: pos.appearsAfter,
   225→      };
   226→      this.entities.set(npcId, entity);
   227→    }
   228→
   229→    // Load saved progress
   230→    await saveManager.loadProgress();
   231→    const gameState = saveManager.getGameState();
   232→    this.chapterPhase = gameState.chapterPhase;
   233→    this.timeOfDay = gameState.timeOfDay;
   234→    
   235→    if (this.chapterPhase !== 'intro') {
   236→      this.introShown = true;
   237→    }
   238→
   239→    // Update entity visibility based on completed objectives
   240→    this._updateEntityVisibility();
   241→
   242→    // Mark already-triggered zones
   243→    this.triggerZones.forEach(zone => {
   244→      if (zone.requiresObjective && saveManager.isObjectiveCompleted(zone.requiresObjective)) {
   245→        // The prerequisite is done, but this zone itself may not have been triggered yet
   246→        // Only mark as triggered if the objective THIS zone completes is also done
   247→      }
   248→    });
   249→
   250→    // Bind input handlers
   251→    this._bindInput();
   252→
   253→    // Start auto-save
   254→    saveManager.startAutoSave();
   255→
   256→    // Emit game ready
   257→    gameEvents.emit('game:ready');
   258→
   259→    // Initialize some particles
   260→    this._initParticles();
   261→
   262→    // Start game loop
   263→    this.lastTimestamp = performance.now();
   264→    this._gameLoop(this.lastTimestamp);
   265→  }

   316→  private _handleInteract(): void {
   317→    if (this.dialogueActive) return;
   318→    if (this.interactCooldown > 0) return;
   319→    
   320→    // Check if player is near an NPC they can talk to
   321→    if (!this.player) return;
   322→
   323→    for (const [npcId, npc] of this.entities) {
   324→      if (npc.isPlayer || !npc.visible) continue;
   325→      
   326→      const dist = this._distanceBetween(this.player, npc);
   327→      if (dist < this.tileSize * 2) {
   328→        this.interactCooldown = 500;
   329→        
   330→        // Determine which dialogue to trigger
   331→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   332→          this._startDialogue('mang-tenyo-first');
   333→          return;
   334→        }
   335→        // Mang Tenyo can be talked to again but won't repeat the quest dialogue
   336→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   337→          this._startDialogue('mang-tenyo-repeat');
   338→          return;
   339→        }
   340→      }
   341→    }
   342→  }

   344→  private _distanceBetween(a: Entity, b: Entity): number {
   345→    const dx = a.x - b.x;
   346→    const dy = a.y - b.y;
   347→    return Math.sqrt(dx * dx + dy * dy);
   348→  }

   350→  private _startDialogue(dialogueId: string): void {
   351→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   352→    if (!dialogue) {
   353→      console.warn(`Dialogue not found: ${dialogueId}`);
   354→      return;
   355→    }
   356→
   357→    this.dialogueActive = true;
   358→    this.currentDialogueId = dialogueId;
   359→    this.currentDialogueLineIndex = 0;
   360→
   361→    gameEvents.emit('dialogue:start', {
   362→      dialogueId,
   363→      line: dialogue.lines[0],
   364→      lineIndex: 0,
   365→      totalLines: dialogue.lines.length,
   366→    });
   367→  }
   368→
   369→  advanceDialogue(): void {
   370→    if (!this.currentDialogueId) return;
   371→    
   372→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];
   373→    if (!dialogue) return;
   374→
   375→    this.currentDialogueLineIndex++;
   376→
   377→    if (this.currentDialogueLineIndex >= dialogue.lines.length) {
   378→      // Dialogue complete
   379→      this._endDialogue(dialogue);
   380→      return;
   381→    }
   382→
   383→    const line = dialogue.lines[this.currentDialogueLineIndex];
   384→    
   385→    // Check for time transition
   386→    if (line.isTimeTransition) {
   387→      this.timeOfDay = 'morning';
   388→      this.chapterPhase = 'ibarra-sighting';
   389→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   390→      gameEvents.emit('time:transition', 'morning');
   391→    }
   392→
   393→    // Check for Ibarra sighting
   394→    if (line.isIbarraSighting) {
   395→      this.ibarraAppeared = true;
   396→      // Show Ibarra NPC
   397→      const ibaraEntity = this.entities.get('ibara');
   398→      if (ibaraEntity) {
   399→        ibaraEntity.visible = true;
   400→      }
   401→    }
   402→
   403→    gameEvents.emit('dialogue:line', {
   404→      dialogueId: this.currentDialogueId,
   405→      line,
   406→      lineIndex: this.currentDialogueLineIndex,
   407→      totalLines: dialogue.lines.length,
   408→    });
   409→  }

   411→  private _endDialogue(dialogue: { unlocksObjective?: string | null; id: string }): void {
   412→    this.dialogueActive = false;
   413→    this.currentDialogueId = null;
   414→    this.currentDialogueLineIndex = 0;
   415→
   416→    // Update chapter phase based on dialogue completion
   417→    if (dialogue.id === 'intro-text') {
   418→      this.chapterPhase = 'explore';
   419→      saveManager.updateGameState({ chapterPhase: 'explore' });
   420→    }
   421→
   422→    // Complete objective if dialogue unlocks one
   423→    if (dialogue.unlocksObjective) {
   424→      saveManager.completeObjective(dialogue.unlocksObjective);
   425→      
   426→      // Update entity visibility after objectives
   427→      this._updateEntityVisibility();
   428→
   429→      // Update chapter phase based on quest progress
   430→      if (dialogue.id === 'mang-tenyo-first') {
   431→        this.chapterPhase = 'explore';
   432→        saveManager.updateGameState({ chapterPhase: 'explore' });
   433→      }
   434→      if (dialogue.id === 'market-gossip') {
   435→        this.chapterPhase = 'gossip';
   436→        saveManager.updateGameState({ chapterPhase: 'gossip' });
   437→      }
   438→    }
   439→
   440→    gameEvents.emit('dialogue:end', dialogue.id);
   441→
   442→    // Check if this was the gossip dialogue - enable Ibarra sighting zone
   443→    if (dialogue.id === 'market-gossip') {
   444→      // The market gossip zone should now be triggered
   445→      this.triggerZones.forEach(z => {
   446→        if (z.id === 'market-gossip') z.triggered = true;
   447→      });
   448→    }
   449→
   450→    // Check if chapter is complete after Ibarra sighting dialogue
   451→    if (dialogue.id === 'ibarra-sighting') {
   452→      this._triggerChapterComplete();
   453→    }
   454→  }

   456→  private _updateEntityVisibility(): void {
   457→    for (const [npcId, entity] of this.entities) {
   458→      if (entity.appearsAfterObjective) {
   459→        entity.visible = saveManager.isObjectiveCompleted(entity.appearsAfterObjective);
   460→      }
   461→    }
   462→    
   463→    // Special: Ibarra appears after gossip objective AND time transition
   464→    const ibara = this.entities.get('ibara');
   465→    if (ibara && this.timeOfDay === 'morning' && saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   466→      ibara.visible = true;
   467→    }
   468→  }

   502→  private _gameLoop(timestamp: number): void {
   503→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   504→    this.lastTimestamp = timestamp;
   505→    this.gameTime += deltaTime;
   506→
   507→    this._update(deltaTime);
   508→    this._render();
   509→
   510→    this.animationFrameId = requestAnimationFrame(this._gameLoop.bind(this));
   511→  }
   512→
   513→  private _update(dt: number): void {
   514→    if (!this.player) return;
   515→    
   516→    // Update interact cooldown
   517→    if (this.interactCooldown > 0) {
   518→      this.interactCooldown -= dt * 1000;
   519→    }
   520→
   521→    // Don't allow movement during dialogue
   522→    if (this.dialogueActive) return;
   523→
   524→    // Handle intro state
   525→    if (this.chapterPhase === 'intro' && !this.introShown) {
   526→      this.introShown = true;
   527→      this._startDialogue('intro-text');
   528→      return;
   529→    }
   530→
   531→    // Player movement
   532→    let moveX = 0;
   533→    let moveY = 0;
   534→
   535→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   536→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   537→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   538→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   539→
   540→    // Normalize diagonal movement
   541→    if (moveX !== 0 && moveY !== 0) {
   542→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   543→      moveX /= len;
   544→      moveY /= len;
   545→    }
   546→
   547→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   548→
   549→    if (this.player.isMoving) {
   550→      // Determine direction from movement vector
   551→      this.player.direction = this._vectorToDirection(moveX, moveY);
   552→      
   553→      const newX = this.player.x + moveX * this.player.speed * dt;
   554→      const newY = this.player.y + moveY * this.player.speed * dt;
   555→
   556→      // Collision check
   557→      if (!this._isBlocked(newX, newY)) {
   558→        this.player.x = newX;
   559→        this.player.y = newY;
   560→      }
   561→
   562→      // Update animation
   563→      this.player.animationTimer += dt;
   564→      const animConfig = characterData.characters.student.animations.Walk;
   565→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   566→        this.player.animationTimer = 0;
   567→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   568→      }
   569→    } else {
   570→      // Idle animation
   571→      this.player.animationTimer += dt;
   572→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   573→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   574→        this.player.animationTimer = 0;
   575→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   576→      }
   577→    }
   578→
   579→    // Update camera to follow player
   580→    this.cameraX = this.player.x - this.viewWidth / 2;
   581→    this.cameraY = this.player.y - this.viewHeight / 2;
   582→
   583→    // Clamp camera to map bounds
   584→    const mapPixelWidth = mapData.width * this.tileSize;
   585→    const mapPixelHeight = mapData.height * this.tileSize;
   586→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   587→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   588→
   589→    // NPC idle animations
   590→    for (const [npcId, npc] of this.entities) {
   591→      if (npc.isPlayer || !npc.visible) continue;
   592→      npc.animationTimer += dt;
   593→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   594→      if (animConfig) {
   595→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   596→          npc.animationTimer = 0;
   597→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   598→        }
   599→      }
   600→    }
   601→
   602→    // Check trigger zones
   603→    this._checkTriggerZones();
   604→
   605→    // Update particles
   606→    this._updateParticles(dt);
   607→
   608→    // Update game state in SaveManager
   609→    saveManager.updateGameState({
   610→      playerPosition: {
   611→        row: Math.floor(this.player.y / this.tileSize),
   612→        col: Math.floor(this.player.x / this.tileSize),
   613→      },
   614→      playerDirection: this.player.direction,
   615→      currentDialogue: this.currentDialogueId,
   616→      timeOfDay: this.timeOfDay,
   617→      chapterPhase: this.chapterPhase,
   618→    });
   619→  }

   621→  private _vectorToDirection(x: number, y: number): Direction {
   622→    // Map movement vector to 8-direction name
   623→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   624→    
   625→    if (angle >= -22.5 && angle < 22.5) return 'east';
   626→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   627→    if (angle >= 67.5 && angle < 112.5) return 'south';
   628→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   629→    if (angle >= 157.5 || angle < -157.5) return 'west';
   630→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   631→    if (angle >= -112.5 && angle < -67.5) return 'north';
   632→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   633→    return 'south';
   634→  }

   636→  private _isBlocked(x: number, y: number): boolean {
   637→    // Check collision map
   638→    const col = Math.floor(x / this.tileSize);
   639→    const row = Math.floor(y / this.tileSize);
   640→    
   641→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   642→    
   643→    // Check player sprite footprint (check a small area around center)
   644→    const checks = [
   645→      { x: x - 16, y: y - 16 },
   646→      { x: x + 16, y: y - 16 },
   647→      { x: x - 16, y: y + 16 },
   648→      { x: x + 16, y: y + 16 },
   649→    ];
   650→    
   651→    for (const check of checks) {
   652→      const cCol = Math.floor(check.x / this.tileSize);
   653→      const cRow = Math.floor(check.y / this.tileSize);
   654→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   655→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   656→    }
   657→    
   658→    return false;
   659→  }

   661→  private _checkTriggerZones(): void {
   662→    if (!this.player || this.dialogueActive) return;
   663→    
   664→    const playerRow = Math.floor(this.player.y / this.tileSize);
   665→    const playerCol = Math.floor(this.player.x / this.tileSize);
   666→
   667→    for (const zone of this.triggerZones) {
   668→      if (zone.triggered) continue;
   669→
   670→      // Check prerequisite objective
   671→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   672→
   673→      // Check if player is in the zone
   674→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   675→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   676→
   677→      // Or within radius of NPC
   678→      let nearNpc = false;
   679→      if (zone.radius > 0 && zone.npcId) {
   680→        const npc = this.entities.get(zone.npcId);
   681→        if (npc && npc.visible) {
   682→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   683→        }
   684→      }
   685→
   686→      if (inZone || nearNpc) {
   687→        zone.triggered = true;
   688→
   689→        switch (zone.type) {
   690→          case 'npc-interact':
   691→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   692→            break;
   693→          case 'auto-trigger':
   694→            // Generic auto-trigger: if zone has a dialogueId, start that dialogue
   695→            if (zone.dialogueId) {
   696→              // Special handling for Ibarra sighting (time transition)
   697→              if (zone.id === 'ibarra-sighting') {
   698→                this.timeOfDay = 'morning';
   699→                this.chapterPhase = 'ibarra-sighting';
   700→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   701→                this._updateEntityVisibility();
   702→              }
   703→              this._startDialogue(zone.dialogueId);
   704→            } else {
   705→              // Legacy hardcoded triggers as fallback
   706→              if (zone.id === 'market-gossip') {
   707→                this._startDialogue('market-gossip');
   708→              } else if (zone.id === 'ibarra-sighting') {
   709→                this.timeOfDay = 'morning';
   710→                this.chapterPhase = 'ibarra-sighting';
   711→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   712→                this._updateEntityVisibility();
   713→                this._startDialogue('ibarra-sighting');
   714→              }
   715→            }
   716→            break;
   717→        }
   718→      }
   719→    }
   720→  }

   724→  private _initParticles(): void {

   731→  private _spawnParticle(): void {
   732→    if (!this.player) return;
   733→    // Spawn particle near the camera view area
   734→    const cx = this.cameraX + this.viewWidth / 2;
   735→    const cy = this.cameraY + this.viewHeight / 2;
   736→    this.particles.push({
   737→      x: cx + (Math.random() - 0.5) * this.viewWidth * 0.8,
   738→      y: cy + (Math.random() - 0.5) * this.viewHeight * 0.8,
   739→      vx: (Math.random() - 0.5) * 8, // Slow drift
   740→      vy: -Math.random() * 4 - 1, // Slight upward drift
   741→      alpha: 0,
   742→      size: Math.random() * 2.5 + 1,
   743→      life: 0,
   744→      maxLife: Math.random() * 6 + 4, // 4-10 seconds
   745→    });
   746→  }
   747→
   748→  private _updateParticles(dt: number): void {
   749→    // Spawn new particles periodically
   750→    this.particleSpawnTimer += dt;
   751→    if (this.particleSpawnTimer > 0.3 && this.particles.length < 25) {
   752→      this.particleSpawnTimer = 0;
   753→      this._spawnParticle();
   754→    }
   755→
   756→    // Update existing particles
   757→    for (let i = this.particles.length - 1; i >= 0; i--) {
   758→      const p = this.particles[i];
   759→      p.life += dt;
   760→      p.x += p.vx * dt;
   761→      p.y += p.vy * dt;
   762→      // Gentle oscillation
   763→      p.x += Math.sin(this.gameTime * 0.5 + i) * 0.3;
   764→
   765→      // Fade in then fade out
   766→      const lifeRatio = p.life / p.maxLife;
   767→      if (lifeRatio < 0.2) {
   768→        p.alpha = lifeRatio / 0.2 * 0.4; // Fade in
   769→      } else if (lifeRatio > 0.7) {
   770→        p.alpha = (1 - lifeRatio) / 0.3 * 0.4; // Fade out
   771→      } else {
   772→        p.alpha = 0.4; // Visible
   773→      }
   774→
   775→      // Remove dead particles
   776→      if (p.life >= p.maxLife) {
   777→        this.particles.splice(i, 1);
   778→      }
   779→    }
   780→  }

   857→  private _renderMap(ctx: CanvasRenderingContext2D): void {
   858→    // Only render visible tiles (camera culling)
   859→    const startCol = Math.floor(this.cameraX / this.tileSize);
   860→    const startRow = Math.floor(this.cameraY / this.tileSize);
   861→    const endCol = Math.min(mapData.width, startCol + Math.ceil(this.viewWidth / this.tileSize) + 1);
   862→    const endRow = Math.min(mapData.height, startRow + Math.ceil(this.viewHeight / this.tileSize) + 1);
   863→
   864→    // Draw ground layer with detailed patterns
   865→    for (let row = startRow; row < endRow; row++) {
   866→      for (let col = startCol; col < endCol; col++) {
   867→        if (row < 0 || col < 0) continue;
   868→        const tileType = mapData.layers.ground[row]?.[col];
   869→        if (tileType && tileType !== 0) {
   870→          const x = col * this.tileSize - this.cameraX;
   871→          const y = row * this.tileSize - this.cameraY;
   872→          
   873→          switch (tileType) {
   874→            case 1: // Dirt/stone (ochre/tan) - building interior floor
   875→              this._renderDirtFloor(ctx, x, y, row, col);
   876→              break;
   877→            case 2: // Grass (green) - outskirts/wild areas
   878→              this._renderGrass(ctx, x, y, row, col);
   879→              break;
   880→            case 3: // Plaza (tan/dirt) - walkable area
   881→              this._renderPlazaCobblestone(ctx, x, y, row, col);
   882→              break;
   883→            case 4: // Market stalls (brown)
   884→              this._renderMarketStalls(ctx, x, y, row, col);
   885→              break;
   886→            default:
   887→              const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
   888→              ctx.fillStyle = color;
   889→              ctx.fillRect(x, y, this.tileSize, this.tileSize);
   890→          }
   891→        }
   892→      }
   893→    }
   894→
   895→    // Draw buildings layer (flat base color - details rendered separately)
   896→    for (let row = startRow; row < endRow; row++) {
   897→      for (let col = startCol; col < endCol; col++) {
   898→        if (row < 0 || col < 0) continue;
   899→        const tileType = mapData.layers.buildings[row]?.[col];
   900→        if (tileType && tileType !== 0) {
   901→          const x = col * this.tileSize - this.cameraX;
   902→          const y = row * this.tileSize - this.cameraY;
   903→
   904→          // Base building fill
   905→          const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
   906→          ctx.fillStyle = color;
   907→          ctx.fillRect(x, y, this.tileSize, this.tileSize);
   908→        }
   909→      }
   910→    }
   911→  }

   915→  private _renderGrass(ctx: CanvasRenderingContext2D, x: number, y: number, row: number, col: number): void {
   916→    const ts = this.tileSize;
   917→    const h = tileHash(row, col, 42);
   918→    
   919→    // Base grass color with slight variation per tile
   920→    const baseGreen = h < 0.5 ? '#7BAF6E' : '#6B9E5E';
   921→    ctx.fillStyle = baseGreen;
   922→    ctx.fillRect(x, y, ts, ts);
   923→    
   924→    // Second shade patches for depth variation
   925→    const h2 = tileHash(row, col, 99);
   926→    ctx.fillStyle = h2 < 0.3 ? '#5C8B4E' : h2 < 0.6 ? '#8BC47B' : '#6B9E5E';
   927→    // Random patch placement
   928→    const patchX = x + tileHash(row, col, 7) * (ts - 16);
   929→    const patchY = y + tileHash(row, col, 13) * (ts - 16);
   930→    ctx.fillRect(patchX, patchY, 16, 16);
   931→    
   932→    // Grass blade clusters (multiple small blades)
   933→    ctx.fillStyle = '#8FD47F';
   934→    const bladeCount = 3 + Math.floor(tileHash(row, col, 23) * 4);
   935→    for (let i = 0; i < bladeCount; i++) {
   936→      const bx = x + tileHash(row, col, i * 7 + 1) * (ts - 6);
   937→      const by = y + tileHash(row, col, i * 11 + 3) * (ts - 10);
   938→      const bladeH = 5 + tileHash(row, col, i * 3 + 5) * 7;
   939→      ctx.fillRect(bx, by - bladeH, 1.5, bladeH);
   940→      // Second blade slightly offset
   941→      ctx.fillRect(bx + 2, by - bladeH * 0.7, 1, bladeH * 0.7);
   942→    }
   943→    
   944→    // Darker grass blades for depth
   945→    ctx.fillStyle = '#4E7A3E';
   946→    for (let i = 0; i < 2; i++) {
   947→      const bx = x + tileHash(row, col, i * 17 + 50) * (ts - 6);
   948→      const by = y + tileHash(row, col, i * 19 + 60) * (ts - 8);
   949→      ctx.fillRect(bx, by - 5, 1.5, 5);

   951→

   953→    const flowerChance = tileHash(row, col, 77);
   954→    if (flowerChance < 0.15) {
   955→      // Small wildflower
   956→      const flowerX = x + tileHash(row, col, 88) * (ts - 8) + 4;

   960→      ctx.beginPath();
   961→      ctx.arc(flowerX, flowerY, 2.5, 0, Math.PI * 2);
   962→      ctx.fill();
   963→      // Flower center
   964→      ctx.fillStyle = '#FFE066';
   965→      ctx.beginPath();
   966→      ctx.arc(flowerX, flowerY, 1, 0, Math.PI * 2);
   967→      ctx.fill();
   968→    }
   969→    
   970→    // Very subtle, near-invisible grid line (for alignment only)
   971→    ctx.strokeStyle = 'rgba(0,0,0,0.02)';
   972→    ctx.lineWidth = 0.25;
   973→    ctx.strokeRect(x, y, ts, ts);
   974→  }

   976→  private _renderPlazaCobblestone(ctx: CanvasRenderingContext2D, x: number, y: number, row: number, col: number): void {
   977→    const ts = this.tileSize;
   978→    
   979→    // Base tan/dirt color
   980→    ctx.fillStyle = '#C4A76C';
   981→    ctx.fillRect(x, y, ts, ts);
   982→    
   983→    // Warm dirt undertone
   984→    ctx.fillStyle = '#B89858';
   985→    const patchH = tileHash(row, col, 33);
   986→    if (patchH < 0.4) {
   987→      ctx.fillRect(x + 8, y + 8, ts - 16, ts - 16);
   988→    }
   989→    
   990→    // Cobblestone pattern - individual stones
   991→    const stoneW = 14;
   992→    const stoneH = 10;
   993→    const gap = 2;
   994→    
   995→    for (let sy = 0; sy < ts; sy += stoneH + gap) {
   996→      const offset = ((sy / (stoneH + gap)) % 2 === 0) ? 0 : stoneW / 2; // Brick-like offset
   997→      for (let sx = offset; sx < ts; sx += stoneW + gap) {
   998→        const sh = tileHash(row, col, sy * 100 + sx);
   999→        // Stone color variations
  1000→        const stoneColors = ['#D2B48C', '#C4A76C', '#B89560', '#A88950', '#C9AE7A'];
  1001→        ctx.fillStyle = stoneColors[Math.floor(sh * stoneColors.length)];
  1002→        
  1003→        // Draw rounded stone
  1004→        const sw = Math.min(stoneW, ts - sx);
  1005→        const sht = Math.min(stoneH, ts - sy);
  1006→        if (sw > 0 && sht > 0) {
  1007→          ctx.beginPath();
  1008→          const r = 2;
  1009→          ctx.roundRect(x + sx, y + sy, sw, sht, r);
  1010→          ctx.fill();
  1011→          
  1012→          // Subtle stone edge shadow
  1013→          ctx.fillStyle = 'rgba(0,0,0,0.06)';
  1014→          ctx.fillRect(x + sx, y + sy + sht - 1.5, sw, 1.5);
  1015→          
  1016→          // Stone highlight on top
  1017→          ctx.fillStyle = 'rgba(255,255,255,0.04)';
  1018→          ctx.fillRect(x + sx, y + sy, sw, 2);
  1019→        }
  1020→      }
  1021→    }
  1022→    
  1023→    // Dirt/gravel fill between stones
  1024→    ctx.fillStyle = '#A88950';
  1025→    for (let i = 0; i < 3; i++) {
  1026→      const gx = x + tileHash(row, col, i * 29 + 100) * ts;
  1027→      const gy = y + tileHash(row, col, i * 31 + 110) * ts;
  1028→      ctx.fillRect(gx, gy, 2, 2);
  1029→    }
  1030→    
  1031→    // Very subtle grid line
  1032→    ctx.strokeStyle = 'rgba(0,0,0,0.02)';
  1033→    ctx.lineWidth = 0.25;
  1034→    ctx.strokeRect(x, y, ts, ts);
  1035→  }
