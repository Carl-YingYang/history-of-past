   127→// Deterministic hash for tile decoration patterns
   128→function tileHash(row: number, col: number, seed: number = 0): number {
   129→  let h = seed + row * 374761393 + col * 668265263;
   130→  h = (h ^ (h >> 13)) * 1274126177;
   131→  h = h ^ (h >> 16);
   132→  return (h & 0x7FFFFFFF) / 0x7FFFFFFF; // Returns 0..1
   133→}

   203→  private _computeBuildingGroups(): void {
   204→    // For each building tile type, find contiguous rectangular groups
   205→    const buildingTypes = [5, 6]; // Church and Market building types
   206→    for (const tileType of buildingTypes) {
   207→      const visited: Set<string> = new Set();
   208→      const groups: { minRow: number; minCol: number; maxRow: number; maxCol: number }[] = [];
   209→      
   210→      for (let row = 0; row < mapData.height; row++) {
   211→        for (let col = 0; col < mapData.width; col++) {
   212→          const key = `${row},${col}`;
   213→          if (visited.has(key)) continue;
   214→          if (mapData.layers.buildings[row]?.[col] !== tileType) continue;
   215→          
   216→          // Find bounding box of contiguous group
   217→          let minRow = row, maxRow = row, minCol = col, maxCol = col;
   218→          const toVisit: [number, number][] = [[row, col]];
   219→          
   220→          while (toVisit.length > 0) {
   221→            const [r, c] = toVisit.pop()!;
   222→            const k = `${r},${c}`;
   223→            if (visited.has(k)) continue;
   224→            if (mapData.layers.buildings[r]?.[c] !== tileType) continue;
   225→            visited.add(k);
   226→            minRow = Math.min(minRow, r);
   227→            maxRow = Math.max(maxRow, r);
   228→            minCol = Math.min(minCol, c);
   229→            maxCol = Math.max(maxCol, c);
   230→            toVisit.push([r - 1, c], [r + 1, c], [r, c - 1], [r, c + 1]);
   231→          }
   232→          
   233→          groups.push({ minRow, minCol, maxRow, maxCol });
   234→        }
   235→      }
   236→      
   237→      this.buildingGroups.set(tileType, groups);
   238→    }
   239→  }

   259→      x: spawn.col * this.tileSize + this.tileSize / 2,

   269→      dialogueTriggered: false,

   272→
   273→    // Create NPC entities
   274→    const npcPositions = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   275→    for (const [npcId, pos] of Object.entries(npcPositions)) {
   276→      const entity: Entity = {
   277→        id: npcId,
   278→        x: pos.col * this.tileSize + this.tileSize / 2,
   279→        y: pos.row * this.tileSize + this.tileSize / 2,
   280→        direction: pos.direction as Direction,
   281→        characterKey: npcId,
   282→        isMoving: false,
   283→        animationFrame: 0,
   284→        animationTimer: 0,
   285→        speed: 0,
   286→        visible: !pos.appearsAfter || saveManager.isObjectiveCompleted(pos.appearsAfter),
   287→        isPlayer: false,
   288→        dialogueTriggered: false,
   289→        appearsAfterObjective: pos.appearsAfter,
   290→      };
   291→      this.entities.set(npcId, entity);
   292→    }
   293→
   294→    // Load saved progress
   295→    await saveManager.loadProgress();
   296→    const gameState = saveManager.getGameState();
   297→    this.chapterPhase = gameState.chapterPhase;
   298→    this.timeOfDay = gameState.timeOfDay;
   299→    
   300→    if (this.chapterPhase !== 'intro') {
   301→      this.introShown = true;
   302→    }
   303→
   304→    // Update entity visibility based on completed objectives
   305→    this._updateEntityVisibility();
   306→
   307→    // Mark already-triggered zones
   308→    this.triggerZones.forEach(zone => {
   309→      if (zone.requiresObjective && saveManager.isObjectiveCompleted(zone.requiresObjective)) {
   310→        // The prerequisite is done, but this zone itself may not have been triggered yet
   311→        // Only mark as triggered if the objective THIS zone completes is also done
   312→      }
   313→    });
   314→
   315→    // Bind input handlers
   316→    this._bindInput();
   317→
   318→    // Listen for objective completion events to spawn celebration bursts
   319→    gameEvents.on('quest:objectiveComplete', (objectiveId: unknown) => {
   320→      const id = objectiveId as string;
   321→      const loc = OBJECTIVE_LOCATIONS[id];
   322→      if (loc) {
   323→        // Spawn burst at the objective's location in world coords
   324→        this._spawnCelebrationBurst(
   325→          loc.col * this.tileSize + this.tileSize / 2,
   326→          loc.row * this.tileSize + this.tileSize / 2
   327→        );
   328→      }
   329→    });
   330→
   331→    // Start auto-save
   332→    saveManager.startAutoSave();
   333→
   334→    // Emit game ready
   335→    gameEvents.emit('game:ready');
   336→
   337→    // Initialize some particles
   338→    this._initParticles();
   339→
   340→    // Initialize fireflies (persistent ambient atmosphere)
   341→    this._initFireflies();
   342→
   343→    // Start game loop
   344→    this.lastTimestamp = performance.now();
   345→    this._gameLoop(this.lastTimestamp);
   346→  }

   348→  /**
   349→   * Spawn a golden celebration burst at the given world position.
   350→   * Used when an objective is completed.
   351→   */
   352→  private _spawnCelebrationBurst(worldX: number, worldY: number): void {
   353→    const colors = ['#FFD700', '#FFA500', '#FFEC8B', '#FFEE00', '#FFC125'];
   354→    const count = 28;
   355→    for (let i = 0; i < count; i++) {
   356→      const angle = (i / count) * Math.PI * 2 + Math.random() * 0.3;
   357→      const speed = 60 + Math.random() * 80;
   358→      this.burstParticles.push({
   359→        x: worldX,
   360→        y: worldY,
   361→        vx: Math.cos(angle) * speed,
   362→        vy: Math.sin(angle) * speed - 30, // Slight upward bias
   363→        life: 0,
   364→        maxLife: 1.0 + Math.random() * 0.6,
   365→        color: colors[Math.floor(Math.random() * colors.length)],
   366→        size: 2 + Math.random() * 3,
   367→        gravity: 80,
   368→      });
   369→    }
   370→  }
   371→
   372→  private _updateBurstParticles(dt: number): void {
   373→    for (let i = this.burstParticles.length - 1; i >= 0; i--) {
   374→      const p = this.burstParticles[i];
   375→      p.life += dt;
   376→      p.vy += p.gravity * dt;     // gravity pulls down
   377→      p.x += p.vx * dt;
   378→      p.y += p.vy * dt;
   379→      // Air resistance
   380→      p.vx *= 0.96;
   381→      p.vy *= 0.96;
   382→      if (p.life >= p.maxLife) {
   383→        this.burstParticles.splice(i, 1);
   384→      }
   385→    }
   386→  }
   387→
   388→  private _renderBurstParticles(ctx: CanvasRenderingContext2D): void {
   389→    for (const p of this.burstParticles) {
   390→      const screenX = p.x - this.cameraX;
   391→      const screenY = p.y - this.cameraY;
   392→      if (screenX < -20 || screenX > this.viewWidth + 20 || screenY < -20 || screenY > this.viewHeight + 20) continue;
   393→      const lifeRatio = p.life / p.maxLife;
   394→      const alpha = Math.max(0, 1 - lifeRatio);
   395→      // Glow
   396→      ctx.globalAlpha = alpha * 0.4;
   397→      ctx.fillStyle = p.color;
   398→      ctx.beginPath();
   399→      ctx.arc(screenX, screenY, p.size * 2.5, 0, Math.PI * 2);
   400→      ctx.fill();
   401→      // Core
   402→      ctx.globalAlpha = alpha;
   403→      ctx.fillStyle = p.color;
   404→      ctx.beginPath();
   405→      ctx.arc(screenX, screenY, p.size, 0, Math.PI * 2);
   406→      ctx.fill();
   407→    }
   408→    ctx.globalAlpha = 1;
   409→  }

   460→  private _handleInteract(): void {
   461→    if (this.dialogueActive) return;
   462→    if (this.interactCooldown > 0) return;
   463→    
   464→    // Check if player is near an NPC they can talk to
   465→    if (!this.player) return;
   466→
   467→    for (const [npcId, npc] of this.entities) {
   468→      if (npc.isPlayer || !npc.visible) continue;
   469→      
   470→      const dist = this._distanceBetween(this.player, npc);
   471→      if (dist < this.tileSize * 2) {
   472→        this.interactCooldown = 500;
   473→        // Track NPC interaction for exploration stats
   474→        this.npcsTalkedTo.add(npcId);
   475→        
   476→        // Determine which dialogue to trigger
   477→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   478→          this._startDialogue('mang-tenyo-first');
   479→          return;
   480→        }
   481→        // Mang Tenyo can be talked to again — different dialogue based on progress
   482→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   483→          // If gossip objective completed, give more insightful dialogue
   484→          if (saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   485→            this._startDialogue('mang-tenyo-after-gossip');
   486→          } else {
   487→            this._startDialogue('mang-tenyo-repeat');
   488→          }
   489→          return;
   490→        }
   491→      }
   492→    }
   493→  }

   495→  private _distanceBetween(a: Entity, b: Entity): number {
   496→    const dx = a.x - b.x;
   497→    const dy = a.y - b.y;
   498→    return Math.sqrt(dx * dx + dy * dy);
   499→  }

   501→  private _startDialogue(dialogueId: string): void {
   502→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   503→    if (!dialogue) {
   504→      console.warn(`Dialogue not found: ${dialogueId}`);
   505→      return;
   506→    }
   507→
   508→    this.dialogueActive = true;
   509→    this.currentDialogueId = dialogueId;
   510→    this.currentDialogueLineIndex = 0;
   511→
   512→    gameEvents.emit('dialogue:start', {
   513→      dialogueId,
   514→      line: dialogue.lines[0],
   515→      lineIndex: 0,
   516→      totalLines: dialogue.lines.length,
   517→    });
   518→  }
   519→
   520→  advanceDialogue(): void {
   521→    if (!this.currentDialogueId) return;
   522→    
   523→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];
   524→    if (!dialogue) return;
   525→
   526→    this.currentDialogueLineIndex++;
   527→
   528→    if (this.currentDialogueLineIndex >= dialogue.lines.length) {
   529→      // Dialogue complete
   530→      this._endDialogue(dialogue);
   531→      return;
   532→    }
   533→
   534→    const line = dialogue.lines[this.currentDialogueLineIndex];
   535→    
   536→    // Check for time transition
   537→    if (line.isTimeTransition) {
   538→      this.timeOfDay = 'morning';
   539→      this.chapterPhase = 'ibarra-sighting';
   540→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   541→      gameEvents.emit('time:transition', 'morning');
   542→    }
   543→
   544→    // Check for Ibarra sighting
   545→    if (line.isIbarraSighting) {
   546→      this.ibarraAppeared = true;
   547→      // Show Ibarra NPC
   548→      const ibaraEntity = this.entities.get('ibara');
   549→      if (ibaraEntity) {
   550→        ibaraEntity.visible = true;
   551→      }
   552→    }
   553→
   554→    gameEvents.emit('dialogue:line', {
   555→      dialogueId: this.currentDialogueId,
   556→      line,
   557→      lineIndex: this.currentDialogueLineIndex,
   558→      totalLines: dialogue.lines.length,
   559→    });
   560→  }

   562→  private _endDialogue(dialogue: { unlocksObjective?: string | null; id: string }): void {
   563→    this.dialogueActive = false;
   564→    this.currentDialogueId = null;
   565→    this.currentDialogueLineIndex = 0;
   566→
   567→    // Update chapter phase based on dialogue completion
   568→    if (dialogue.id === 'intro-text') {
   569→      this.chapterPhase = 'explore';
   570→      saveManager.updateGameState({ chapterPhase: 'explore' });
   571→    }
   572→
   573→    // Complete objective if dialogue unlocks one
   574→    if (dialogue.unlocksObjective) {

   685→  private _gameLoop(timestamp: number): void {
   686→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   687→    this.lastTimestamp = timestamp;
   688→    this.gameTime += deltaTime;
   689→
   690→    this._update(deltaTime);
   691→    this._render();
   692→
   693→    this.animationFrameId = requestAnimationFrame(this._gameLoop.bind(this));
   694→  }

   696→  private _update(dt: number): void {
   697→    if (!this.player) return;
   698→    
   699→    // Update interact cooldown
   700→    if (this.interactCooldown > 0) {
   701→      this.interactCooldown -= dt * 1000;
   702→    }
   703→
   704→    // Don't allow movement during dialogue
   705→    if (this.dialogueActive) return;
   706→
   707→    // Handle intro state
   708→    if (this.chapterPhase === 'intro' && !this.introShown) {
   709→      this.introShown = true;
   710→      this._startDialogue('intro-text');
   711→      return;
   712→    }
   713→
   714→    // Player movement
   715→    let moveX = 0;
   716→    let moveY = 0;
   717→
   718→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   719→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   720→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   721→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   722→
   723→    // Normalize diagonal movement
   724→    if (moveX !== 0 && moveY !== 0) {
   725→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   726→      moveX /= len;
   727→      moveY /= len;
   728→    }
   729→
   730→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   731→
   732→    if (this.player.isMoving) {
   733→      // Determine direction from movement vector
   734→      this.player.direction = this._vectorToDirection(moveX, moveY);
   735→      
   736→      const newX = this.player.x + moveX * this.player.speed * dt;
   737→      const newY = this.player.y + moveY * this.player.speed * dt;
   738→
   739→      // Collision check
   740→      if (!this._isBlocked(newX, newY)) {
   741→        this.player.x = newX;
   742→        this.player.y = newY;
   743→      }
   744→
   745→      // Update animation
   746→      this.player.animationTimer += dt;
   747→      const animConfig = characterData.characters.student.animations.Walk;
   748→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   749→        this.player.animationTimer = 0;
   750→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   751→      }
   752→    } else {
   753→      // Idle animation
   754→      this.player.animationTimer += dt;
   755→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   756→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   757→        this.player.animationTimer = 0;
   758→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   759→      }
   760→    }
   761→
   762→    // Update camera to follow player
   763→    this.cameraX = this.player.x - this.viewWidth / 2;
   764→    this.cameraY = this.player.y - this.viewHeight / 2;
   765→
   766→    // Clamp camera to map bounds
   767→    const mapPixelWidth = mapData.width * this.tileSize;
   768→    const mapPixelHeight = mapData.height * this.tileSize;
   769→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   770→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   771→
   772→    // NPC idle animations
   773→    for (const [npcId, npc] of this.entities) {
   774→      if (npc.isPlayer || !npc.visible) continue;
   775→      npc.animationTimer += dt;
   776→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   777→      if (animConfig) {
   778→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   779→          npc.animationTimer = 0;
   780→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   781→        }
   782→      }
   783→    }
   784→
   785→    // Check trigger zones
   786→    this._checkTriggerZones();
   787→
   788→    // Update particles
   789→    this._updateParticles(dt);
   790→
   791→    // Update fireflies (ambient glowing insects)
   792→    this._updateFireflies(dt);
   793→
   794→    // Update drifting leaves (occasional foliage)
   795→    this._updateDriftLeaves(dt);
   796→
   797→    // Update burst particles (celebration effect)
   798→    this._updateBurstParticles(dt);
   799→
   800→    // Update game state in SaveManager
   801→    saveManager.updateGameState({
   802→      playerPosition: {
   803→        row: Math.floor(this.player.y / this.tileSize),
   804→        col: Math.floor(this.player.x / this.tileSize),
   805→      },
   806→      playerDirection: this.player.direction,
   807→      currentDialogue: this.currentDialogueId,
   808→      timeOfDay: this.timeOfDay,
   809→      chapterPhase: this.chapterPhase,
   810→    });
   811→
   812→    // Track exploration stats — count unique tiles visited
   813→    const tileRow = Math.floor(this.player.y / this.tileSize);
   814→    const tileCol = Math.floor(this.player.x / this.tileSize);
   815→    const tileKey = `${tileRow},${tileCol}`;
   816→    this.visitedTiles.add(tileKey);
   817→
   818→    // Flush stats to localStorage every ~2s (debounced)
   819→    this.statsFlushTimer += dt;
   820→    if (this.statsFlushTimer > 2.0) {
   821→      this.statsFlushTimer = 0;
   822→      try {
   823→        localStorage.setItem('noor-stats', JSON.stringify({
   824→          tilesExplored: this.visitedTiles.size,
   825→          npcsTalkedTo: this.npcsTalkedTo.size,
   826→        }));
   827→      } catch {
   828→        // localStorage may be unavailable; ignore
   829→      }
   830→    }
   831→  }

   833→  private _vectorToDirection(x: number, y: number): Direction {
   834→    // Map movement vector to 8-direction name
   835→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   836→    
   837→    if (angle >= -22.5 && angle < 22.5) return 'east';
   838→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   839→    if (angle >= 67.5 && angle < 112.5) return 'south';
   840→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   841→    if (angle >= 157.5 || angle < -157.5) return 'west';
   842→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   843→    if (angle >= -112.5 && angle < -67.5) return 'north';
   844→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   845→    return 'south';
   846→  }

   848→  private _isBlocked(x: number, y: number): boolean {
   849→    // Check collision map
   850→    const col = Math.floor(x / this.tileSize);
   851→    const row = Math.floor(y / this.tileSize);
   852→    
   853→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   854→    
   855→    // Check player sprite footprint (check a small area around center)
   856→    const checks = [
   857→      { x: x - 16, y: y - 16 },
   858→      { x: x + 16, y: y - 16 },
   859→      { x: x - 16, y: y + 16 },
   860→      { x: x + 16, y: y + 16 },
   861→    ];
   862→    
   863→    for (const check of checks) {
   864→      const cCol = Math.floor(check.x / this.tileSize);
   865→      const cRow = Math.floor(check.y / this.tileSize);
   866→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   867→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   868→    }
   869→    
   870→    return false;
   871→  }

   873→  private _checkTriggerZones(): void {
   874→    if (!this.player || this.dialogueActive) return;
   875→    
   876→    const playerRow = Math.floor(this.player.y / this.tileSize);
   877→    const playerCol = Math.floor(this.player.x / this.tileSize);
   878→
   879→    for (const zone of this.triggerZones) {
   880→      if (zone.triggered) continue;
   881→
   882→      // Check prerequisite objective
   883→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   884→
   885→      // Check if player is in the zone
   886→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   887→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   888→
   889→      // Or within radius of NPC
   890→      let nearNpc = false;
   891→      if (zone.radius > 0 && zone.npcId) {
   892→        const npc = this.entities.get(zone.npcId);
   893→        if (npc && npc.visible) {
   894→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   895→        }
   896→      }
   897→
   898→      if (inZone || nearNpc) {
   899→        zone.triggered = true;
   900→
   901→        switch (zone.type) {
   902→          case 'npc-interact':
   903→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   904→            break;
   905→          case 'auto-trigger':
   906→            // Generic auto-trigger: if zone has a dialogueId, start that dialogue
   907→            if (zone.dialogueId) {
   908→              // Special handling for Ibarra sighting (time transition)
   909→              if (zone.id === 'ibarra-sighting') {
   910→                this.timeOfDay = 'morning';
   911→                this.chapterPhase = 'ibarra-sighting';
   912→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   913→                this._updateEntityVisibility();
   914→              }
   915→              this._startDialogue(zone.dialogueId);
   916→            } else {
   917→              // Legacy hardcoded triggers as fallback
   918→              if (zone.id === 'market-gossip') {
   919→                this._startDialogue('market-gossip');
   920→              } else if (zone.id === 'ibarra-sighting') {
   921→                this.timeOfDay = 'morning';
   922→                this.chapterPhase = 'ibarra-sighting';
   923→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   924→                this._updateEntityVisibility();
   925→                this._startDialogue('ibarra-sighting');
   926→              }
   927→            }
   928→            break;
   929→        }
   930→      }
   931→    }
   932→  }

   936→  private _initParticles(): void {
   937→    // Seed some initial particles

   940→    }

   943→  private _spawnParticle(): void {
   944→    if (!this.player) return;
   945→    // Spawn particle near the camera view area
   946→    const cx = this.cameraX + this.viewWidth / 2;
   947→    const cy = this.cameraY + this.viewHeight / 2;
   948→    this.particles.push({
   949→      x: cx + (Math.random() - 0.5) * this.viewWidth * 0.8,
   950→      y: cy + (Math.random() - 0.5) * this.viewHeight * 0.8,
   951→      vx: (Math.random() - 0.5) * 8, // Slow drift
   952→      vy: -Math.random() * 4 - 1, // Slight upward drift
   953→      alpha: 0,
   954→      size: Math.random() * 2.5 + 1,
   955→      life: 0,
   956→      maxLife: Math.random() * 6 + 4, // 4-10 seconds
   957→    });
   958→  }
   959→
   960→  private _updateParticles(dt: number): void {
   961→    // Spawn new particles periodically
   962→    this.particleSpawnTimer += dt;
   963→    if (this.particleSpawnTimer > 0.3 && this.particles.length < 25) {
   964→      this.particleSpawnTimer = 0;
   965→      this._spawnParticle();
   966→    }
   967→
   968→    // Update existing particles
   969→    for (let i = this.particles.length - 1; i >= 0; i--) {
   970→      const p = this.particles[i];
   971→      p.life += dt;
   972→      p.x += p.vx * dt;
   973→      p.y += p.vy * dt;
   974→      // Gentle oscillation
   975→      p.x += Math.sin(this.gameTime * 0.5 + i) * 0.3;
   976→
   977→      // Fade in then fade out
   978→      const lifeRatio = p.life / p.maxLife;
   979→      if (lifeRatio < 0.2) {
   980→        p.alpha = lifeRatio / 0.2 * 0.4; // Fade in
   981→      } else if (lifeRatio > 0.7) {
   982→        p.alpha = (1 - lifeRatio) / 0.3 * 0.4; // Fade out
   983→      } else {
   984→        p.alpha = 0.4; // Visible
   985→      }
   986→
   987→      // Remove dead particles
   988→      if (p.life >= p.maxLife) {
   989→        this.particles.splice(i, 1);
   990→      }
   991→    }
   992→  }

   994→  // ==================== FIREFLIES ====================
   995→
   996→  private _initFireflies(): void {
   997→    // Pre-spawn fireflies scattered around the map (persistent)
   998→    if (!this.player) return;
   999→    const MAP_W = mapData.width * this.tileSize;
  1000→    const MAP_H = mapData.height * this.tileSize;
  1001→    for (let i = 0; i < 18; i++) {
  1002→      const fx = Math.random() * MAP_W;
  1003→      const fy = Math.random() * MAP_H;
  1004→      this.fireflies.push({
  1005→        x: fx,
  1006→        y: fy,
  1007→        baseX: fx,
  1008→        baseY: fy,
  1009→        phase: Math.random() * Math.PI * 2,
  1010→        blinkSpeed: 0.8 + Math.random() * 1.6, // 0.8-2.4 rad/s
  1011→        driftPhase: Math.random() * Math.PI * 2,
  1012→        brightness: 0,
  1013→      });
  1014→    }
  1015→  }
  1016→
  1017→  private _updateFireflies(dt: number): void {
  1018→    // Note: gameTime is already updated in the main _update() loop
  1019→
  1020→    // Occasionally relocate fireflies that are too far from the player's view
  1021→    // so we always have some nearby fireflies
  1022→    this.fireflySpawnTimer += dt;
  1023→    if (this.fireflySpawnTimer > 1.5 && this.player) {
  1024→      this.fireflySpawnTimer = 0;
  1025→      // Despawn fireflies that wandered too far
  1026→      this.fireflies = this.fireflies.filter(f => {
  1027→        const dx = f.x - this.player!.x;
  1028→        const dy = f.y - this.player!.y;
  1029→        return Math.sqrt(dx * dx + dy * dy) < this.viewWidth * 1.2;
  1030→      });
  1031→      // Spawn a few new ones near the player's view
  1032→      while (this.fireflies.length < 18 && this.player) {
  1033→        const angle = Math.random() * Math.PI * 2;
  1034→        const radius = this.viewWidth * 0.4 + Math.random() * this.viewWidth * 0.3;
  1035→        const fx = this.player.x + Math.cos(angle) * radius;
  1036→        const fy = this.player.y + Math.sin(angle) * radius;
  1037→        this.fireflies.push({
  1038→          x: fx,
  1039→          y: fy,
  1040→          baseX: fx,
  1041→          baseY: fy,
  1042→          phase: Math.random() * Math.PI * 2,
  1043→          blinkSpeed: 0.8 + Math.random() * 1.6,
  1044→          driftPhase: Math.random() * Math.PI * 2,
  1045→          brightness: 0,
  1046→        });
  1047→      }
  1048→    }
  1049→
  1050→    // Update firefly positions and brightness
  1051→    for (const f of this.fireflies) {
  1052→      // Sinusoidal drift around the anchor point
  1053→      const driftRadius = 18;
  1054→      f.x = f.baseX + Math.cos(this.gameTime * 0.6 + f.driftPhase) * driftRadius;
  1055→      f.y = f.baseY + Math.sin(this.gameTime * 0.4 + f.driftPhase * 1.3) * driftRadius * 0.7;
  1056→
  1057→      // Slowly migrate the anchor point (gentle wandering)
  1058→      f.baseX += Math.sin(this.gameTime * 0.15 + f.driftPhase) * 0.15;
  1059→      f.baseY += Math.cos(this.gameTime * 0.18 + f.phase) * 0.1;
  1060→
  1061→      // Blink brightness using a smooth wave (0..1)
  1062→      // Use sin^4 for sharper on/off transitions
  1063→      const raw = Math.sin(this.gameTime * f.blinkSpeed + f.phase);
  1064→      const norm = (raw + 1) / 2; // 0..1
  1065→      f.brightness = Math.pow(norm, 3); // Sharp blink curve
  1066→    }
  1067→  }
  1068→
  1069→  private _renderFireflies(ctx: CanvasRenderingContext2D): void {
  1070→    // Fireflies only show during afternoon (golden hour) — they evoke a warm evening feel
  1071→    if (this.timeOfDay !== 'afternoon') return;
  1072→
  1073→    ctx.save();
  1074→    ctx.globalCompositeOperation = 'lighter'; // Additive blending for glow
  1075→
  1076→    for (const f of this.fireflies) {
  1077→      const screenX = f.x - this.cameraX;
  1078→      const screenY = f.y - this.cameraY;
  1079→
  1080→      // Skip off-screen fireflies
  1081→      if (screenX < -30 || screenX > this.viewWidth + 30 || screenY < -30 || screenY > this.viewHeight + 30) continue;
  1082→
  1083→      const b = f.brightness;
  1084→      if (b < 0.02) continue;
  1085→
  1086→      // Outer glow halo (large soft circle)
  1087→      const glowRadius = 12;

  1109→  private _updateDriftLeaves(dt: number): void {
  1110→    // Occasionally spawn a new leaf
  1111→    this.leafSpawnTimer += dt;
  1112→    if (this.leafSpawnTimer > 2.5 + Math.random() * 3 && this.player && this.driftLeaves.length < 8) {
  1113→      this.leafSpawnTimer = 0;
  1114→      // Spawn from the right side (wind blowing from east to west)
  1115→      const spawnX = this.player.x + this.viewWidth * 0.6 + Math.random() * 100;
  1116→      const spawnY = this.player.y - this.viewHeight * 0.3 + Math.random() * this.viewHeight * 0.6;
  1117→      const colors = ['#8B5A2B', '#A0522D', '#6B4423', '#C17840', '#9C6B3F'];
  1118→      this.driftLeaves.push({
  1119→        x: spawnX,
  1120→        y: spawnY,
  1121→        vx: -25 - Math.random() * 20, // Wind blows leftward
  1122→        vy: 8 + Math.random() * 8,    // Gentle downward drift
  1123→        rotation: Math.random() * Math.PI * 2,
  1124→        rotationSpeed: (Math.random() - 0.5) * 3,
  1125→        size: 5 + Math.random() * 4,
  1126→        color: colors[Math.floor(Math.random() * colors.length)],
  1127→        alpha: 0,
  1128→        life: 0,
  1129→        maxLife: 8 + Math.random() * 4,
  1130→      });
  1131→    }
  1132→
  1133→    // Update leaves
  1134→    for (let i = this.driftLeaves.length - 1; i >= 0; i--) {
  1135→      const leaf = this.driftLeaves[i];
  1136→      leaf.life += dt;
  1137→
  1138→      // Leaf motion: drifts left and downward, with sinusoidal sway
  1139→      leaf.x += leaf.vx * dt + Math.sin(this.gameTime * 1.5 + i) * 0.5;
  1140→      leaf.y += leaf.vy * dt + Math.cos(this.gameTime * 1.2 + i * 0.7) * 0.3;
  1141→      leaf.rotation += leaf.rotationSpeed * dt;
  1142→
  1143→      // Fade in/out
  1144→      const lifeRatio = leaf.life / leaf.maxLife;
  1145→      if (lifeRatio < 0.15) {
  1146→        leaf.alpha = lifeRatio / 0.15 * 0.85;
  1147→      } else if (lifeRatio > 0.7) {
  1148→        leaf.alpha = (1 - lifeRatio) / 0.3 * 0.85;
  1149→      } else {
  1150→        leaf.alpha = 0.85;
  1151→      }
  1152→
  1153→      // Remove dead leaves
  1154→      if (leaf.life >= leaf.maxLife) {
  1155→        this.driftLeaves.splice(i, 1);
  1156→      }
  1157→    }
  1158→  }

  1160→  private _renderDriftLeaves(ctx: CanvasRenderingContext2D): void {

  1263→  /**
  1264→   * Render a golden arrow at the screen edge pointing toward the next incomplete objective.
  1265→   * Only shown when no dialogue is active and the player isn't standing in the objective zone.
  1266→   */
  1267→  private _renderWaypointArrow(ctx: CanvasRenderingContext2D): void {
  1268→    if (!this.player || this.dialogueActive) return;
  1269→    if (this.chapterPhase === 'intro' || this.chapterPhase === 'complete') return;
  1270→
  1271→    // Find the next incomplete objective
  1272→    const objectivesOrder = ['obj.ch1.follow_tenyo', 'obj.ch1.overhear_gossip', 'obj.ch1.see_ibarra'];
  1273→    const nextObjId = objectivesOrder.find(id => !saveManager.isObjectiveCompleted(id));
  1274→    if (!nextObjId) return;
  1275→
  1276→    const loc = OBJECTIVE_LOCATIONS[nextObjId];
  1277→    if (!loc) return;
  1278→
  1279→    const targetX = loc.col * this.tileSize + this.tileSize / 2;
  1280→    const targetY = loc.row * this.tileSize + this.tileSize / 2;
  1281→
  1282→    // Vector from player to target
  1283→    const dx = targetX - this.player.x;
  1284→    const dy = targetY - this.player.y;
  1285→    const dist = Math.sqrt(dx * dx + dy * dy);
  1286→
  1287→    // Don't show arrow if very close
  1288→    if (dist < this.tileSize * 2.5) return;
  1289→
  1290→    // Compute target position on screen (may be off-screen)
  1291→    const screenTargetX = targetX - this.cameraX;
  1292→    const screenTargetY = targetY - this.cameraY;
  1293→
  1294→    const cx = this.viewWidth / 2;
  1295→    const cy = this.viewHeight / 2;
  1296→
  1297→    // If the target is on screen, don't show the edge arrow (the indicator is enough)
  1298→    const onScreen = screenTargetX > 40 && screenTargetX < this.viewWidth - 40 &&
  1299→                     screenTargetY > 80 && screenTargetY < this.viewHeight - 80;
  1300→    if (onScreen) return;
  1301→
  1302→    // Direction from center of screen to target
  1303→    const angle = Math.atan2(dy, dx);
  1304→
  1305→    // Place arrow at a fixed radius from screen center, clamped to a margin inside the screen
  1306→    const margin = 80;
  1307→    const maxRadiusX = this.viewWidth / 2 - margin;
  1308→    const maxRadiusY = this.viewHeight / 2 - margin;
  1309→    // Project the direction onto the screen-edge ellipse
  1310→    // Solve for t such that (t*cos(angle), t*sin(angle)) lies on the ellipse (maxRX, maxRY)
  1311→    const cosA = Math.cos(angle);
  1312→    const sinA = Math.sin(angle);
  1313→    const t = 1 / Math.sqrt((cosA * cosA) / (maxRadiusX * maxRadiusX) + (sinA * sinA) / (maxRadiusY * maxRadiusY));
  1314→    const arrowX = cx + cosA * t;
  1315→    const arrowY = cy + sinA * t;
  1316→
  1317→    // Pulsing effect
  1318→    const pulse = Math.sin(this.gameTime * 4) * 0.15 + 0.85;
  1319→
  1320→    ctx.save();
  1321→    ctx.translate(arrowX, arrowY);
  1322→    ctx.rotate(angle);
  1323→
  1324→    // Outer glow
  1325→    ctx.fillStyle = `rgba(255,215,0,${0.15 * pulse})`;
  1326→    ctx.beginPath();
  1327→    ctx.arc(0, 0, 28, 0, Math.PI * 2);
  1328→    ctx.fill();
  1329→
  1330→    // Arrow shadow
  1331→    ctx.fillStyle = 'rgba(0,0,0,0.5)';
  1332→    ctx.beginPath();
  1333→    ctx.moveTo(18, 0);
  1334→    ctx.lineTo(-8, -12);
  1335→    ctx.lineTo(-4, 0);
  1336→    ctx.lineTo(-8, 12);
  1337→    ctx.closePath();
  1338→    ctx.fill();
  1339→
  1340→    // Arrow body
  1341→    ctx.fillStyle = `rgba(255,215,0,${pulse})`;
  1342→    ctx.strokeStyle = 'rgba(180,120,0,1)';
  1343→    ctx.lineWidth = 1.5;
  1344→    ctx.beginPath();
  1345→    ctx.moveTo(15, 0);
  1346→    ctx.lineTo(-10, -10);
  1347→    ctx.lineTo(-6, 0);
  1348→    ctx.lineTo(-10, 10);
  1349→    ctx.closePath();
  1350→    ctx.fill();
  1351→    ctx.stroke();
  1352→
  1353→    ctx.restore();
  1354→
  1355→    // Distance label below arrow
  1356→    const distTiles = Math.round(dist / this.tileSize);
  1357→    ctx.font = 'bold 10px "Geist", sans-serif';
  1358→    ctx.fillStyle = 'rgba(0,0,0,0.6)';
  1359→    ctx.fillRect(arrowX - 22, arrowY + 14, 44, 14);
  1360→    ctx.fillStyle = '#FFD700';
  1361→    ctx.textAlign = 'center';
  1362→    ctx.fillText(`${distTiles}m →`, arrowX, arrowY + 24);
  1363→  }
