   240→   240→  async init(canvas: HTMLCanvasElement): Promise<void> {
   241→   241→    this.canvas = canvas;
   242→   242→
   243→   243→    // Set canvas to internal resolution
   244→   244→    canvas.width = INTERNAL_WIDTH;
   245→   245→    canvas.height = INTERNAL_HEIGHT;
   246→   246→    this.ctx = canvas.getContext('2d')!;
   247→   247→    this.ctx.imageSmoothingEnabled = false;
   248→   248→
   249→   249→    // Create offscreen buffer (same internal resolution)
   250→   250→    this.offscreen = document.createElement('canvas');
   251→   251→    this.offscreen.width = INTERNAL_WIDTH;
   252→   252→    this.offscreen.height = INTERNAL_HEIGHT;
   253→   253→    this.offCtx = this.offscreen.getContext('2d')!;
   254→   254→    this.offCtx.imageSmoothingEnabled = false;
   255→   255→
   256→   256→    // Load all assets via assetManager
   257→   257→    await assetManager.loadAll();
   258→   258→
   259→   259→    // Load character sprites via spriteLoader
   260→   260→    await spriteLoader.loadCharacter('student');
   261→   261→    await spriteLoader.loadCharacter('ibara');
   262→   262→
   263→   263→    // Load save data
   264→   264→    await saveManager.loadProgress();
   265→   265→    saveManager.startAutoSave();
   266→   266→
   267→   267→    // Parse map data
   268→   268→    this._parseMapData();
   269→   269→
   270→   270→    // Parse NPC positions from mapData
   271→   271→    this._parseNPCData();
   272→   272→
   273→   273→    // Parse buildings and props
   274→   274→    this._parseBuildingsAndProps();
   275→   275→
   276→   276→    // Parse trigger zones
   277→   277→    this._parseTriggerZones();
   278→   278→
   279→   279→    // Load NPC interaction counts from localStorage
   280→   280→    this._loadNpcInteractions();
   281→   281→
   282→   282→    // Initialize player position from save data if available
   283→   283→    this._initPlayerFromSave();
   284→   284→
   285→   285→    // Set up input listeners
   286→   286→    this._setupInputListeners();
   287→   287→
   288→   288→    // Initialize camera to center on player
   289→   289→    this._snapCameraToPlayer();
   290→   290→
   291→   291→    // Start game loop
   292→   292→    this.running = true;
   293→   293→    this.lastTime = performance.now();
   294→   294→    this._gameLoop(this.lastTime);
   295→   295→
   296→   296→    this.initialized = true;

   299→   299→    // Start ambient sounds

   301→   301→  }

   310→   310→    saveManager.stopAutoSave();

   317→   317→  private _parseMapData(): void {
   318→   318→    const layers = mapData.layers as any;
   319→   319→    this.groundLayer = layers.ground;
   320→   320→    this.buildingsLayer = layers.buildings;
   321→   321→    this.collisionLayer = layers.collision;
   322→   322→    this.overlayLayer = layers.overlay || Array.from({ length: mapData.height }, () => Array(mapData.width).fill(0));

   326→   326→
   327→   327→  private _parseNPCData(): void {
   328→   328→    this.npcs = [];
   329→   329→    const npcPositions = mapData.npcPositions as Record<string, any>;
   330→   330→    for (const [id, pos] of Object.entries(npcPositions)) {
   331→   331→      const charKey = id;
   332→   332→      const charConfig = (characterData.characters as any)[charKey] || (characterData.characters as any)[id];
   333→   333→      const displayName = charConfig?.displayName || id;
   334→   334→
   335→   335→      // Check visibility condition
   336→   336→      let visible = true;
   337→   337→      if (pos.appearsAfter) {
   338→   338→        visible = saveManager.isObjectiveCompleted(pos.appearsAfter);
   339→   339→      }
   340→   340→
   341→   341→      const row = pos.row as number;
   342→   342→      const col = pos.col as number;
   343→   343→
   344→   344→      this.npcs.push({
   345→   345→        id,
   346→   346→        characterKey: charKey,
   347→   347→        row,
   348→   348→        col,
   349→   349→        facing: this._toDirection4(pos.direction as string),
   350→   350→        visible,
   351→   351→        appearsAfterObjective: pos.appearsAfter,
   352→   352→        spriteSheet: pos.spriteSheet || charConfig?.spriteSheet,
   353→   353→        animFrame: 0,
   354→   354→        animTimer: 0,
   355→   355→        displayName,
   356→   356→        pixelX: col * TILE_SIZE,
   357→   357→        pixelY: row * TILE_SIZE,
   358→   358→      });
   359→   359→    }
   360→   360→  }

   421→   421→  private _initPlayerFromSave(): void {
   422→   422→    const saveData = saveManager.getSaveData();
   423→   423→    if (saveData.gameState.playerPosition) {
   424→   424→      const pos = saveData.gameState.playerPosition;
   425→   425→      // Validate position is within map and walkable
   426→   426→      if (pos.row >= 0 && pos.row < this.mapHeight &&
   427→   427→          pos.col >= 0 && pos.col < this.mapWidth &&
   428→   428→          !this._isCollision(pos.row, pos.col)) {
   429→   429→        this.player.row = pos.row;
   430→   430→        this.player.col = pos.col;
   431→   431→      }
   432→   432→    }
   433→   433→    // Recalculate pixel position
   434→   434→    this.player.pixelX = this.player.col * TILE_SIZE;
   435→   435→    this.player.pixelY = this.player.row * TILE_SIZE;
   436→   436→    this.player.slideFromRow = this.player.row;
   437→   437→    this.player.slideFromCol = this.player.col;
   438→   438→    this.player.slideToRow = this.player.row;
   439→   439→    this.player.slideToCol = this.player.col;

   445→   445→
   446→   446→  private _keydownHandler: ((e: KeyboardEvent) => void) | null = null;

   448→   448→
   449→   449→  private _setupInputListeners(): void {
   450→   450→    this._keydownHandler = (e: KeyboardEvent) => this._onKeyDown(e);
   451→   451→    this._keyupHandler = (e: KeyboardEvent) => this._onKeyUp(e);
   452→   452→    window.addEventListener('keydown', this._keydownHandler);
   453→   453→    window.addEventListener('keyup', this._keyupHandler);
   454→   454→  }
   455→   455→
   456→   456→  private _removeInputListeners(): void {
   457→   457→    if (this._keydownHandler) window.removeEventListener('keydown', this._keydownHandler);
   458→   458→    if (this._keyupHandler) window.removeEventListener('keyup', this._keyupHandler);
   459→   459→  }
   460→   460→
   461→   461→  private _onKeyDown(e: KeyboardEvent): void {
   462→   462→    // Ignore input when focus is in text fields
   463→   463→    const target = e.target as HTMLElement;
   464→   464→    if (target && (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable)) return;
   465→   465→
   466→   466→    const key = e.key.toLowerCase();
   467→   467→
   468→   468→    // Prevent default for game keys (stop page scrolling)
   469→   469→    if (KEY_TO_DIR[key] || key === ' ' || key === 'enter') {
   470→   470→      e.preventDefault();
   471→   471→    }
   472→   472→
   473→   473→    this.keysDown.add(key);
   474→   474→
   475→   475→    // Interaction key (Space/Enter)
   476→   476→    if (key === ' ' || key === 'enter') {
   477→   477→      if (!this.interactPressed) {
   478→   478→        this.interactPressed = true;
   479→   479→        this._handleInteract();
   480→   480→      }
   481→   481→    }
   482→   482→  }
   483→   483→
   484→   484→  private _onKeyUp(e: KeyboardEvent): void {
   485→   485→    const key = e.key.toLowerCase();
   486→   486→    this.keysDown.delete(key);
   487→   487→
   488→   488→    if (key === ' ' || key === 'enter') {
   489→   489→      this.interactPressed = false;
   490→   490→    }
   491→   491→  }
   492→   492→
   493→   493→  // Touch / external controls
   494→   494→  setMoveDirection(dir: 'up' | 'down' | 'left' | 'right' | null): void {
   495→   495→    if (!dir) {
   496→   496→      this.touchDir = null;
   497→   497→      return;
   498→   498→    }
   499→   499→    const mapping: Record<string, Direction4> = {
   500→   500→      up: 'north', down: 'south', left: 'west', right: 'east'
   501→   501→    };
   502→   502→    this.touchDir = mapping[dir] || null;
   503→   503→  }
   504→   504→
   505→   505→  triggerInteract(): void {
   506→   506→    this._handleInteract();
   507→   507→  }

   509→   509→  // ═══════════════════════════════════════════════════════════════════
   510→   510→  // MOVEMENT SYSTEM (Pokémon Grid Movement)
   511→   511→  // ═══════════════════════════════════════════════════════════════════
   512→   512→
   513→   513→  private _getDesiredDirection(): Direction4 | null {
   514→   514→    // Check keyboard keys
   515→   515→    for (const key of this.keysDown) {
   516→   516→      const dir = KEY_TO_DIR[key];
   517→   517→      if (dir) return dir;
   518→   518→    }
   519→   519→    // Check touch direction
   520→   520→    return this.touchDir;
   521→   521→  }
   522→   522→
   523→   523→  private _tryMove(dir: Direction4, now: number): void {
   524→   524→    // Update facing direction immediately
   525→   525→    this.player.facing = dir;
   526→   526→
   527→   527→    if (this.player.isSliding) {
   528→   528→      // Buffer the direction for when slide completes
   529→   529→      this.player.bufferedDir = dir;
   530→   530→      return;
   531→   531→    }
   532→   532→
   533→   533→    // Check pause between moves
   534→   534→    if (now - this.player.lastMoveTime < SLIDE_PAUSE) return;
   535→   535→
   536→   536→    // Calculate target tile
   537→   537→    const offset = DIR_OFFSET[dir];
   538→   538→    const targetRow = this.player.row + offset.dr;
   539→   539→    const targetCol = this.player.col + offset.dc;
   540→   540→
   541→   541→    // Check bounds
   542→   542→    if (targetRow < 0 || targetRow >= this.mapHeight || targetCol < 0 || targetCol >= this.mapWidth) return;
   543→   543→
   544→   544→    // Check collision
   545→   545→    if (this._isCollision(targetRow, targetCol)) return;
   546→   546→
   547→   547→    // Check if NPC occupies target tile
   548→   548→    if (this._isNpcAtTile(targetRow, targetCol)) return;
   549→   549→
   550→   550→    // Check if prop occupies target tile
   551→   551→    if (this._isPropAtTile(targetRow, targetCol)) return;
   552→   552→
   553→   553→    // Start sliding
   554→   554→    this.player.isSliding = true;
   555→   555→    this.player.slideStart = now;
   556→   556→    this.player.slideFromRow = this.player.row;
   557→   557→    this.player.slideFromCol = this.player.col;
   558→   558→    this.player.slideToRow = targetRow;
   559→   559→    this.player.slideToCol = targetCol;
   560→   560→    this.player.bufferedDir = null;
   561→   561→
   562→   562→    // Play footstep sound
   563→   563→    if (now - this.lastFootstepTime > this.FOOTSTEP_INTERVAL) {
   564→   564→      soundManager.playFootstep();
   565→   565→      this.lastFootstepTime = now;
   566→   566→    }
   567→   567→  }
   568→   568→
   569→   569→  private _updateMovement(now: number, dt: number): void {
   570→   570→    if (this.dialogue.active) return; // No movement during dialogue
   571→   571→
   572→   572→    // Process desired direction
   573→   573→    const desired = this._getDesiredDirection();
   574→   574→    if (desired) {
   575→   575→      this._tryMove(desired, now);
   576→   576→    }
   577→   577→
   578→   578→    // Update slide interpolation
   579→   579→    if (this.player.isSliding) {
   580→   580→      const elapsed = now - this.player.slideStart;
   581→   581→      const t = Math.min(1, elapsed / SLIDE_DURATION);
   582→   582→
   583→   583→      // Interpolate pixel position
   584→   584→      const fromX = this.player.slideFromCol * TILE_SIZE;
   585→   585→      const fromY = this.player.slideFromRow * TILE_SIZE;
   586→   586→      const toX = this.player.slideToCol * TILE_SIZE;
   587→   587→      const toY = this.player.slideToRow * TILE_SIZE;
   588→   588→
   589→   589→      this.player.pixelX = fromX + (toX - fromX) * t;
   590→   590→      this.player.pixelY = fromY + (toY - fromY) * t;
   591→   591→
   592→   592→      // Update walk animation
   593→   593→      this.player.animTimer += dt;
   594→   594→      if (this.player.animTimer >= WALK_ANIM_RATE) {
   595→   595→        this.player.animTimer -= WALK_ANIM_RATE;
   596→   596→        this.player.animFrame++;
   597→   597→      }
   598→   598→
   599→   599→      // Check if slide is complete
   600→   600→      if (t >= 1) {
   601→   601→        this.player.isSliding = false;
   602→   602→        this.player.row = this.player.slideToRow;
   603→   603→        this.player.col = this.player.slideToCol;
   604→   604→        this.player.pixelX = this.player.col * TILE_SIZE;
   605→   605→        this.player.pixelY = this.player.row * TILE_SIZE;
   606→   606→        this.player.lastMoveTime = now;
   607→   607→
   608→   608→        // Emit player:move event
   609→   609→        gameEvents.emit('player:move', { row: this.player.row, col: this.player.col });
   610→   610→
   611→   611→        // Update save state
   612→   612→        saveManager.updateGameState({
   613→   613→          playerPosition: { row: this.player.row, col: this.player.col },
   614→   614→          playerDirection: this.player.facing,
   615→   615→        });
   616→   616→
   617→   617→        // Check trigger zones
   618→   618→        this._checkTriggerZones(now);
   619→   619→
   620→   620→        // Process buffered direction
   621→   621→        if (this.player.bufferedDir) {
   622→   622→          const bufDir = this.player.bufferedDir;
   623→   623→          this.player.bufferedDir = null;
   624→   624→          this._tryMove(bufDir, now);
   625→   625→        } else {
   626→   626→          // Switch to idle
   627→   627→          this.player.animFrame = 0;
   628→   628→          this.player.animTimer = 0;
   629→   629→        }
   630→   630→      }
   631→   631→    } else {
   632→   632→      // Idle animation
   633→   633→      this.player.animTimer += dt;
   634→   634→      if (this.player.animTimer >= IDLE_ANIM_RATE) {
   635→   635→        this.player.animTimer -= IDLE_ANIM_RATE;
   636→   636→        this.player.animFrame++;
   637→   637→      }
   638→   638→
   639→   639→      // Snap pixel position to tile (safety)
   640→   640→      this.player.pixelX = this.player.col * TILE_SIZE;
   641→   641→      this.player.pixelY = this.player.row * TILE_SIZE;
   642→   642→    }
   643→   643→  }

   649→   649→  private _isCollision(row: number, col: number): boolean {
   650→   650→    if (row < 0 || row >= this.mapHeight || col < 0 || col >= this.mapWidth) return true;
   651→   651→    return this.collisionLayer[row]?.[col] === 1;
   652→   652→  }

   654→   654→  private _isNpcAtTile(row: number, col: number): boolean {
   655→   655→    return this.npcs.some(npc => npc.visible && npc.row === row && npc.col === col);
   656→   656→  }
   657→   657→
   658→   658→  private _isPropAtTile(row: number, col: number): boolean {
   659→   659→    for (const prop of this.props) {
   660→   660→      if (row >= prop.row && row < prop.row + prop.height &&
   661→   661→          col >= prop.col && col < prop.col + prop.width) {
   662→   662→        return true;
   663→   663→      }
   664→   664→    }
   665→   665→    return false;
   666→   666→  }

   669→   669→  // CAMERA SYSTEM
   670→   670→  // ═══════════════════════════════════════════════════════════════════
   671→   671→
   672→   672→  private _updateCamera(): void {
   673→   673→    // Camera target: center on player
   674→   674→    const playerCenterX = this.player.pixelX + TILE_SIZE / 2;
   675→   675→    const playerCenterY = this.player.pixelY + TILE_SIZE / 2;
   676→   676→
   677→   677→    this.camera.targetX = playerCenterX - INTERNAL_WIDTH / 2;
   678→   678→    this.camera.targetY = playerCenterY - INTERNAL_HEIGHT / 2;
   679→   679→
   680→   680→    // Clamp to map edges
   681→   681→    const maxCamX = this.mapWidth * TILE_SIZE - INTERNAL_WIDTH;
   682→   682→    const maxCamY = this.mapHeight * TILE_SIZE - INTERNAL_HEIGHT;
   683→   683→    this.camera.targetX = Math.max(0, Math.min(maxCamX, this.camera.targetX));
   684→   684→    this.camera.targetY = Math.max(0, Math.min(maxCamY, this.camera.targetY));
   685→   685→
   686→   686→    // Smooth lerp
   687→   687→    this.camera.x += (this.camera.targetX - this.camera.x) * CAMERA_LERP;
   688→   688→    this.camera.y += (this.camera.targetY - this.camera.y) * CAMERA_LERP;
   689→   689→
   690→   690→    // Clamp again (for precision)
   691→   691→    this.camera.x = Math.max(0, Math.min(maxCamX, this.camera.x));
   692→   692→    this.camera.y = Math.max(0, Math.min(maxCamY, this.camera.y));
   693→   693→  }
   694→   694→
   695→   695→  private _snapCameraToPlayer(): void {
   696→   696→    const playerCenterX = this.player.pixelX + TILE_SIZE / 2;
   697→   697→    const playerCenterY = this.player.pixelY + TILE_SIZE / 2;
   698→   698→    this.camera.x = playerCenterX - INTERNAL_WIDTH / 2;
   699→   699→    this.camera.y = playerCenterY - INTERNAL_HEIGHT / 2;
   700→   700→    const maxCamX = this.mapWidth * TILE_SIZE - INTERNAL_WIDTH;
   701→   701→    const maxCamY = this.mapHeight * TILE_SIZE - INTERNAL_HEIGHT;
   702→   702→    this.camera.x = Math.max(0, Math.min(maxCamX, this.camera.x));
   703→   703→    this.camera.y = Math.max(0, Math.min(maxCamY, this.camera.y));
   704→   704→    this.camera.targetX = this.camera.x;
   705→   705→    this.camera.targetY = this.camera.y;
   706→   706→  }

   712→   712→  private _updateNPCs(dt: number): void {
   713→   713→    for (const npc of this.npcs) {

   715→   715→      if (npc.appearsAfterObjective) {
   716→   716→        const wasVisible = npc.visible;

   718→   718→        if (npc.visible && !wasVisible) {
   719→   719→          gameEvents.emit('ibarra:appear');
   720→   720→        }
   721→   721→      }
   722→   722→
   723→   723→      // Update idle animation
   724→   724→      npc.animTimer += dt;
   725→   725→      if (npc.animTimer >= IDLE_ANIM_RATE) {
   726→   726→        npc.animTimer -= IDLE_ANIM_RATE;
   727→   727→        npc.animFrame++;
   728→   728→      }
   729→   729→
   730→   730→      // Update pixel position (NPCs don't move, but snap to tile)
   731→   731→      npc.pixelX = npc.col * TILE_SIZE;
   732→   732→      npc.pixelY = npc.row * TILE_SIZE;
   733→   733→    }
   734→   734→  }

   738→   738→  // ═══════════════════════════════════════════════════════════════════
   739→   739→
   740→   740→  private _handleInteract(): void {
   741→   741→    if (this.dialogue.active) {
   742→   742→      // Advance dialogue
   743→   743→      this.advanceDialogue();
   744→   744→      return;
   745→   745→    }
   746→   746→
   747→   747→    // Try to talk to an adjacent NPC
   748→   748→    const facing = this.player.facing;
   749→   749→    const offset = DIR_OFFSET[facing];
   750→   750→    const targetRow = this.player.row + offset.dr;
   751→   751→    const targetCol = this.player.col + offset.dc;
   752→   752→
   753→   753→    // Check NPCs at target tile or current tile
   754→   754→    let closestNpc: NPCState | null = null;
   755→   755→    let closestDist = Infinity;
   756→   756→
   757→   757→    for (const npc of this.npcs) {
   758→   758→      if (!npc.visible) continue;
   759→   759→      const dist = Math.abs(npc.row - targetRow) + Math.abs(npc.col - targetCol);
   760→   760→      if (dist === 0 && dist < closestDist) {
   761→   761→        closestNpc = npc;
   762→   762→        closestDist = dist;
   763→   763→      }
   764→   764→    }
   765→   765→
   766→   766→    // Also check NPCs on player's tile (standing on top of them)
   767→   767→    if (!closestNpc) {
   768→   768→      for (const npc of this.npcs) {
   769→   769→        if (!npc.visible) continue;
   770→   770→        const dist = Math.abs(npc.row - this.player.row) + Math.abs(npc.col - this.player.col);
   771→   771→        if (dist <= 1 && dist < closestDist) {
   772→   772→          closestNpc = npc;
   773→   773→          closestDist = dist;
   774→   774→        }
   775→   775→      }
   776→   776→    }
   777→   777→
   778→   778→    if (closestNpc) {
   779→   779→      // Face NPC toward player
   780→   780→      closestNpc.facing = this._oppositeDirection(this.player.facing);
   781→   781→      this._startNpcDialogue(closestNpc);
   782→   782→    }
   783→   783→  }

   785→   785→  private _startNpcDialogue(npc: NPCState): void {
   786→   786→    // Increment interaction count
   787→   787→    this.npcInteractionCounts[npc.id] = (this.npcInteractionCounts[npc.id] || 0) + 1;
   788→   788→    this._saveNpcInteractions();
   789→   789→
   790→   790→    // Determine which dialogue to use
   791→   791→    let dialogueId: string | null = null;
   792→   792→    const npcId = npc.id;
   793→   793→
   794→   794→    // Check for first-time dialogues based on NPC ID
   795→   795→    if (npcId === 'mang-tenyo') {
   796→   796→      if (!saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   797→   797→        dialogueId = 'mang-tenyo-first';
   798→   798→      } else if (!saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   799→   799→        dialogueId = 'mang-tenyo-repeat';
   800→   800→      } else {
   801→   801→        dialogueId = 'mang-tenyo-after-gossip';
   802→   802→      }
   803→   803→    } else if (npcId === 'vendor-1' || npcId === 'vendor-2' || npcId === 'kitchen-staff-1' || npcId === 'kitchen-staff-2') {
   804→   804→      // Vendors share market-gossip dialogue
   805→   805→      if (saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo') && !saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   806→   806→        dialogueId = 'market-gossip';
   807→   807→      } else {
   808→   808→        // Generic vendor repeat
   809→   809→        dialogueId = 'mang-tenyo-repeat'; // Fallback
   810→   810→      }
   811→   811→    } else if (npcId === 'ibara') {
   812→   812→      // Ibarra sighting
   813→   813→      dialogueId = 'ibarra-sighting';
   814→   814→    }
   815→   815→
   816→   816→    if (dialogueId && (dialogueData.dialogues as any)[dialogueId]) {
   817→   817→      this._startDialogue(dialogueId, npcId);
   818→   818→    }
   819→   819→  }

   821→   821→  private _startDialogue(dialogueId: string, npcId: string | null): void {
   822→   822→    const dialogueObj = (dialogueData.dialogues as any)[dialogueId];
   823→   823→    if (!dialogueObj) return;
   824→   824→
   825→   825→    const lines: DialogueLine[] = dialogueObj.lines.map((l: any) => ({
   826→   826→      speaker: l.speaker,
   827→   827→      text: l.text,
   828→   828→      translation: l.translation,
   829→   829→      isTimeTransition: l.isTimeTransition,
   830→   830→      isIbarraSighting: l.isIbarraSighting,
   831→   831→    }));
   832→   832→
   833→   833→    this.dialogue = {
   834→   834→      active: true,
   835→   835→      dialogueId,
   836→   836→      lines,
   837→   837→      currentLineIndex: 0,
   838→   838→      npcId,
   839→   839→      triggered: true,
   840→   840→    };
   841→   841→
   842→   842→    // Emit dialogue:start event
   843→   843→    const firstLine = lines[0];
   844→   844→    gameEvents.emit('dialogue:start', {
   845→   845→      dialogueId,
   846→   846→      line: firstLine,
   847→   847→      lineIndex: 0,
   848→   848→      totalLines: lines.length,
   849→   849→    });
   850→   850→
   851→   851→    // Complete any objectives this dialogue unlocks
   852→   852→    if (dialogueObj.unlocksObjective) {
   853→   853→      saveManager.completeObjective(dialogueObj.unlocksObjective);
   854→   854→      saveManager.saveProgress();
   855→   855→    }
   856→   856→  }
   857→   857→
   858→   858→  advanceDialogue(): void {
   859→   859→    if (!this.dialogue.active) return;
   860→   860→
   861→   861→    const nextIndex = this.dialogue.currentLineIndex + 1;
   862→   862→
   863→   863→    if (nextIndex >= this.dialogue.lines.length) {
   864→   864→      // End dialogue
   865→   865→      const wasTimeTransition = this.dialogue.lines.some(l => l.isTimeTransition);
   866→   866→
   867→   867→      this.dialogue = {
   868→   868→        active: false,
   869→   869→        dialogueId: null,
   870→   870→        lines: [],
   871→   871→        currentLineIndex: 0,
   872→   872→        npcId: null,
   873→   873→        triggered: false,
   874→   874→      };
   875→   875→
   876→   876→      gameEvents.emit('dialogue:end');
   877→   877→
   878→   878→      // Handle time transition
   879→   879→      if (wasTimeTransition) {
   880→   880→        gameEvents.emit('time:transition', 'morning');
   881→   881→        saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   882→   882→        soundManager.play('time-transition');
   883→   883→      }
   884→   884→
   885→   885→      return;
   886→   886→    }
   887→   887→
   888→   888→    // Advance to next line
   889→   889→    this.dialogue.currentLineIndex = nextIndex;
   890→   890→    const line = this.dialogue.lines[nextIndex];
   891→   891→
   892→   892→    gameEvents.emit('dialogue:line', {
   893→   893→      dialogueId: this.dialogue.dialogueId,
   894→   894→      line,
   895→   895→      lineIndex: nextIndex,
   896→   896→      totalLines: this.dialogue.lines.length,
   897→   897→    });
   898→   898→
   899→   899→    // Handle time transition in dialogue line
   900→   900→    if (line.isTimeTransition) {
   901→   901→      gameEvents.emit('time:transition', 'morning');
   902→   902→      saveManager.updateGameState({ timeOfDay: 'morning' });
   903→   903→    }
   904→   904→  }

   906→   906→  private _checkTriggerZones(now: number): void {
   907→   907→    for (const zone of this.triggerZones) {
   908→   908→      if (zone.triggered) continue;
   909→   909→
   910→   910→      // Check objective requirement
   911→   911→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   912→   912→
   913→   913→      // Check if player is in the zone
   914→   914→      const inZone = this.player.row >= zone.row && this.player.row < zone.row + zone.height &&
   915→   915→                     this.player.col >= zone.col && this.player.col < zone.col + zone.width;
   916→   916→
   917→   917→      // Or within radius
   918→   918→      const centerRow = zone.row + (zone.height - 1) / 2;
   919→   919→      const centerCol = zone.col + (zone.width - 1) / 2;
   920→   920→      const dist = Math.sqrt(
   921→   921→        Math.pow(this.player.row - centerRow, 2) +
   922→   922→        Math.pow(this.player.col - centerCol, 2)
   923→   923→      );
   924→   924→      const inRadius = zone.radius > 0 && dist <= zone.radius;
   925→   925→
   926→   926→      if (inZone || inRadius) {
   927→   927→        zone.triggered = true;
   928→   928→
   929→   929→        if (zone.dialogueId) {
   930→   930→          this._startDialogue(zone.dialogueId, zone.npcId || null);
   931→   931→        }
   932→   932→
   933→   933→        gameEvents.emit('interaction:trigger', { zoneId: zone.id, type: zone.type });
   934→   934→
   935→   935→        // If this triggers a time transition, handle it
   936→   936→        if (zone.triggersTimeTransition) {
   937→   937→          // Time transition will be handled by the dialogue lines
   938→   938→        }
   939→   939→      }
   940→   940→    }
   941→   941→  }

   945→   945→  // ═══════════════════════════════════════════════════════════════════
   946→   946→
   947→   947→  private _updateAmbient(): void {
   948→   948→    // Check if player is near market area (rows 17-22, cols 3-22)
   949→   949→    const nearMarket = this.player.row >= 16 && this.player.row <= 23 &&
   950→   950→                       this.player.col >= 2 && this.player.col <= 23;
   951→   951→
   952→   952→    if (nearMarket && !this.wasNearMarket) {
   953→   953→      soundManager.startMarketAmbient();
   954→   954→      this.wasNearMarket = true;
   955→   955→    } else if (!nearMarket && this.wasNearMarket) {
   956→   956→      soundManager.stopMarketAmbient();
   957→   957→      this.wasNearMarket = false;
   958→   958→    }
   959→   959→  }
   960→   960→

   965→   965→  private _render(): void {
   966→   966→    if (!this.offCtx || !this.ctx) return;
   967→   967→
   968→   968→    const ctx = this.offCtx;
   969→   969→
   970→   970→    // Clear
   971→   971→    ctx.fillStyle = '#1a1a2e';
   972→   972→    ctx.fillRect(0, 0, INTERNAL_WIDTH, INTERNAL_HEIGHT);
   973→   973→
   974→   974→    const camX = Math.round(this.camera.x);
   975→   975→    const camY = Math.round(this.camera.y);
   976→   976→
   977→   977→    // Calculate visible tile range
   978→   978→    const startCol = Math.floor(camX / TILE_SIZE);
   979→   979→    const startRow = Math.floor(camY / TILE_SIZE);
   980→   980→    const endCol = Math.ceil((camX + INTERNAL_WIDTH) / TILE_SIZE);
   981→   981→    const endRow = Math.ceil((camY + INTERNAL_HEIGHT) / TILE_SIZE);
   982→   982→
   983→   983→    // ── LAYER 1: Ground tiles ──
   984→   984→    this._renderGroundTiles(ctx, camX, camY, startRow, startRow, endRow, endCol);
   985→   985→
   986→   986→    // ── LAYER 2: Building sprites ──
   987→   987→    this._renderBuildings(ctx, camX, camY);
   988→   988→
   989→   989→    // ── LAYER 3: Props (below characters) ──
   990→   990→    this._renderProps(ctx, camX, camY);
   991→   991→
   992→   992→    // ── LAYER 4: Y-sorted characters (NPCs + Player) ──
   993→   993→    this._renderCharacters(ctx, camX, camY);
   994→   994→
   995→   995→    // ── LAYER 5: Overlay (shadows) ──
   996→   996→    this._renderOverlay(ctx, camX, camY, startRow, endRow, endCol);
   997→   997→
   998→   998→    // ── Copy to display canvas ──
   999→   999→    this.ctx.imageSmoothingEnabled = false;
  1000→  1000→    this.ctx.drawImage(this.offscreen!, 0, 0);
  1001→  1001→  }

  1003→  1003→  private _renderGroundTiles(
  1004→  1004→    ctx: CanvasRenderingContext2D, camX: number, camY: number,
  1005→  1005→    _startRow: number, startRow: number, endRow: number, endCol: number
  1006→  1006→  ): void {
  1007→  1007→    for (let row = startRow; row <= endRow && row < this.mapHeight; row++) {
  1008→  1008→      if (row < 0) continue;
  1009→  1009→      for (let col = 0; col <= endCol && col < this.mapWidth; col++) {
  1010→  1010→        if (col < 0) continue;
  1011→  1011→
  1012→  1012→        const tileType = this.groundLayer[row]?.[col] || 0;
  1013→  1013→        if (tileType === 0) continue; // Empty tile
  1014→  1014→
  1015→  1015→        const screenX = col * TILE_SIZE - camX;
  1016→  1016→        const screenY = row * TILE_SIZE - camY;
  1017→  1017→
  1018→  1018→        // Render tile image from assetManager
  1019→  1019→        const tileAsset = assetManager.getTile(tileType);
  1020→  1020→        if (tileAsset) {
  1021→  1021→          ctx.drawImage(tileAsset.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1022→  1022→        } else {
  1023→  1023→          // Fallback: colored rectangle
  1024→  1024→          const fallbackColors: Record<number, string> = {
  1025→  1025→            1: '#C4A76C', 2: '#8FBC8F', 3: '#D2B48C', 4: '#A0522D',
  1026→  1026→            5: '#696969', 6: '#8B4513', 7: '#6B8E6B', 8: '#555555',
  1027→  1027→            9: '#B8A070', 10: '#4A90A4', 11: '#7CB342', 12: '#00000033',
  1028→  1028→          };
  1029→  1029→          ctx.fillStyle = fallbackColors[tileType] || '#FF00FF';
  1030→  1030→          ctx.fillRect(screenX, screenY, TILE_SIZE, TILE_SIZE);
  1031→  1031→        }
  1032→  1032→
  1033→  1033→        // Also check buildings layer (render building tile type on ground)
  1034→  1034→        const buildingType = this.buildingsLayer[row]?.[col] || 0;
  1035→  1035→        if (buildingType > 0 && buildingType !== 5 && buildingType !== 6) {
  1036→  1036→          const buildingTile = assetManager.getTile(buildingType);
  1037→  1037→          if (buildingTile) {
  1038→  1038→            ctx.drawImage(buildingTile.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1039→  1039→          }
  1040→  1040→        }
  1041→  1041→      }
  1042→  1042→    }
  1043→  1043→  }

  1084→  1084→  private _renderCharacters(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {
  1085→  1085→    // Collect all renderable characters with their Y position for sorting
  1086→  1086→    interface Renderable {
  1087→  1087→      type: 'player' | 'npc';
  1088→  1088→      y: number;
  1089→  1089→      data: any;
  1090→  1090→    }
  1091→  1091→
  1092→  1092→    const renderables: Renderable[] = [];
  1093→  1093→
  1094→  1094→    // Add player
  1095→  1095→    renderables.push({
  1096→  1096→      type: 'player',
  1097→  1097→      y: this.player.pixelY + TILE_SIZE, // Bottom of sprite for Y-sorting
  1098→  1098→      data: this.player,
  1099→  1099→    });
  1100→  1100→
  1101→  1101→    // Add visible NPCs
  1102→  1102→    for (const npc of this.npcs) {
  1103→  1103→      if (!npc.visible) continue;
  1104→  1104→      renderables.push({
  1105→  1105→        type: 'npc',
  1106→  1106→        y: npc.pixelY + TILE_SIZE,
  1107→  1107→        data: npc,
  1108→  1108→      });
  1109→  1109→    }
  1110→  1110→
  1111→  1111→    // Sort by Y position (top to bottom)
  1112→  1112→    renderables.sort((a, b) => a.y - b.y);
  1113→  1113→
  1114→  1114→    // Render in sorted order
  1115→  1115→    for (const r of renderables) {
  1116→  1116→      if (r.type === 'player') {
  1117→  1117→        this._renderPlayer(ctx, camX, camY);
  1118→  1118→      } else {
  1119→  1119→        this._renderNPC(ctx, r.data as NPCState, camX, camY);
  1120→  1120→      }
  1121→  1121→    }
  1122→  1122→  }

  1124→  1124→  private _renderPlayer(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {
  1125→  1125→    const screenX = this.player.pixelX - camX;
  1126→  1126→    const screenY = this.player.pixelY - camY;
  1127→  1127→
  1128→  1128→    // Try to get player sprite from spriteLoader
  1129→  1129→    const facing = DIR_TO_FACING[this.player.facing];
  1130→  1130→    const isMoving = this.player.isSliding;
  1131→  1131→    const animName = isMoving ? 'Walk' : 'Breathing_Idle';
  1132→  1132→
  1133→  1133→    const frameCount = spriteLoader.getAnimationFrameCount('student', animName);
  1134→  1134→    const frameIndex = frameCount > 0 ? this.player.animFrame % frameCount : 0;
  1135→  1135→
  1136→  1136→    const frame = spriteLoader.getFrame('student', animName, facing, frameIndex);
  1137→  1137→
  1138→  1138→    if (frame) {
  1139→  1139→      // Draw the sprite frame scaled to CHAR_RENDER_SIZE
  1140→  1140→      const drawSize = CHAR_RENDER_SIZE;
  1141→  1141→      const offsetX = (TILE_SIZE - drawSize) / 2;
  1142→  1142→      const offsetY = TILE_SIZE - drawSize; // Character bottom aligns with tile bottom
  1143→  1143→
  1144→  1144→      ctx.drawImage(
  1145→  1145→        frame,
  1146→  1146→        0, 0, frame.width, frame.height,
  1147→  1147→        screenX + offsetX, screenY + offsetY, drawSize, drawSize
  1148→  1148→      );
  1149→  1149→    } else {
  1150→  1150→      // Fallback: draw a colored character
  1151→  1151→      this._drawPlaceholderCharacter(ctx, screenX, screenY, '#4A90D9', 'P');
  1152→  1152→    }
  1153→  1153→  }

  1160→  1160→    if (npc.spriteSheet) {

  1162→  1162→      if (spriteSet) {
  1163→  1163→        const facing = DIR_TO_FACING[npc.facing];

  1166→  1166→
  1167→  1167→        if (dirFrames && dirFrames.length > 0) {
  1168→  1168→          const frameIndex = npc.animFrame % dirFrames.length;

  1172→  1172→          const drawWidth = CHAR_RENDER_SIZE;

  1174→  1174→          const offsetX = (TILE_SIZE - drawWidth) / 2;
  1175→  1175→          const offsetY = TILE_SIZE - drawHeight;

  1177→  1177→          ctx.drawImage(

  1186→  1186→
  1187→  1187→    // Try spriteLoader as fallback (for ibara)

  1189→  1189→    const facing = DIR_TO_FACING[npc.facing];

  1194→  1194→      const offsetX = (TILE_SIZE - drawSize) / 2;
  1195→  1195→      const offsetY = TILE_SIZE - drawSize;
  1196→  1196→
  1197→  1197→      ctx.drawImage(
  1198→  1198→        frame,
  1199→  1199→        0, 0, frame.width, frame.height,
  1200→  1200→        screenX + offsetX, screenY + offsetY, drawSize, drawSize
  1201→  1201→      );
  1202→  1202→    } else {
  1203→  1203→      // Final fallback: colored placeholder
  1204→  1204→      const npcColors: Record<string, string> = {
  1205→  1205→        'mang-tenyo': '#8B4513',
  1206→  1206→        'vendor-1': '#D2691E',
