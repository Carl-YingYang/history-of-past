   128→// Deterministic hash for tile decoration patterns
   129→function tileHash(row: number, col: number, seed: number = 0): number {
   130→  let h = seed + row * 374761393 + col * 668265263;
   131→  h = (h ^ (h >> 13)) * 1274126177;
   132→  h = h ^ (h >> 16);
   133→  return (h & 0x7FFFFFFF) / 0x7FFFFFFF; // Returns 0..1
   134→}

   224→  private _computeBuildingGroups(): void {
   225→    // For each building tile type, find contiguous rectangular groups
   226→    const buildingTypes = [5, 6]; // Church and Market building types
   227→    for (const tileType of buildingTypes) {
   228→      const visited: Set<string> = new Set();
   229→      const groups: { minRow: number; minCol: number; maxRow: number; maxCol: number }[] = [];
   230→      
   231→      for (let row = 0; row < mapData.height; row++) {
   232→        for (let col = 0; col < mapData.width; col++) {
   233→          const key = `${row},${col}`;
   234→          if (visited.has(key)) continue;
   235→          if (mapData.layers.buildings[row]?.[col] !== tileType) continue;
   236→          
   237→          // Find bounding box of contiguous group
   238→          let minRow = row, maxRow = row, minCol = col, maxCol = col;
   239→          const toVisit: [number, number][] = [[row, col]];
   240→          
   241→          while (toVisit.length > 0) {
   242→            const [r, c] = toVisit.pop()!;
   243→            const k = `${r},${c}`;
   244→            if (visited.has(k)) continue;
   245→            if (mapData.layers.buildings[r]?.[c] !== tileType) continue;
   246→            visited.add(k);
   247→            minRow = Math.min(minRow, r);
   248→            maxRow = Math.max(maxRow, r);
   249→            minCol = Math.min(minCol, c);
   250→            maxCol = Math.max(maxCol, c);
   251→            toVisit.push([r - 1, c], [r + 1, c], [r, c - 1], [r, c + 1]);
   252→          }
   253→          
   254→          groups.push({ minRow, minCol, maxRow, maxCol });
   255→        }
   256→      }
   257→      
   258→      this.buildingGroups.set(tileType, groups);
   259→    }
   260→  }

   262→  async init(canvas: HTMLCanvasElement): Promise<void> {
   263→    this.canvas = canvas;
   264→    this.ctx = canvas.getContext('2d')!;
   265→    
   266→    // Set canvas size
   267→    this.resizeCanvas();
   268→    window.addEventListener('resize', () => this.resizeCanvas());
   269→
   270→    // Load all required sprites
   271→    const charsToLoad = ['student', 'ibara', 'mang-tenyo', 'kitchen-staff-1', 'kitchen-staff-2'];
   272→    for (const charId of charsToLoad) {
   273→      await spriteLoader.loadCharacter(charId);
   274→    }
   275→
   276→    // Create player entity
   277→    const spawn = mapData.playerSpawn;
   278→    this.player = {
   279→      id: 'player',
   280→      x: spawn.col * this.tileSize + this.tileSize / 2,
   281→      y: spawn.row * this.tileSize + this.tileSize / 2,
   282→      direction: 'north',
   283→      characterKey: 'student',
   284→      isMoving: false,
   285→      animationFrame: 0,
   286→      animationTimer: 0,
   287→      speed: 120,
   288→      visible: true,
   289→      isPlayer: true,
   290→      dialogueTriggered: false,
   291→    };
   292→    this.entities.set('player', this.player);
   293→
   294→    // Create NPC entities
   295→    const npcPositions = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   296→    for (const [npcId, pos] of Object.entries(npcPositions)) {
   297→      const entity: Entity = {
   298→        id: npcId,
   299→        x: pos.col * this.tileSize + this.tileSize / 2,
   300→        y: pos.row * this.tileSize + this.tileSize / 2,
   301→        direction: pos.direction as Direction,
   302→        characterKey: npcId,
   303→        isMoving: false,
   304→        animationFrame: 0,
   305→        animationTimer: 0,
   306→        speed: 0,
   307→        visible: !pos.appearsAfter || saveManager.isObjectiveCompleted(pos.appearsAfter),
   308→        isPlayer: false,
   309→        dialogueTriggered: false,
   310→        appearsAfterObjective: pos.appearsAfter,
   311→      };
   312→      this.entities.set(npcId, entity);
   313→    }
   314→
   315→    // Load saved progress
   316→    await saveManager.loadProgress();
   317→    const gameState = saveManager.getGameState();
   318→    this.chapterPhase = gameState.chapterPhase;
   319→    this.timeOfDay = gameState.timeOfDay;
   320→
   321→    // Sync the nature ambient mode with the loaded time of day. The actual
   322→    // ambient loop is started later by soundManager.initOnUserGesture()
   323→    // (called from page.tsx on the first user gesture, per browser autoplay
   324→    // policy), but we set the correct mode up-front so the first chirps match
   325→    // the in-game time.
   326→    soundManager.setNatureMode(this.timeOfDay === 'morning' || this.timeOfDay === 'afternoon' ? 'day' : 'night');
   327→    
   328→    if (this.chapterPhase !== 'intro') {
   329→      this.introShown = true;
   330→    }
   331→
   332→    // Update entity visibility based on completed objectives
   333→    this._updateEntityVisibility();
   334→
   335→    // Mark already-triggered zones
   336→    this.triggerZones.forEach(zone => {
   337→      if (zone.requiresObjective && saveManager.isObjectiveCompleted(zone.requiresObjective)) {
   338→        // The prerequisite is done, but this zone itself may not have been triggered yet
   339→        // Only mark as triggered if the objective THIS zone completes is also done
   340→      }
   341→    });
   342→
   343→    // Bind input handlers
   344→    this._bindInput();
   345→
   346→    // Listen for objective completion events to spawn celebration bursts
   347→    gameEvents.on('quest:objectiveComplete', (objectiveId: unknown) => {
   348→      const id = objectiveId as string;
   349→      const loc = OBJECTIVE_LOCATIONS[id];
   350→      if (loc) {
   351→        // Spawn burst at the objective's location in world coords
   352→        this._spawnCelebrationBurst(
   353→          loc.col * this.tileSize + this.tileSize / 2,
   354→          loc.row * this.tileSize + this.tileSize / 2
   355→        );
   356→      }
   357→    });
   358→
   359→    // === Ambient sound system event hooks (Task 9-b) ===
   360→    // Ring the church bell when a chapter completes. The soundManager handles
   361→    // its own gating (soundEnabled / ambientEnabled) so we always emit.
   362→    gameEvents.on('chapter:complete', () => {
   363→      soundManager.playChurchBell();
   364→    });
   365→
   366→    // Keep the nature ambient mode in sync with the in-game time of day.
   367→    // Currently the engine only models 'morning' and 'afternoon' (both 'day'),
   368→    // but we still re-affirm the mode so future 'evening'/'night' states would
   369→    // automatically switch to cricket ambience.
   370→    gameEvents.on('time:transition', (time: unknown) => {
   371→      const t = time as 'morning' | 'afternoon' | 'evening' | 'night';
   372→      const mode: 'day' | 'night' = (t === 'evening' || t === 'night') ? 'night' : 'day';
   373→      soundManager.setNatureMode(mode);
   374→    });
   375→
   376→    // Start auto-save
   377→    saveManager.startAutoSave();
   378→
   379→    // Emit game ready
   380→    gameEvents.emit('game:ready');
   381→
   382→    // Initialize some particles
   383→    this._initParticles();
   384→
   385→    // Initialize fireflies (persistent ambient atmosphere)
   386→    this._initFireflies();
   387→
   388→    // Start game loop
   389→    this.lastTimestamp = performance.now();
   390→    this._gameLoop(this.lastTimestamp);
   391→  }

   393→  /**
   394→   * Spawn a golden celebration burst at the given world position.
   395→   * Used when an objective is completed.
   396→   */
   397→  private _spawnCelebrationBurst(worldX: number, worldY: number): void {
   398→    const colors = ['#FFD700', '#FFA500', '#FFEC8B', '#FFEE00', '#FFC125'];
   399→    const count = 28;
   400→    for (let i = 0; i < count; i++) {
   401→      const angle = (i / count) * Math.PI * 2 + Math.random() * 0.3;
   402→      const speed = 60 + Math.random() * 80;
   403→      this.burstParticles.push({
   404→        x: worldX,
   405→        y: worldY,
   406→        vx: Math.cos(angle) * speed,
   407→        vy: Math.sin(angle) * speed - 30, // Slight upward bias
   408→        life: 0,
   409→        maxLife: 1.0 + Math.random() * 0.6,
   410→        color: colors[Math.floor(Math.random() * colors.length)],
   411→        size: 2 + Math.random() * 3,
   412→        gravity: 80,
   413→      });
   414→    }
   415→  }
   416→
   417→  private _updateBurstParticles(dt: number): void {
   418→    for (let i = this.burstParticles.length - 1; i >= 0; i--) {
   419→      const p = this.burstParticles[i];
   420→      p.life += dt;
   421→      p.vy += p.gravity * dt;     // gravity pulls down
   422→      p.x += p.vx * dt;
   423→      p.y += p.vy * dt;
   424→      // Air resistance
   425→      p.vx *= 0.96;
   426→      p.vy *= 0.96;
   427→      if (p.life >= p.maxLife) {
   428→        this.burstParticles.splice(i, 1);
   429→      }
   430→    }
   431→  }
   432→
   433→  private _renderBurstParticles(ctx: CanvasRenderingContext2D): void {
   434→    for (const p of this.burstParticles) {
   435→      const screenX = p.x - this.cameraX;
   436→      const screenY = p.y - this.cameraY;
   437→      if (screenX < -20 || screenX > this.viewWidth + 20 || screenY < -20 || screenY > this.viewHeight + 20) continue;
   438→      const lifeRatio = p.life / p.maxLife;
   439→      const alpha = Math.max(0, 1 - lifeRatio);
   440→      // Glow
   441→      ctx.globalAlpha = alpha * 0.4;
   442→      ctx.fillStyle = p.color;
   443→      ctx.beginPath();
   444→      ctx.arc(screenX, screenY, p.size * 2.5, 0, Math.PI * 2);
   445→      ctx.fill();
   446→      // Core
   447→      ctx.globalAlpha = alpha;
   448→      ctx.fillStyle = p.color;
   449→      ctx.beginPath();
   450→      ctx.arc(screenX, screenY, p.size, 0, Math.PI * 2);
   451→      ctx.fill();
   452→    }
   453→    ctx.globalAlpha = 1;
   454→  }

   501→  triggerInteract(): void {

   505→  private _handleInteract(): void {
   506→    if (this.dialogueActive) return;
   507→    if (this.interactCooldown > 0) return;
   508→    
   509→    // Check if player is near an NPC they can talk to
   510→    if (!this.player) return;
   511→
   512→    for (const [npcId, npc] of this.entities) {
   513→      if (npc.isPlayer || !npc.visible) continue;
   514→      
   515→      const dist = this._distanceBetween(this.player, npc);
   516→      if (dist < this.tileSize * 2) {
   517→        this.interactCooldown = 500;
   518→        // Track NPC interaction for exploration stats
   519→        this.npcsTalkedTo.add(npcId);
   520→        
   521→        // Determine which dialogue to trigger
   522→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   523→          this._startDialogue('mang-tenyo-first');
   524→          return;
   525→        }
   526→        // Mang Tenyo can be talked to again — different dialogue based on progress
   527→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   528→          // If gossip objective completed, give more insightful dialogue
   529→          if (saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   530→            this._startDialogue('mang-tenyo-after-gossip');
   531→          } else {
   532→            this._startDialogue('mang-tenyo-repeat');
   533→          }
   534→          return;
   535→        }
   536→      }
   537→    }
   538→  }

   540→  private _distanceBetween(a: Entity, b: Entity): number {
   541→    const dx = a.x - b.x;
   542→    const dy = a.y - b.y;
   543→    return Math.sqrt(dx * dx + dy * dy);
   544→  }

   546→  private _startDialogue(dialogueId: string): void {
   547→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   548→    if (!dialogue) {
   549→      console.warn(`Dialogue not found: ${dialogueId}`);
   550→      return;
   551→    }
   552→
   553→    this.dialogueActive = true;
   554→    this.currentDialogueId = dialogueId;
   555→    this.currentDialogueLineIndex = 0;
   556→
   557→    gameEvents.emit('dialogue:start', {
   558→      dialogueId,
   559→      line: dialogue.lines[0],
   560→      lineIndex: 0,
   561→      totalLines: dialogue.lines.length,
   562→    });
   563→  }

   565→  advanceDialogue(): void {
   566→    if (!this.currentDialogueId) return;
   567→    
   568→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];
   569→    if (!dialogue) return;
   570→
   571→    this.currentDialogueLineIndex++;
   572→
   573→    if (this.currentDialogueLineIndex >= dialogue.lines.length) {
   574→      // Dialogue complete
   575→      this._endDialogue(dialogue);
   576→      return;
   577→    }
   578→
   579→    const line = dialogue.lines[this.currentDialogueLineIndex];
   580→    
   581→    // Check for time transition
   582→    if (line.isTimeTransition) {
   583→      this.timeOfDay = 'morning';
   584→      this.chapterPhase = 'ibarra-sighting';
   585→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   586→      gameEvents.emit('time:transition', 'morning');
   587→    }
   588→
   589→    // Check for Ibarra sighting
   590→    if (line.isIbarraSighting) {
   591→      this.ibarraAppeared = true;
   592→      // Show Ibarra NPC
   593→      const ibaraEntity = this.entities.get('ibara');
   594→      if (ibaraEntity) {
   595→        ibaraEntity.visible = true;
   596→      }
   597→    }
   598→
   599→    gameEvents.emit('dialogue:line', {
   600→      dialogueId: this.currentDialogueId,
   601→      line,
   602→      lineIndex: this.currentDialogueLineIndex,
   603→      totalLines: dialogue.lines.length,
   604→    });
   605→  }

   655→        'ch1'

   730→  private _gameLoop(timestamp: number): void {
   731→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   732→    this.lastTimestamp = timestamp;
   733→    this.gameTime += deltaTime;
   734→
   735→    this._update(deltaTime);
   736→    this._render();
   737→
   738→    this.animationFrameId = requestAnimationFrame(this._gameLoop.bind(this));
   739→  }

   741→  private _update(dt: number): void {
   742→    if (!this.player) return;
   743→    
   744→    // Update interact cooldown
   745→    if (this.interactCooldown > 0) {
   746→      this.interactCooldown -= dt * 1000;
   747→    }
   748→
   749→    // Don't allow movement during dialogue
   750→    if (this.dialogueActive) return;
   751→
   752→    // Handle intro state
   753→    if (this.chapterPhase === 'intro' && !this.introShown) {
   754→      this.introShown = true;
   755→      this._startDialogue('intro-text');
   756→      return;
   757→    }
   758→
   759→    // Player movement
   760→    let moveX = 0;
   761→    let moveY = 0;
   762→
   763→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   764→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   765→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   766→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   767→
   768→    // Normalize diagonal movement
   769→    if (moveX !== 0 && moveY !== 0) {
   770→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   771→      moveX /= len;
   772→      moveY /= len;
   773→    }
   774→
   775→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   776→
   777→    if (this.player.isMoving) {
   778→      // Determine direction from movement vector
   779→      this.player.direction = this._vectorToDirection(moveX, moveY);
   780→      
   781→      const newX = this.player.x + moveX * this.player.speed * dt;
   782→      const newY = this.player.y + moveY * this.player.speed * dt;
   783→
   784→      // Collision check
   785→      const blocked = this._isBlocked(newX, newY);
   786→      if (!blocked) {
   787→        this.player.x = newX;
   788→        this.player.y = newY;
   789→
   790→        // === Footstep ambient sound (Task 9-b) ===
   791→        // Throttle to ~400ms between steps. Only play when the player actually
   792→        // moved (not when blocked by a wall) so shuffling into a corner doesn't
   793→        // spam footsteps.
   794→        const nowMs = performance.now();
   795→        if (nowMs - this.lastFootstepTime >= 400) {
   796→          this.lastFootstepTime = nowMs;
   797→          soundManager.playFootstep();
   798→        }
   799→      }
   800→
   801→      // Update animation
   802→      this.player.animationTimer += dt;
   803→      const animConfig = characterData.characters.student.animations.Walk;
   804→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   805→        this.player.animationTimer = 0;
   806→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   807→      }
   808→    } else {
   809→      // Idle animation
   810→      this.player.animationTimer += dt;
   811→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   812→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   813→        this.player.animationTimer = 0;
   814→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   815→      }
   816→    }
   817→
   818→    // Update camera to follow player
   819→    this.cameraX = this.player.x - this.viewWidth / 2;
   820→    this.cameraY = this.player.y - this.viewHeight / 2;
   821→
   822→    // Clamp camera to map bounds
   823→    const mapPixelWidth = mapData.width * this.tileSize;
   824→    const mapPixelHeight = mapData.height * this.tileSize;
   825→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   826→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   827→
   828→    // NPC idle animations
   829→    for (const [npcId, npc] of this.entities) {
   830→      if (npc.isPlayer || !npc.visible) continue;
   831→      npc.animationTimer += dt;
   832→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   833→      if (animConfig) {
   834→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   835→          npc.animationTimer = 0;
   836→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   837→        }
   838→      }
   839→    }
   840→
   841→    // Check trigger zones
   842→    this._checkTriggerZones();
   843→
   844→    // === Market ambient sound proximity check (Task 9-b) ===
   845→    // If the player is within 4 tiles (Chebyshev distance) of the Market
   846→    // building's bounding box, ensure the market ambient loop is running;
   847→    // otherwise stop it. The soundManager.start/stopMarketAmbient() calls are
   848→    // idempotent so it's safe to call them every frame.
   849→    this._updateMarketAmbient();
   850→
   851→    // Update particles
   852→    this._updateParticles(dt);
   853→
   854→    // Update fireflies (ambient glowing insects)
   855→    this._updateFireflies(dt);
   856→
   857→    // Update drifting leaves (occasional foliage)
   858→    this._updateDriftLeaves(dt);
   859→
   860→    // Update burst particles (celebration effect)
   861→    this._updateBurstParticles(dt);
   862→
   863→    // Update game state in SaveManager
   864→    saveManager.updateGameState({
   865→      playerPosition: {
   866→        row: Math.floor(this.player.y / this.tileSize),
   867→        col: Math.floor(this.player.x / this.tileSize),
   868→      },
   869→      playerDirection: this.player.direction,
   870→      currentDialogue: this.currentDialogueId,
   871→      timeOfDay: this.timeOfDay,
   872→      chapterPhase: this.chapterPhase,
   873→    });
   874→
   875→    // Track exploration stats — count unique tiles visited
   876→    const tileRow = Math.floor(this.player.y / this.tileSize);
   877→    const tileCol = Math.floor(this.player.x / this.tileSize);
   878→    const tileKey = `${tileRow},${tileCol}`;
   879→    this.visitedTiles.add(tileKey);
   880→
   881→    // Flush stats to localStorage every ~2s (debounced)
   882→    this.statsFlushTimer += dt;
   883→    if (this.statsFlushTimer > 2.0) {
   884→      this.statsFlushTimer = 0;
   885→      try {
   886→        localStorage.setItem('noor-stats', JSON.stringify({
   887→          tilesExplored: this.visitedTiles.size,
   888→          npcsTalkedTo: this.npcsTalkedTo.size,
   889→        }));
   890→      } catch {
   891→        // localStorage may be unavailable; ignore
   892→      }
   893→    }
   894→  }
   895→
   896→  private _vectorToDirection(x: number, y: number): Direction {
   897→    // Map movement vector to 8-direction name
   898→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   899→    
   900→    if (angle >= -22.5 && angle < 22.5) return 'east';
   901→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   902→    if (angle >= 67.5 && angle < 112.5) return 'south';
   903→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   904→    if (angle >= 157.5 || angle < -157.5) return 'west';
   905→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   906→    if (angle >= -112.5 && angle < -67.5) return 'north';
   907→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   908→    return 'south';
   909→  }

   911→  private _isBlocked(x: number, y: number): boolean {
   912→    // Check collision map
   913→    const col = Math.floor(x / this.tileSize);
   914→    const row = Math.floor(y / this.tileSize);
   915→    
   916→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   917→    
   918→    // Check player sprite footprint (check a small area around center)
   919→    const checks = [
   920→      { x: x - 16, y: y - 16 },
   921→      { x: x + 16, y: y - 16 },
   922→      { x: x - 16, y: y + 16 },
   923→      { x: x + 16, y: y + 16 },
   924→    ];
   925→    
   926→    for (const check of checks) {
   927→      const cCol = Math.floor(check.x / this.tileSize);
   928→      const cRow = Math.floor(check.y / this.tileSize);
   929→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   930→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   931→    }
   932→    
   933→    return false;
   934→  }

   936→  private _checkTriggerZones(): void {
   937→    if (!this.player || this.dialogueActive) return;
   938→    
   939→    const playerRow = Math.floor(this.player.y / this.tileSize);
   940→    const playerCol = Math.floor(this.player.x / this.tileSize);
   941→
   942→    for (const zone of this.triggerZones) {
   943→      if (zone.triggered) continue;
   944→
   945→      // Check prerequisite objective
   946→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   947→
   948→      // Check if player is in the zone
   949→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   950→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   951→
   952→      // Or within radius of NPC
   953→      let nearNpc = false;
   954→      if (zone.radius > 0 && zone.npcId) {
   955→        const npc = this.entities.get(zone.npcId);
   956→        if (npc && npc.visible) {
   957→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   958→        }
   959→      }
   960→
   961→      if (inZone || nearNpc) {
   962→        zone.triggered = true;
   963→
   964→        switch (zone.type) {
   965→          case 'npc-interact':
   966→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   967→            break;
   968→          case 'auto-trigger':
   969→            // Generic auto-trigger: if zone has a dialogueId, start that dialogue
   970→            if (zone.dialogueId) {
   971→              // Special handling for Ibarra sighting (time transition)
   972→              if (zone.id === 'ibarra-sighting') {
   973→                this.timeOfDay = 'morning';
   974→                this.chapterPhase = 'ibarra-sighting';
   975→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   976→                this._updateEntityVisibility();
   977→              }
   978→              this._startDialogue(zone.dialogueId);
   979→            } else {
   980→              // Legacy hardcoded triggers as fallback
   981→              if (zone.id === 'market-gossip') {
   982→                this._startDialogue('market-gossip');
   983→              } else if (zone.id === 'ibarra-sighting') {
   984→                this.timeOfDay = 'morning';
   985→                this.chapterPhase = 'ibarra-sighting';
   986→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   987→                this._updateEntityVisibility();
   988→                this._startDialogue('ibarra-sighting');
   989→              }
   990→            }
   991→            break;
   992→        }
   993→      }
   994→    }
   995→  }
   996→
   997→  /**
   998→   * Start or stop the Market ambient sound loop based on the player's
   999→   * proximity to the Market building. The player must be within 4 tiles
  1000→   * (Chebyshev distance) of the building's bounding box.
  1001→   *
  1002→   * The soundManager calls are idempotent — startMarketAmbient() is a no-op
  1003→   * if already running, stopMarketAmbient() is a no-op if already stopped —
  1004→   * so it's safe to call this every frame.
  1005→   */
  1006→  private _updateMarketAmbient(): void {
  1007→    if (!this.player || !this.marketBounds) {
  1008→      soundManager.stopMarketAmbient();
  1009→      return;
  1010→    }
  1011→
  1012→    const playerRow = Math.floor(this.player.y / this.tileSize);
  1013→    const playerCol = Math.floor(this.player.x / this.tileSize);
  1014→    const mb = this.marketBounds;
  1015→
  1016→    // Chebyshev distance from the player's tile to the Market bounding box.
  1017→    // 0 means the player is standing on/in the market; >0 means they're that
  1018→    // many tiles away from the nearest edge.
  1019→    const dx = Math.max(mb.col - playerCol, 0, playerCol - (mb.col + mb.width - 1));
  1020→    const dy = Math.max(mb.row - playerRow, 0, playerRow - (mb.row + mb.height - 1));
  1021→    const dist = Math.max(dx, dy);
  1022→
  1023→    if (dist <= 4) {
  1024→      soundManager.startMarketAmbient();
  1025→    } else {
  1026→      soundManager.stopMarketAmbient();
  1027→    }
  1028→  }

  1032→  private _initParticles(): void {
  1033→    // Seed some initial particles
  1034→    for (let i = 0; i < 15; i++) {
  1035→      this._spawnParticle();

  1038→
  1039→  private _spawnParticle(): void {
  1040→    if (!this.player) return;
  1041→    // Spawn particle near the camera view area
  1042→    const cx = this.cameraX + this.viewWidth / 2;
  1043→    const cy = this.cameraY + this.viewHeight / 2;
  1044→    this.particles.push({
  1045→      x: cx + (Math.random() - 0.5) * this.viewWidth * 0.8,
  1046→      y: cy + (Math.random() - 0.5) * this.viewHeight * 0.8,
  1047→      vx: (Math.random() - 0.5) * 8, // Slow drift
  1048→      vy: -Math.random() * 4 - 1, // Slight upward drift
  1049→      alpha: 0,
  1050→      size: Math.random() * 2.5 + 1,
  1051→      life: 0,
  1052→      maxLife: Math.random() * 6 + 4, // 4-10 seconds
  1053→    });
  1054→  }
  1055→
  1056→  private _updateParticles(dt: number): void {
  1057→    // Spawn new particles periodically
  1058→    this.particleSpawnTimer += dt;
  1059→    if (this.particleSpawnTimer > 0.3 && this.particles.length < 25) {
  1060→      this.particleSpawnTimer = 0;
  1061→      this._spawnParticle();
  1062→    }
  1063→
  1064→    // Update existing particles
  1065→    for (let i = this.particles.length - 1; i >= 0; i--) {
  1066→      const p = this.particles[i];
  1067→      p.life += dt;
  1068→      p.x += p.vx * dt;
  1069→      p.y += p.vy * dt;
  1070→      // Gentle oscillation
  1071→      p.x += Math.sin(this.gameTime * 0.5 + i) * 0.3;
  1072→
  1073→      // Fade in then fade out
  1074→      const lifeRatio = p.life / p.maxLife;
  1075→      if (lifeRatio < 0.2) {
  1076→        p.alpha = lifeRatio / 0.2 * 0.4; // Fade in
  1077→      } else if (lifeRatio > 0.7) {
  1078→        p.alpha = (1 - lifeRatio) / 0.3 * 0.4; // Fade out
  1079→      } else {
  1080→        p.alpha = 0.4; // Visible
  1081→      }
  1082→
  1083→      // Remove dead particles

  1092→  private _initFireflies(): void {
  1093→    // Pre-spawn fireflies scattered around the map (persistent)
  1094→    if (!this.player) return;
  1095→    const MAP_W = mapData.width * this.tileSize;
  1096→    const MAP_H = mapData.height * this.tileSize;
  1097→    for (let i = 0; i < 18; i++) {
  1098→      const fx = Math.random() * MAP_W;
  1099→      const fy = Math.random() * MAP_H;
  1100→      this.fireflies.push({
  1101→        x: fx,
  1102→        y: fy,
  1103→        baseX: fx,
  1104→        baseY: fy,
  1105→        phase: Math.random() * Math.PI * 2,
  1106→        blinkSpeed: 0.8 + Math.random() * 1.6, // 0.8-2.4 rad/s
  1107→        driftPhase: Math.random() * Math.PI * 2,
  1108→        brightness: 0,
  1109→      });
  1110→    }
  1111→  }
  1112→
  1113→  private _updateFireflies(dt: number): void {
  1114→    // Note: gameTime is already updated in the main _update() loop
  1115→
  1116→    // Occasionally relocate fireflies that are too far from the player's view
  1117→    // so we always have some nearby fireflies
  1118→    this.fireflySpawnTimer += dt;
  1119→    if (this.fireflySpawnTimer > 1.5 && this.player) {
  1120→      this.fireflySpawnTimer = 0;
  1121→      // Despawn fireflies that wandered too far
  1122→      this.fireflies = this.fireflies.filter(f => {
  1123→        const dx = f.x - this.player!.x;
  1124→        const dy = f.y - this.player!.y;
  1125→        return Math.sqrt(dx * dx + dy * dy) < this.viewWidth * 1.2;
  1126→      });
  1127→      // Spawn a few new ones near the player's view
  1128→      while (this.fireflies.length < 18 && this.player) {
  1129→        const angle = Math.random() * Math.PI * 2;
  1130→        const radius = this.viewWidth * 0.4 + Math.random() * this.viewWidth * 0.3;
  1131→        const fx = this.player.x + Math.cos(angle) * radius;
  1132→        const fy = this.player.y + Math.sin(angle) * radius;
  1133→        this.fireflies.push({
  1134→          x: fx,
  1135→          y: fy,
  1136→          baseX: fx,
  1137→          baseY: fy,
  1138→          phase: Math.random() * Math.PI * 2,
  1139→          blinkSpeed: 0.8 + Math.random() * 1.6,
  1140→          driftPhase: Math.random() * Math.PI * 2,
  1141→          brightness: 0,
  1142→        });
  1143→      }
  1144→    }
  1145→
  1146→    // Update firefly positions and brightness
  1147→    for (const f of this.fireflies) {
  1148→      // Sinusoidal drift around the anchor point
  1149→      const driftRadius = 18;
  1150→      f.x = f.baseX + Math.cos(this.gameTime * 0.6 + f.driftPhase) * driftRadius;
  1151→      f.y = f.baseY + Math.sin(this.gameTime * 0.4 + f.driftPhase * 1.3) * driftRadius * 0.7;
  1152→
  1153→      // Slowly migrate the anchor point (gentle wandering)
  1154→      f.baseX += Math.sin(this.gameTime * 0.15 + f.driftPhase) * 0.15;
  1155→      f.baseY += Math.cos(this.gameTime * 0.18 + f.phase) * 0.1;
  1156→
  1157→      // Blink brightness using a smooth wave (0..1)
  1158→      // Use sin^4 for sharper on/off transitions
  1159→      const raw = Math.sin(this.gameTime * f.blinkSpeed + f.phase);
  1160→      const norm = (raw + 1) / 2; // 0..1
  1161→      f.brightness = Math.pow(norm, 3); // Sharp blink curve
  1162→    }
  1163→  }
  1164→
  1165→  private _renderFireflies(ctx: CanvasRenderingContext2D): void {
  1166→    // Fireflies only show during afternoon (golden hour) — they evoke a warm evening feel
  1167→    if (this.timeOfDay !== 'afternoon') return;
  1168→
  1169→    ctx.save();
  1170→    ctx.globalCompositeOperation = 'lighter'; // Additive blending for glow
  1171→
  1172→    for (const f of this.fireflies) {
  1173→      const screenX = f.x - this.cameraX;
  1174→      const screenY = f.y - this.cameraY;
  1175→
  1176→      // Skip off-screen fireflies
  1177→      if (screenX < -30 || screenX > this.viewWidth + 30 || screenY < -30 || screenY > this.viewHeight + 30) continue;
  1178→
  1179→      const b = f.brightness;
  1180→      if (b < 0.02) continue;
  1181→
  1182→      // Outer glow halo (large soft circle)
  1183→      const glowRadius = 12;
  1184→      const glowGrad = ctx.createRadialGradient(screenX, screenY, 0, screenX, screenY, glowRadius);
  1185→      glowGrad.addColorStop(0, `rgba(255, 240, 160, ${0.5 * b})`);
  1186→      glowGrad.addColorStop(0.3, `rgba(255, 220, 100, ${0.25 * b})`);

  1188→      ctx.fillStyle = glowGrad;
  1189→      ctx.beginPath();
  1190→      ctx.arc(screenX, screenY, glowRadius, 0, Math.PI * 2);
  1191→      ctx.fill();
  1192→
  1193→      // Core bright dot
  1194→      ctx.fillStyle = `rgba(255, 255, 220, ${0.95 * b})`;
  1195→      ctx.beginPath();
  1196→      ctx.arc(screenX, screenY, 1.5, 0, Math.PI * 2);
  1197→      ctx.fill();
  1198→    }
  1199→
  1200→    ctx.restore();

  1205→  private _updateDriftLeaves(dt: number): void {
  1206→    // Occasionally spawn a new leaf
  1207→    this.leafSpawnTimer += dt;
  1208→    if (this.leafSpawnTimer > 2.5 + Math.random() * 3 && this.player && this.driftLeaves.length < 8) {
  1209→      this.leafSpawnTimer = 0;
  1210→      // Spawn from the right side (wind blowing from east to west)
  1211→      const spawnX = this.player.x + this.viewWidth * 0.6 + Math.random() * 100;
  1212→      const spawnY = this.player.y - this.viewHeight * 0.3 + Math.random() * this.viewHeight * 0.6;
  1213→      const colors = ['#8B5A2B', '#A0522D', '#6B4423', '#C17840', '#9C6B3F'];
  1214→      this.driftLeaves.push({
  1215→        x: spawnX,
  1216→        y: spawnY,
  1217→        vx: -25 - Math.random() * 20, // Wind blows leftward
  1218→        vy: 8 + Math.random() * 8,    // Gentle downward drift
  1219→        rotation: Math.random() * Math.PI * 2,
  1220→        rotationSpeed: (Math.random() - 0.5) * 3,
  1221→        size: 5 + Math.random() * 4,
  1222→        color: colors[Math.floor(Math.random() * colors.length)],
  1223→        alpha: 0,
  1224→        life: 0,
  1225→        maxLife: 8 + Math.random() * 4,
  1226→      });
  1227→    }
  1228→
  1229→    // Update leaves
  1230→    for (let i = this.driftLeaves.length - 1; i >= 0; i--) {
  1231→      const leaf = this.driftLeaves[i];
  1232→      leaf.life += dt;
  1233→
  1234→      // Leaf motion: drifts left and downward, with sinusoidal sway
  1235→      leaf.x += leaf.vx * dt + Math.sin(this.gameTime * 1.5 + i) * 0.5;
  1236→      leaf.y += leaf.vy * dt + Math.cos(this.gameTime * 1.2 + i * 0.7) * 0.3;
  1237→      leaf.rotation += leaf.rotationSpeed * dt;
  1238→
  1239→      // Fade in/out
  1240→      const lifeRatio = leaf.life / leaf.maxLife;
  1241→      if (lifeRatio < 0.15) {
  1242→        leaf.alpha = lifeRatio / 0.15 * 0.85;
  1243→      } else if (lifeRatio > 0.7) {
  1244→        leaf.alpha = (1 - lifeRatio) / 0.3 * 0.85;
  1245→      } else {
  1246→        leaf.alpha = 0.85;
  1247→      }
  1248→
  1249→      // Remove dead leaves
  1250→      if (leaf.life >= leaf.maxLife) {
  1251→        this.driftLeaves.splice(i, 1);
  1252→      }
  1253→    }
  1254→  }

  1363→  private _renderWaypointArrow(ctx: CanvasRenderingContext2D): void {
  1364→    if (!this.player || this.dialogueActive) return;
  1365→    if (this.chapterPhase === 'intro' || this.chapterPhase === 'complete') return;
  1366→

  1369→    const nextObjId = objectivesOrder.find(id => !saveManager.isObjectiveCompleted(id));
  1370→    if (!nextObjId) return;
  1371→
  1372→    const loc = OBJECTIVE_LOCATIONS[nextObjId];
  1373→    if (!loc) return;
  1374→
  1375→    const targetX = loc.col * this.tileSize + this.tileSize / 2;
  1376→    const targetY = loc.row * this.tileSize + this.tileSize / 2;
  1377→
  1378→    // Vector from player to target
  1379→    const dx = targetX - this.player.x;
  1380→    const dy = targetY - this.player.y;
  1381→    const dist = Math.sqrt(dx * dx + dy * dy);
  1382→
  1383→    // Don't show arrow if very close
  1384→    if (dist < this.tileSize * 2.5) return;
  1385→

  1387→    const screenTargetX = targetX - this.cameraX;
  1388→    const screenTargetY = targetY - this.cameraY;

  1391→    const cy = this.viewHeight / 2;

  1394→    const onScreen = screenTargetX > 40 && screenTargetX < this.viewWidth - 40 &&
  1395→                     screenTargetY > 80 && screenTargetY < this.viewHeight - 80;
  1396→    if (onScreen) return;
  1397→

  1399→    const angle = Math.atan2(dy, dx);
  1400→

  1404→    const maxRadiusY = this.viewHeight / 2 - margin;

  1407→    const cosA = Math.cos(angle);
  1408→    const sinA = Math.sin(angle);
  1409→    const t = 1 / Math.sqrt((cosA * cosA) / (maxRadiusX * maxRadiusX) + (sinA * sinA) / (maxRadiusY * maxRadiusY));
  1410→    const arrowX = cx + cosA * t;
  1411→    const arrowY = cy + sinA * t;

  1426→    // Arrow shadow

  1433→    ctx.closePath();
  1434→    ctx.fill();
  1435→
  1436→    // Arrow body
  1437→    ctx.fillStyle = `rgba(255,215,0,${pulse})`;
  1438→    ctx.strokeStyle = 'rgba(180,120,0,1)';
  1439→    ctx.lineWidth = 1.5;
  1440→    ctx.beginPath();
  1441→    ctx.moveTo(15, 0);
  1442→    ctx.lineTo(-10, -10);
  1443→    ctx.lineTo(-6, 0);
  1444→    ctx.lineTo(-10, 10);
  1445→    ctx.closePath();
  1446→    ctx.fill();
  1447→    ctx.stroke();
  1448→
  1449→    ctx.restore();
  1450→
  1451→    // Distance label below arrow
  1452→    const distTiles = Math.round(dist / this.tileSize);
  1453→    ctx.font = 'bold 10px "Geist", sans-serif';
  1454→    ctx.fillStyle = 'rgba(0,0,0,0.6)';

  1545→  private _renderGrass(ctx: CanvasRenderingContext2D, x: number, y: number, row: number, col: number): void {

  1556→    ctx.fillStyle = h2 < 0.3 ? '#5C8B4E' : h2 < 0.6 ? '#8BC47B' : '#6B9E5E';
