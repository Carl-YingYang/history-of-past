     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useMemo } from 'react';
     4→import { useUIStore } from './UIManager';
     5→import { useGameStore } from '@/stores/gameStore';
     6→import mapData from '@/data/mapData.json';
     7→import characterData from '@/data/characters.json';
     8→
     9→/**
    10→ * Minimap - Enhanced overview of the San Diego map with building labels,
    11→ * NPC markers, discovery markers, compass, zoom levels, and visual polish.
    12→ *
    13→ * Task 9-b additions:
    14→ *   - Undiscovered buildings render as a pulsing gray dashed circle with a
    15→ *     "?" glyph instead of their label, so the player can see there's
    16→ *     *something* there but hasn't found it yet.
    17→ *   - Hovering any building marker shows a tooltip ("Undiscovered location"
    18→ *     for ? markers, or the building name for discovered ones).
    19→ *   - Legend gets a "❓ Undiscovered" entry.
    20→ *   - Discovery counter at the bottom reflects the actual building count
    21→ *     from `mapData.buildingLabels` (currently 3).
    22→ */
    23→
    24→// Zoom level cell sizes
    25→const ZOOM_LEVELS: Record<string, { cellSize: number; label: string }> = {
    26→  'small':  { cellSize: 8,  label: 'S' },
    27→  'medium': { cellSize: 12, label: 'M' },
    28→  'large':  { cellSize: 16, label: 'L' },
    29→};
    30→
    31→// Shape of a discovery-log entry persisted by DiscoveryLogPanel under
    32→// 'noor-discovery-log'. Only the fields we actually read here.
    33→interface DiscoveryLogEntry {
    34→  id: string;
    35→  name: string;
    36→  type: string;
    37→  position: { x: number; y: number };
    38→  timestamp: number;
    39→  note?: string;
    40→}
    41→
    42→// Information about the building marker currently under the cursor. Used to
    43→// render a positioned tooltip overlay.
    44→interface HoveredMarker {
    45→  cssX: number;       // mouse X relative to canvas (CSS px)
    46→  cssY: number;       // mouse Y relative to canvas (CSS px)
    47→  label: string;
    48→  discovered: boolean;
    49→}
    50→
    51→/**
    52→ * Build the canonical discovery-log id for a building label, matching the
    53→ * format used by DiscoveryLogPanel/Minimap when dispatching 'noor:discovery'
    54→ * events: `bldg-<slugified-label>`.
    55→ */
    56→function getBuildingDiscoveryId(label: string): string {
    57→  return `bldg-${label.toLowerCase().replace(/[^a-z0-9]+/g, '-')}`;
    58→}
    59→
    60→/** Load the discovery log from localStorage (SSR-safe). */
    61→function loadDiscoveryLog(): DiscoveryLogEntry[] {
    62→  if (typeof window === 'undefined') return [];
    63→  try {
    64→    const raw = localStorage.getItem('noor-discovery-log');
    65→    if (!raw) return [];
    66→    const parsed = JSON.parse(raw);
    67→    if (!Array.isArray(parsed)) return [];
    68→    return parsed.filter(
    69→      (d): d is DiscoveryLogEntry =>
    70→        d && typeof d === 'object' && typeof d.id === 'string'
    71→    );
    72→  } catch {
    73→    return [];
    74→  }
    75→}
    76→
    77→// NPC color mapping from characters.json
    78→const NPC_COLORS: Record<string, string> = {};
    79→for (const [id, char] of Object.entries(characterData.characters)) {
    80→  if (char.placeholder && char.placeholderColor) {
    81→    NPC_COLORS[id] = char.placeholderColor;
    82→  } else if (id === 'ibara') {
    83→    NPC_COLORS[id] = '#FFD700'; // Gold for Ibarra
    84→  }
    85→}
    86→// Fallback for known NPC ids not in characters
    87→NPC_COLORS['mang-tenyo'] = NPC_COLORS['mang-tenyo'] || '#8B4513';
    88→NPC_COLORS['kitchen-staff-1'] = NPC_COLORS['kitchen-staff-1'] || '#D2691E';
    89→NPC_COLORS['kitchen-staff-2'] = NPC_COLORS['kitchen-staff-2'] || '#CD853F';
    90→NPC_COLORS['ibara'] = NPC_COLORS['ibara'] || '#FFD700';
    91→
    92→export default function Minimap() {
    93→  const { activePanel, togglePanel } = useUIStore();
    94→  const isOpen = activePanel === 'minimap';
    95→  const { completedObjectives } = useGameStore();
    96→
    97→  const [playerPos, setPlayerPos] = useState({ row: 15, col: 10 });
    98→  const [playerDirection, setPlayerDirection] = useState<string>('south');
    99→  const [zoomLevel, setZoomLevel] = useState<string>('medium');
   100→  // Lazy initializer: load discovered locations from localStorage on first render
   101→  const [discoveredLocations, setDiscoveredLocations] = useState<string[]>(() => {
   102→    try {
   103→      const saved = localStorage.getItem('noor-discovered-locations');
   104→      if (saved) return JSON.parse(saved);
   105→    } catch (e) {
   106→      // Ignore
   107→    }
   108→    return [];
   109→  });
   110→  // Discovery log (canonical source per Task 9-b). We also keep the legacy
   111→  // `noor-discovered-locations` list as a fallback for backward compatibility.
   112→  const [discoveryLog, setDiscoveryLog] = useState<DiscoveryLogEntry[]>(() => loadDiscoveryLog());
   113→  // Building marker currently under the cursor (for tooltip rendering).
   114→  const [hoveredMarker, setHoveredMarker] = useState<HoveredMarker | null>(null);
   115→  const canvasRef = useRef<HTMLCanvasElement>(null);
   116→  const rafRef = useRef<number | null>(null);
   117→  const discoveredRef = useRef(discoveredLocations);
   118→  // Keep ref in sync with state via effect (not during render)
   119→  useEffect(() => {
   120→    discoveredRef.current = discoveredLocations;
   121→  }, [discoveredLocations]);
   122→
   123→  // Set of building discovery-log ids that have been recorded. Computed from
   124→  // `discoveryLog`; falls back to `discoveredLocations` (label-based) so a
   125→  // discovery recorded through either channel counts as discovered.
   126→  const discoveredBuildingIds = useMemo(() => {
   127→    const ids = new Set<string>();
   128→    for (const entry of discoveryLog) ids.add(entry.id);
   129→    // Also include any buildings recorded by label in the legacy list.
   130→    for (const label of discoveredLocations) {
   131→      ids.add(getBuildingDiscoveryId(label));
   132→    }
   133→    return ids;
   134→  }, [discoveryLog, discoveredLocations]);
   135→
   136→  // Listen for live discovery-log updates so the minimap re-renders as soon
   137→  // as a new building is discovered (without waiting for the next poll).
   138→  useEffect(() => {
   139→    const handler = () => setDiscoveryLog(loadDiscoveryLog());
   140→    window.addEventListener('noor:discovery-updated', handler as EventListener);
   141→    window.addEventListener('noor:discovery', handler as EventListener);
   142→    return () => {
   143→      window.removeEventListener('noor:discovery-updated', handler as EventListener);
   144→      window.removeEventListener('noor:discovery', handler as EventListener);
   145→    };
   146→  }, []);
   147→
   148→  // Listen for player position updates and check for discoveries in callback
   149→  useEffect(() => {
   150→    const handlePositionUpdate = (e: Event) => {
   151→      const detail = (e as CustomEvent).detail;
   152→      if (detail?.playerPos) setPlayerPos(detail.playerPos);
   153→      if (detail?.playerDirection) setPlayerDirection(detail.playerDirection);
   154→    };
   155→    window.addEventListener('noor:positionUpdate', handlePositionUpdate);
   156→
   157→    // Poll player position from save data every 500ms
   158→    // Discovery check is done within the polling callback (not directly in effect body)
   159→    const interval = setInterval(() => {
   160→      try {
   161→        const save = localStorage.getItem('noor-save');
   162→        if (save) {
   163→          const data = JSON.parse(save);
   164→          if (data.gameState?.playerPosition) {
   165→            const pos = data.gameState.playerPosition;
   166→            setPlayerPos(pos);
   167→
   168→            // Check for location discovery within this callback
   169→            const buildingLabelsData = mapData.buildingLabels;
   170→            const currentDiscoveries = discoveredRef.current;
   171→            const newDiscoveries: string[] = [];
   172→
   173→            for (const building of buildingLabelsData) {
   174→              const centerRow = building.row + Math.floor(building.height / 2);
   175→              const centerCol = building.col + Math.floor(building.width / 2);
   176→              const dist = Math.abs(pos.row - centerRow) + Math.abs(pos.col - centerCol);
   177→
   178→              if (dist <= 3 && !currentDiscoveries.includes(building.label)) {
   179→                newDiscoveries.push(building.label);
   180→              }
   181→            }
   182→
   183→            if (newDiscoveries.length > 0) {
   184→              const updated = [...currentDiscoveries, ...newDiscoveries];
   185→              setDiscoveredLocations(updated);
   186→              discoveredRef.current = updated;
   187→              try {
   188→                localStorage.setItem('noor-discovered-locations', JSON.stringify(updated));
   189→              } catch (e) {
   190→                // Ignore
   191→              }
   192→
   193→              // Dispatch 'noor:discovery' events so the DiscoveryLogPanel
   194→              // (and any other listener) can record full discovery metadata.
   195→              for (const building of buildingLabelsData) {
   196→                if (newDiscoveries.includes(building.label)) {
   197→                  const centerRow = building.row + Math.floor(building.height / 2);
   198→                  const centerCol = building.col + Math.floor(building.width / 2);
   199→                  window.dispatchEvent(new CustomEvent('noor:discovery', {
   200→                    detail: {
   201→                      id: `bldg-${building.label.toLowerCase().replace(/[^a-z0-9]+/g, '-')}`,
   202→                      name: building.label,
   203→                      type: 'building',
   204→                      position: { x: centerCol, y: centerRow },
   205→                      timestamp: Date.now(),
   206→                      note: building.sublabel || undefined,
   207→                    },
   208→                  }));
   209→                }
   210→              }
   211→            }
   212→          }
   213→          if (data.gameState?.playerDirection) {
   214→            setPlayerDirection(data.gameState.playerDirection);
   215→          }
   216→        }
   217→      } catch (e) {
   218→        // Ignore
   219→      }
   220→    }, 500);
   221→
   222→    return () => {
   223→      window.removeEventListener('noor:positionUpdate', handlePositionUpdate);
   224→      clearInterval(interval);
   225→    };
   226→  }, []);
   227→
   228→  // Render minimap to canvas
   229→  useEffect(() => {
   230→    if (!isOpen || !canvasRef.current) return;
   231→
   232→    const canvas = canvasRef.current;
   233→    const ctx = canvas.getContext('2d');
   234→    if (!ctx) return;
   235→
   236→    const { cellSize } = ZOOM_LEVELS[zoomLevel];
   237→    const canvasW = mapData.width * cellSize;
   238→    const canvasH = mapData.height * cellSize;
   239→    canvas.width = canvasW;
   240→    canvas.height = canvasH;
   241→
   242→    let lastDraw = 0;
   243→
   244→    const draw = (t: number) => {
   245→      if (t - lastDraw < 80) {
   246→        rafRef.current = requestAnimationFrame(draw);
   247→        return;
   248→      }
   249→      lastDraw = t;
   250→
   251→      // Clear with dark background
   252→      ctx.fillStyle = '#0a0a0a';
   253→      ctx.fillRect(0, 0, canvasW, canvasH);
   254→
   255→      // Draw tiles with enhanced colors and subtle borders
   256→      for (let row = 0; row < mapData.height; row++) {
   257→        for (let col = 0; col < mapData.width; col++) {
   258→          const groundTile = mapData.layers.ground[row]?.[col];
   259→          const buildingTile = mapData.layers.buildings[row]?.[col];
   260→
   261→          let color = '#1a1a1a';
   262→          let borderColor = '#222222';
   263→
   264→          if (groundTile === 2) { color = '#3a5f3a'; borderColor = '#4a6f4a'; }
   265→          else if (groundTile === 3) { color = '#8B7355'; borderColor = '#9B8365'; }
   266→          else if (groundTile === 1) { color = '#C4A76C'; borderColor = '#D4B77C'; }
   267→          else if (groundTile === 4) { color = '#A0522D'; borderColor = '#B0623D'; }
   268→
   269→          // Buildings
   270→          if (buildingTile === 5) { color = '#696969'; borderColor = '#797979'; }
   271→          if (buildingTile === 6) { color = '#8B4513'; borderColor = '#9B5523'; }
   272→
   273→          ctx.fillStyle = color;
   274→          ctx.fillRect(col * cellSize, row * cellSize, cellSize, cellSize);
   275→
   276→          // Subtle grid lines
   277→          ctx.strokeStyle = borderColor;
   278→          ctx.lineWidth = 0.3;
   279→          ctx.strokeRect(col * cellSize, row * cellSize, cellSize, cellSize);
   280→        }
   281→      }
   282→
   283→      // Draw fountain/water and trees from decoration layer
   284→      const decorationLayer = (mapData.layers as any).decoration;
   285→      if (decorationLayer) {
   286→        for (let row = 0; row < mapData.height; row++) {
   287→          for (let col = 0; col < mapData.width; col++) {
   288→            const decorTile = decorationLayer[row]?.[col];
   289→            if (decorTile === 7) {
   290→              ctx.fillStyle = '#7CB9E8';
   291→              ctx.fillRect(col * cellSize, row * cellSize, cellSize, cellSize);
   292→            }
   293→            if (decorTile === 8) {
   294→              ctx.fillStyle = '#2d4a2d';
   295→              ctx.fillRect(col * cellSize, row * cellSize, cellSize, cellSize);
   296→            }
   297→          }
   298→        }
   299→      }
   300→
   301→      // ---- Building Labels from mapData.buildingLabels ----
   302→      const buildingLabelsData = mapData.buildingLabels;
   303→      const fontSize = zoomLevel === 'large' ? 9 : zoomLevel === 'medium' ? 7 : 6;
   304→      ctx.font = `bold ${fontSize}px monospace`;
   305→      ctx.textAlign = 'center';
   306→
   307→      // Pulsing alpha for the "?" undiscovered markers — oscillates between
   308→      // 0.4 and 0.7 with a ~1.4s period to gently draw the player's attention.
   309→      const pulseT = Date.now() / 1000;
   310→      const questionAlpha = 0.55 + Math.sin(pulseT * Math.PI / 0.7) * 0.15; // ~0.4..0.7
   311→
   312→      for (const building of buildingLabelsData) {
   313→        const centerRow = building.row + Math.floor(building.height / 2);
   314→        const centerCol = building.col + Math.floor(building.width / 2);
   315→        const centerX = centerCol * cellSize + cellSize / 2;
   316→        const centerY = centerRow * cellSize + cellSize / 2;
   317→
   318→        const label = building.label;
   319→        const isDiscovered = discoveredBuildingIds.has(getBuildingDiscoveryId(label));
   320→
   321→        if (!isDiscovered) {
   322→          // ── Undiscovered marker: gray dashed circle + "?" glyph ──
   323→          const qRadius = Math.max(cellSize * 0.9, 6);
   324→          // Soft dark fill so the ? reads against any background
   325→          ctx.globalAlpha = questionAlpha;
   326→          ctx.fillStyle = 'rgba(40, 40, 40, 0.85)';
   327→          ctx.beginPath();
   328→          ctx.arc(centerX, centerY, qRadius, 0, Math.PI * 2);
   329→          ctx.fill();
   330→
   331→          // Dashed gray border
   332→          ctx.strokeStyle = 'rgba(180, 180, 180, 0.9)';
   333→          ctx.lineWidth = 1;
   334→          ctx.setLineDash([2, 2]);
   335→          ctx.beginPath();
   336→          ctx.arc(centerX, centerY, qRadius, 0, Math.PI * 2);
   337→          ctx.stroke();
   338→          ctx.setLineDash([]);
   339→
   340→          // "?" glyph in white/50
   341→          const qFontSize = Math.max(8, cellSize + 2);
   342→          ctx.font = `bold ${qFontSize}px monospace`;
   343→          ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
   344→          ctx.textAlign = 'center';
   345→          ctx.textBaseline = 'middle';
   346→          ctx.fillText('?', centerX, centerY + 1);
   347→          ctx.textBaseline = 'alphabetic';
   348→          ctx.globalAlpha = 1;
   349→          ctx.font = `bold ${fontSize}px monospace`;
   350→          continue; // skip the discovered-label rendering below
   351→        }
   352→
   353→        // Background rectangle for label
   354→        const textWidth = ctx.measureText(label).width;
   355→        ctx.fillStyle = 'rgba(0,0,0,0.8)';
   356→        ctx.fillRect(
   357→          centerX - textWidth / 2 - 2,
   358→          centerY - fontSize / 2 - 1,
   359→          textWidth + 4,
   360→          fontSize + 2
   361→        );
   362→
   363→        // Label text - white/amber
   364→        ctx.fillStyle = '#E8D0A0';
   365→        ctx.fillText(label, centerX, centerY + fontSize / 2 - 1);
   366→
   367→        // Sublabel if present
   368→        if (building.sublabel && zoomLevel !== 'small') {
   369→          const subFontSize = fontSize - 1;
   370→          ctx.font = `${subFontSize}px monospace`;
   371→          const subWidth = ctx.measureText(building.sublabel).width;
   372→          ctx.fillStyle = 'rgba(0,0,0,0.6)';
   373→          ctx.fillRect(
   374→            centerX - subWidth / 2 - 1,
   375→            centerY + fontSize / 2 + 1,
   376→            subWidth + 2,
   377→            subFontSize + 1
   378→          );
   379→          ctx.fillStyle = '#A08060';
   380→          ctx.fillText(building.sublabel, centerX, centerY + fontSize / 2 + subFontSize + 1);
   381→          ctx.font = `bold ${fontSize}px monospace`;
   382→        }
   383→
   384→        // Discovery marker ✦ if player has been near
   385→        const wasDiscoveredLegacy = discoveredLocations.includes(building.label);
   386→        if (wasDiscoveredLegacy) {
   387→          const discoveryFontSize = fontSize + 2;
   388→          ctx.font = `bold ${discoveryFontSize}px serif`;
   389→          ctx.fillStyle = '#FFD700';
   390→          ctx.textAlign = 'right';
   391→          ctx.fillText('✦', centerX + textWidth / 2 + 2, centerY - fontSize / 2 - 2);
   392→          ctx.textAlign = 'center';
   393→          ctx.font = `bold ${fontSize}px monospace`;
   394→        }
   395→      }
   396→
   397→      // ---- NPC Position Markers ----
   398→      const npcPosData = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   399→      for (const [id, pos] of Object.entries(npcPosData)) {
   400→        let visible = true;
   401→        if (pos.appearsAfter) {
   402→          visible = completedObjectives.includes(pos.appearsAfter);
   403→        }
   404→
   405→        if (visible) {
   406→          const npcColor = NPC_COLORS[id] || '#FF6B35';
   407→          const npcX = pos.col * cellSize + cellSize / 2;
   408→          const npcY = pos.row * cellSize + cellSize / 2;
   409→          const npcRadius = cellSize / 2 - 1;
   410→
   411→          // NPC glow
   412→          ctx.globalAlpha = 0.25;
   413→          ctx.fillStyle = npcColor;
   414→          ctx.beginPath();
   415→          ctx.arc(npcX, npcY, npcRadius + 3, 0, Math.PI * 2);
   416→          ctx.fill();
   417→          ctx.globalAlpha = 1;
   418→
   419→          // NPC dot
   420→          ctx.fillStyle = npcColor;
   421→          ctx.beginPath();
   422→          ctx.arc(npcX, npcY, npcRadius, 0, Math.PI * 2);
   423→          ctx.fill();
   424→
   425→          // Small white center highlight
   426→          ctx.fillStyle = 'rgba(255,255,255,0.3)';
   427→          ctx.beginPath();
   428→          ctx.arc(npcX, npcY, Math.max(1, npcRadius - 2), 0, Math.PI * 2);
   429→          ctx.fill();
   430→
   431→          // NPC name label on medium/large zoom
   432→          if (zoomLevel !== 'small') {
   433→            const charInfo = characterData.characters[id as keyof typeof characterData.characters];
   434→            const displayName = charInfo?.displayName || id;
   435→            const nameFontSize = zoomLevel === 'large' ? 7 : 5;
   436→            ctx.font = `${nameFontSize}px monospace`;
   437→            ctx.textAlign = 'center';
   438→            const nameWidth = ctx.measureText(displayName).width;
   439→            ctx.fillStyle = 'rgba(0,0,0,0.7)';
   440→            ctx.fillRect(npcX - nameWidth / 2 - 1, npcY - npcRadius - nameFontSize - 3, nameWidth + 2, nameFontSize + 2);
   441→            ctx.fillStyle = npcColor;
   442→            ctx.fillText(displayName, npcX, npcY - npcRadius - 2);
   443→          }
   444→        }
   445→      }
   446→
   447→      // ---- Player Position (pulsing amber/yellow) ----
   448→      const pulse = (Math.sin(Date.now() / 250) + 1) / 2;
   449→      const px = playerPos.col * cellSize + cellSize / 2;
   450→      const py = playerPos.row * cellSize + cellSize / 2;
   451→      const playerRadius = cellSize / 2;
   452→
   453→      // Outer glow ring (amber/yellow pulsing)
   454→      ctx.globalAlpha = 0.15 + pulse * 0.15;
   455→      ctx.fillStyle = '#FFC107';
   456→      ctx.beginPath();
   457→      ctx.arc(px, py, playerRadius + pulse * 4, 0, Math.PI * 2);
   458→      ctx.fill();
   459→
   460→      // Inner glow
   461→      ctx.globalAlpha = 0.3 + pulse * 0.1;
   462→      ctx.fillStyle = '#FFD54F';
   463→      ctx.beginPath();
   464→      ctx.arc(px, py, playerRadius + 2, 0, Math.PI * 2);
   465→      ctx.fill();
   466→
   467→      ctx.globalAlpha = 1;
   468→
   469→      // Player dot (bright amber)
   470→      ctx.fillStyle = `rgba(255, 193, 7, ${0.85 + pulse * 0.15})`;
   471→      ctx.beginPath();
   472→      ctx.arc(px, py, playerRadius - 1, 0, Math.PI * 2);
   473→      ctx.fill();
   474→
   475→      // Direction indicator arrow
   476→      const dirOffsets: Record<string, { dx: number; dy: number }> = {
   477→        'north': { dx: 0, dy: -3 },
   478→        'south': { dx: 0, dy: 3 },
   479→        'east': { dx: 3, dy: 0 },
   480→        'west': { dx: -3, dy: 0 },
   481→        'north-east': { dx: 2, dy: -2 },
   482→        'north-west': { dx: -2, dy: -2 },
   483→        'south-east': { dx: 2, dy: 2 },
   484→        'south-west': { dx: -2, dy: 2 },
   485→      };
   486→      const dirOffset = dirOffsets[playerDirection] || dirOffsets['south'];
   487→      ctx.strokeStyle = '#FFC107';
   488→      ctx.lineWidth = 2;
   489→      ctx.beginPath();
   490→      ctx.moveTo(px, py);
   491→      ctx.lineTo(px + dirOffset.dx, py + dirOffset.dy);
   492→      ctx.stroke();
   493→
   494→      // Player square outline (amber)
   495→      ctx.strokeStyle = '#FFC107';
   496→      ctx.lineWidth = 1;
   497→      ctx.strokeRect(
   498→        playerPos.col * cellSize + 1,
   499→        playerPos.row * cellSize + 1,
   500→        cellSize - 2,
   501→        cellSize - 2
   502→      );
   503→
   504→      // ---- Compass Indicator ----
   505→      const compassX = canvasW - cellSize * 2;
   506→      const compassY = cellSize;
   507→      const compassR = cellSize * 0.8;
   508→
   509→      // Compass circle background
   510→      ctx.fillStyle = 'rgba(0,0,0,0.6)';
   511→      ctx.beginPath();
   512→      ctx.arc(compassX, compassY, compassR + 2, 0, Math.PI * 2);
   513→      ctx.fill();
   514→
   515→      // Compass circle border
   516→      ctx.strokeStyle = '#D4A574';
   517→      ctx.lineWidth = 1;
   518→      ctx.beginPath();
   519→      ctx.arc(compassX, compassY, compassR, 0, Math.PI * 2);
   520→      ctx.stroke();
   521→
   522→      // Compass needle pointing north
   523→      ctx.fillStyle = '#FFC107';
   524→      ctx.beginPath();
   525→      ctx.moveTo(compassX, compassY - compassR + 2);
   526→      ctx.lineTo(compassX - 2, compassY);

   583→      // Map to canvas-internal coordinates for hit-testing
   584→      const scaleX = canvas.width / rect.width;
   585→      const scaleY = canvas.height / rect.height;
   586→      const cx = cssX * scaleX;
   587→      const cy = cssY * scaleY;
   588→      const { cellSize } = ZOOM_LEVELS[zoomLevel];
   589→
   590→      // Hit-test against each building's center within a generous radius.
   591→      // The radius scales with zoom so small/medium maps remain usable.
   592→      const hitRadius = Math.max(cellSize * 1.5, 10);
   593→      let best: HoveredMarker | null = null;
   594→      let bestDist = hitRadius;
   595→      for (const building of mapData.buildingLabels) {
   596→        const centerRow = building.row + Math.floor(building.height / 2);
   597→        const centerCol = building.col + Math.floor(building.width / 2);
   598→        const bx = centerCol * cellSize + cellSize / 2;
   599→        const by = centerRow * cellSize + cellSize / 2;
   600→        const d = Math.sqrt((cx - bx) ** 2 + (cy - by) ** 2);
   601→        if (d <= bestDist) {
   602→          bestDist = d;
   603→          const isDiscovered = discoveredBuildingIds.has(getBuildingDiscoveryId(building.label));
   604→          best = { cssX, cssY, label: building.label, discovered: isDiscovered };
   605→        }
   606→      }
   607→      setHoveredMarker(best);
   608→    };

   610→    const handleLeave = () => setHoveredMarker(null);
   611→
   612→    canvas.addEventListener('mousemove', handleMove);
   613→    canvas.addEventListener('mouseleave', handleLeave);
   614→    return () => {
   615→      canvas.removeEventListener('mousemove', handleMove);
   616→      canvas.removeEventListener('mouseleave', handleLeave);
