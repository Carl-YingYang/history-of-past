    15→// CONSTANTS

    17→
    18→const INTERNAL_WIDTH = 480;
    19→const INTERNAL_HEIGHT = 320;
    20→const TILE_SIZE = 24;            // Internal pixels per tile (48/2 for 2× scale)
    21→const DISPLAY_TILE_SIZE = 48;    // Display pixels per tile
    22→const SLIDE_DURATION = 150;      // ms to slide between tiles
    23→const SLIDE_PAUSE = 50;          // ms pause between consecutive tiles when key held
    24→const IDLE_ANIM_RATE = 300;      // ms between idle animation frames
    25→const WALK_ANIM_RATE = 100;      // ms between walk animation frames
    26→const CAMERA_LERP = 0.12;        // Camera smoothing factor (0 = no follow, 1 = instant)
    27→const NPC_INTERACT_DIST = 1.5;   // Tile distance to interact with NPC
    28→const CHAR_RENDER_SIZE = 32;     // Character render size in internal pixels
    29→
    30→// 4-direction movement
    31→type Direction4 = 'north' | 'south' | 'east' | 'west';
    32→
    33→const DIR_OFFSET: Record<Direction4, { dr: number; dc: number }> = {
    34→  north: { dr: -1, dc: 0 },
    35→  south: { dr: 1, dc: 0 },
    36→  east:  { dr: 0, dc: 1 },
    37→  west:  { dr: 0, dc: -1 },
    38→};
    39→
    40→// Direction to facing string for sprite lookup
    41→const DIR_TO_FACING: Record<Direction4, string> = {
    42→  north: 'north',
    43→  south: 'south',
    44→  east: 'east',
    45→  west: 'west',
    46→};
    47→
    48→// Key to direction mapping
    49→const KEY_TO_DIR: Record<string, Direction4> = {
    50→  'w': 'north', 'arrowup': 'north',
    51→  's': 'south', 'arrowdown': 'south',
    52→  'd': 'east',  'arrowright': 'east',
    53→  'a': 'west',  'arrowleft': 'west',
    54→};

    60→interface PlayerState {
    61→  row: number;           // Current tile row
    62→  col: number;           // Current tile col
    63→  facing: Direction4;    // Current facing direction
    64→  isSliding: boolean;    // Whether currently sliding between tiles
    65→  slideStart: number;    // Timestamp when slide started
    66→  slideFromRow: number;  // Row we're sliding from
    67→  slideFromCol: number;  // Col we're sliding from
    68→  slideToRow: number;    // Row we're sliding to
    69→  slideToCol: number;    // Col we're sliding to
    70→  pixelX: number;        // Current render pixel X (internal coords)
    71→  pixelY: number;        // Current render pixel Y (internal coords)
    72→  animFrame: number;     // Current animation frame index
    73→  animTimer: number;     // Time accumulator for animation
    74→  bufferedDir: Direction4 | null;  // Buffered next direction during slide
    75→  lastMoveTime: number;  // Timestamp of last completed move
    76→}
    77→

   238→  // ═══════════════════════════════════════════════════════════════════

   240→  async init(canvas: HTMLCanvasElement): Promise<void> {
   241→    this.canvas = canvas;
   242→
   243→    // Set canvas to internal resolution
   244→    canvas.width = INTERNAL_WIDTH;
   245→    canvas.height = INTERNAL_HEIGHT;
   246→    this.ctx = canvas.getContext('2d')!;
   247→    this.ctx.imageSmoothingEnabled = false;
   248→
   249→    // Create offscreen buffer (same internal resolution)
   250→    this.offscreen = document.createElement('canvas');
   251→    this.offscreen.width = INTERNAL_WIDTH;
   252→    this.offscreen.height = INTERNAL_HEIGHT;
   253→    this.offCtx = this.offscreen.getContext('2d')!;
   254→    this.offCtx.imageSmoothingEnabled = false;
   255→
   256→    // Load all assets via assetManager
   257→    await assetManager.loadAll();
   258→
   259→    // Load character sprites via spriteLoader
   260→    await spriteLoader.loadCharacter('student');
   261→    await spriteLoader.loadCharacter('ibara');
   262→
   263→    // Load save data
   264→    await saveManager.loadProgress();
   265→    saveManager.startAutoSave();
   266→
   267→    // Parse map data
   268→    this._parseMapData();
   269→
   270→    // Parse NPC positions from mapData
   271→    this._parseNPCData();
   272→
   273→    // Parse buildings and props
   274→    this._parseBuildingsAndProps();
   275→
   276→    // Parse trigger zones
   277→    this._parseTriggerZones();
   278→
   279→    // Load NPC interaction counts from localStorage
   280→    this._loadNpcInteractions();
   281→
   282→    // Initialize player position from save data if available
   283→    this._initPlayerFromSave();
   284→
   285→    // Set up input listeners
   286→    this._setupInputListeners();
   287→
   288→    // Initialize camera to center on player
   289→    this._snapCameraToPlayer();
   290→
   291→    // Start game loop
   292→    this.running = true;
   293→    this.lastTime = performance.now();
   294→    this._gameLoop(this.lastTime);
   295→
   296→    this.initialized = true;
   297→    gameEvents.emit('game:ready');
   298→
   299→    // Start ambient sounds
   300→    soundManager.initOnUserGesture();
   301→  }
   302→
   303→  destroy(): void {
   304→    this.running = false;
   305→    if (this.animFrameId) {
   306→      cancelAnimationFrame(this.animFrameId);
   307→      this.animFrameId = null;
   308→    }
   309→    this._removeInputListeners();
   310→    saveManager.stopAutoSave();
   311→  }

   317→  private _parseMapData(): void {
   318→    const layers = mapData.layers as any;
   319→    this.groundLayer = layers.ground;
   320→    this.buildingsLayer = layers.buildings;
   321→    this.collisionLayer = layers.collision;
   322→    this.overlayLayer = layers.overlay || Array.from({ length: mapData.height }, () => Array(mapData.width).fill(0));
   323→    this.mapWidth = mapData.width;
   324→    this.mapHeight = mapData.height;
   325→  }
   326→
   327→  private _parseNPCData(): void {
   328→    this.npcs = [];
   329→    const npcPositions = mapData.npcPositions as Record<string, any>;
   330→    for (const [id, pos] of Object.entries(npcPositions)) {
   331→      const charKey = id;
   332→      const charConfig = (characterData.characters as any)[charKey] || (characterData.characters as any)[id];
   333→      const displayName = charConfig?.displayName || id;
   334→
   335→      // Check visibility condition
   336→      let visible = true;
   337→      if (pos.appearsAfter) {
   338→        visible = saveManager.isObjectiveCompleted(pos.appearsAfter);
   339→      }
   340→
   341→      const row = pos.row as number;
   342→      const col = pos.col as number;
   343→
   344→      this.npcs.push({
   345→        id,
   346→        characterKey: charKey,
   347→        row,
   348→        col,
   349→        facing: this._toDirection4(pos.direction as string),
   350→        visible,
   351→        appearsAfterObjective: pos.appearsAfter,
   352→        spriteSheet: pos.spriteSheet || charConfig?.spriteSheet,
   353→        animFrame: 0,
   354→        animTimer: 0,
   355→        displayName,
   356→        pixelX: col * TILE_SIZE,
   357→        pixelY: row * TILE_SIZE,
   358→      });
   359→    }
   360→  }
   361→
   362→  private _parseBuildingsAndProps(): void {
   363→    // Parse buildings from mapData
   364→    this.buildings = [];
   365→    const buildingsData = mapData.buildings as any[];
   366→    if (buildingsData) {
   367→      for (const b of buildingsData) {
   368→        this.buildings.push({
   369→          key: b.key,
   370→          row: b.row,
   371→          col: b.col,
   372→          width: b.width,
   373→          height: b.height,
   374→          spriteWidth: b.spriteWidth || b.width * DISPLAY_TILE_SIZE,
   375→          spriteHeight: b.spriteHeight || b.height * DISPLAY_TILE_SIZE,
   376→        });
   377→      }
   378→    }
   379→
   380→    // Parse props from mapData
   381→    this.props = [];
   382→    const propsData = mapData.props as any[];
   383→    if (propsData) {
   384→      for (const p of propsData) {
   385→        this.props.push({
   386→          key: p.key,
   387→          row: p.row,
   388→          col: p.col,
   389→          width: p.width,
   390→          height: p.height,
   391→          spriteWidth: p.spriteWidth || p.width * DISPLAY_TILE_SIZE,
   392→          spriteHeight: p.spriteHeight || p.height * DISPLAY_TILE_SIZE,
   393→        });
   394→      }
   395→    }
   396→  }
   397→
   398→  private _parseTriggerZones(): void {
   399→    this.triggerZones = [];
   400→    const zones = mapData.triggerZones as any[];
   401→    if (zones) {
   402→      for (const z of zones) {
   403→        this.triggerZones.push({
   404→          id: z.id,
   405→          type: z.type,
   406→          npcId: z.npcId,
   407→          row: z.row,
   408→          col: z.col,
   409→          width: z.width,
   410→          height: z.height,
   411→          radius: z.radius,
   412→          requiresObjective: z.requiresObjective,
   413→          triggersTimeTransition: z.triggersTimeTransition,
   414→          dialogueId: z.dialogueId,
   415→          triggered: false,
   416→        });
   417→      }
   418→    }
   419→  }
   420→
   421→  private _initPlayerFromSave(): void {
   422→    const saveData = saveManager.getSaveData();
   423→    if (saveData.gameState.playerPosition) {
   424→      const pos = saveData.gameState.playerPosition;
   425→      // Validate position is within map and walkable
   426→      if (pos.row >= 0 && pos.row < this.mapHeight &&
   427→          pos.col >= 0 && pos.col < this.mapWidth &&
   428→          !this._isCollision(pos.row, pos.col)) {
   429→        this.player.row = pos.row;
   430→        this.player.col = pos.col;
   431→      }
   432→    }
   433→    // Recalculate pixel position
   434→    this.player.pixelX = this.player.col * TILE_SIZE;
   435→    this.player.pixelY = this.player.row * TILE_SIZE;
   436→    this.player.slideFromRow = this.player.row;
   437→    this.player.slideFromCol = this.player.col;
   438→    this.player.slideToRow = this.player.row;
   439→    this.player.slideToCol = this.player.col;
   440→  }

   443→  // INPUT HANDLING
   444→  // ═══════════════════════════════════════════════════════════════════
   445→
   446→  private _keydownHandler: ((e: KeyboardEvent) => void) | null = null;
   447→  private _keyupHandler: ((e: KeyboardEvent) => void) | null = null;
   448→
   449→  private _setupInputListeners(): void {
   450→    this._keydownHandler = (e: KeyboardEvent) => this._onKeyDown(e);
   451→    this._keyupHandler = (e: KeyboardEvent) => this._onKeyUp(e);
   452→    window.addEventListener('keydown', this._keydownHandler);
   453→    window.addEventListener('keyup', this._keyupHandler);
   454→  }
   455→
   456→  private _removeInputListeners(): void {
   457→    if (this._keydownHandler) window.removeEventListener('keydown', this._keydownHandler);
   458→    if (this._keyupHandler) window.removeEventListener('keyup', this._keyupHandler);
   459→  }
   460→
   461→  private _onKeyDown(e: KeyboardEvent): void {
   462→    // Ignore input when focus is in text fields
   463→    const target = e.target as HTMLElement;
   464→    if (target && (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable)) return;
   465→
   466→    const key = e.key.toLowerCase();
   467→
   468→    // Prevent default for game keys (stop page scrolling)
   469→    if (KEY_TO_DIR[key] || key === ' ' || key === 'enter') {
   470→      e.preventDefault();
   471→    }
   472→
   473→    this.keysDown.add(key);
   474→
   475→    // Interaction key (Space/Enter)
   476→    if (key === ' ' || key === 'enter') {
   477→      if (!this.interactPressed) {
   478→        this.interactPressed = true;
   479→        this._handleInteract();
   480→      }
   481→    }
   482→  }
   483→
   484→  private _onKeyUp(e: KeyboardEvent): void {
   485→    const key = e.key.toLowerCase();
   486→    this.keysDown.delete(key);
   487→
   488→    if (key === ' ' || key === 'enter') {
   489→      this.interactPressed = false;
   490→    }
   491→  }
   492→
   493→  // Touch / external controls
   494→  setMoveDirection(dir: 'up' | 'down' | 'left' | 'right' | null): void {
   495→    if (!dir) {
   496→      this.touchDir = null;
   497→      return;
   498→    }
   499→    const mapping: Record<string, Direction4> = {
   500→      up: 'north', down: 'south', left: 'west', right: 'east'
   501→    };
   502→    this.touchDir = mapping[dir] || null;
   503→  }
   504→
   505→  triggerInteract(): void {
   506→    this._handleInteract();
   507→  }

   509→  // ═══════════════════════════════════════════════════════════════════
   510→  // MOVEMENT SYSTEM (Pokémon Grid Movement)
   511→  // ═══════════════════════════════════════════════════════════════════
   512→
   513→  private _getDesiredDirection(): Direction4 | null {
   514→    // Check keyboard keys
   515→    for (const key of this.keysDown) {
   516→      const dir = KEY_TO_DIR[key];
   517→      if (dir) return dir;
   518→    }
   519→    // Check touch direction
   520→    return this.touchDir;
   521→  }
   522→
   523→  private _tryMove(dir: Direction4, now: number): void {
   524→    // Update facing direction immediately
   525→    this.player.facing = dir;
   526→
   527→    if (this.player.isSliding) {
   528→      // Buffer the direction for when slide completes
   529→      this.player.bufferedDir = dir;
   530→      return;
   531→    }
   532→
   533→    // Check pause between moves
   534→    if (now - this.player.lastMoveTime < SLIDE_PAUSE) return;
   535→
   536→    // Calculate target tile
   537→    const offset = DIR_OFFSET[dir];
   538→    const targetRow = this.player.row + offset.dr;
   539→    const targetCol = this.player.col + offset.dc;
   540→
   541→    // Check bounds
   542→    if (targetRow < 0 || targetRow >= this.mapHeight || targetCol < 0 || targetCol >= this.mapWidth) return;
   543→
   544→    // Check collision
   545→    if (this._isCollision(targetRow, targetCol)) return;
   546→
   547→    // Check if NPC occupies target tile
   548→    if (this._isNpcAtTile(targetRow, targetCol)) return;
   549→
   550→    // Check if prop occupies target tile
   551→    if (this._isPropAtTile(targetRow, targetCol)) return;
   552→
   553→    // Start sliding
   554→    this.player.isSliding = true;
   555→    this.player.slideStart = now;
   556→    this.player.slideFromRow = this.player.row;
   557→    this.player.slideFromCol = this.player.col;
   558→    this.player.slideToRow = targetRow;
   559→    this.player.slideToCol = targetCol;
   560→    this.player.bufferedDir = null;
   561→
   562→    // Play footstep sound
   563→    if (now - this.lastFootstepTime > this.FOOTSTEP_INTERVAL) {
   564→      soundManager.playFootstep();
   565→      this.lastFootstepTime = now;
   566→    }
   567→  }
   568→
   569→  private _updateMovement(now: number, dt: number): void {
   570→    if (this.dialogue.active) return; // No movement during dialogue
   571→
   572→    // Process desired direction
   573→    const desired = this._getDesiredDirection();
   574→    if (desired) {
   575→      this._tryMove(desired, now);
   576→    }
   577→
   578→    // Update slide interpolation
   579→    if (this.player.isSliding) {
   580→      const elapsed = now - this.player.slideStart;
   581→      const t = Math.min(1, elapsed / SLIDE_DURATION);
   582→
   583→      // Interpolate pixel position
   584→      const fromX = this.player.slideFromCol * TILE_SIZE;
   585→      const fromY = this.player.slideFromRow * TILE_SIZE;
   586→      const toX = this.player.slideToCol * TILE_SIZE;
   587→      const toY = this.player.slideToRow * TILE_SIZE;
   588→
   589→      this.player.pixelX = fromX + (toX - fromX) * t;
   590→      this.player.pixelY = fromY + (toY - fromY) * t;
   591→
   592→      // Update walk animation
   593→      this.player.animTimer += dt;
   594→      if (this.player.animTimer >= WALK_ANIM_RATE) {
   595→        this.player.animTimer -= WALK_ANIM_RATE;
   596→        this.player.animFrame++;
   597→      }
   598→
   599→      // Check if slide is complete
   600→      if (t >= 1) {
   601→        this.player.isSliding = false;
   602→        this.player.row = this.player.slideToRow;
   603→        this.player.col = this.player.slideToCol;
   604→        this.player.pixelX = this.player.col * TILE_SIZE;
   605→        this.player.pixelY = this.player.row * TILE_SIZE;
   606→        this.player.lastMoveTime = now;
   607→
   608→        // Emit player:move event
   609→        gameEvents.emit('player:move', { row: this.player.row, col: this.player.col });
   610→
   611→        // Update save state
   612→        saveManager.updateGameState({
   613→          playerPosition: { row: this.player.row, col: this.player.col },
   614→          playerDirection: this.player.facing,
   615→        });
   616→
   617→        // Check trigger zones
   618→        this._checkTriggerZones(now);
   619→
   620→        // Process buffered direction
   621→        if (this.player.bufferedDir) {
   622→          const bufDir = this.player.bufferedDir;
   623→          this.player.bufferedDir = null;
   624→          this._tryMove(bufDir, now);
   625→        } else {
   626→          // Switch to idle
   627→          this.player.animFrame = 0;
   628→          this.player.animTimer = 0;
   629→        }
   630→      }
   631→    } else {
   632→      // Idle animation
   633→      this.player.animTimer += dt;
   634→      if (this.player.animTimer >= IDLE_ANIM_RATE) {
   635→        this.player.animTimer -= IDLE_ANIM_RATE;
   636→        this.player.animFrame++;
   637→      }
   638→
   639→      // Snap pixel position to tile (safety)
   640→      this.player.pixelX = this.player.col * TILE_SIZE;
   641→      this.player.pixelY = this.player.row * TILE_SIZE;
   642→    }
   643→  }

   672→  private _updateCamera(): void {
   673→    // Camera target: center on player
   674→    const playerCenterX = this.player.pixelX + TILE_SIZE / 2;
   675→    const playerCenterY = this.player.pixelY + TILE_SIZE / 2;
   676→
   677→    this.camera.targetX = playerCenterX - INTERNAL_WIDTH / 2;
   678→    this.camera.targetY = playerCenterY - INTERNAL_HEIGHT / 2;
   679→
   680→    // Clamp to map edges
   681→    const maxCamX = this.mapWidth * TILE_SIZE - INTERNAL_WIDTH;
   682→    const maxCamY = this.mapHeight * TILE_SIZE - INTERNAL_HEIGHT;
   683→    this.camera.targetX = Math.max(0, Math.min(maxCamX, this.camera.targetX));
   684→    this.camera.targetY = Math.max(0, Math.min(maxCamY, this.camera.targetY));
   685→
   686→    // Smooth lerp
   687→    this.camera.x += (this.camera.targetX - this.camera.x) * CAMERA_LERP;
   688→    this.camera.y += (this.camera.targetY - this.camera.y) * CAMERA_LERP;
   689→
   690→    // Clamp again (for precision)
   691→    this.camera.x = Math.max(0, Math.min(maxCamX, this.camera.x));
   692→    this.camera.y = Math.max(0, Math.min(maxCamY, this.camera.y));
   693→  }

   695→  private _snapCameraToPlayer(): void {
   696→    const playerCenterX = this.player.pixelX + TILE_SIZE / 2;
   697→    const playerCenterY = this.player.pixelY + TILE_SIZE / 2;
   698→    this.camera.x = playerCenterX - INTERNAL_WIDTH / 2;
   699→    this.camera.y = playerCenterY - INTERNAL_HEIGHT / 2;
   700→    const maxCamX = this.mapWidth * TILE_SIZE - INTERNAL_WIDTH;
   701→    const maxCamY = this.mapHeight * TILE_SIZE - INTERNAL_HEIGHT;
   702→    this.camera.x = Math.max(0, Math.min(maxCamX, this.camera.x));
   703→    this.camera.y = Math.max(0, Math.min(maxCamY, this.camera.y));
   704→    this.camera.targetX = this.camera.x;
   705→    this.camera.targetY = this.camera.y;
   706→  }

   712→  private _updateNPCs(dt: number): void {

   724→      npc.animTimer += dt;
   725→      if (npc.animTimer >= IDLE_ANIM_RATE) {
   726→        npc.animTimer -= IDLE_ANIM_RATE;
   727→        npc.animFrame++;
   728→      }
   729→
   730→      // Update pixel position (NPCs don't move, but snap to tile)

   732→      npc.pixelY = npc.row * TILE_SIZE;

   740→  private _handleInteract(): void {
   741→    if (this.dialogue.active) {
   742→      // Advance dialogue
   743→      this.advanceDialogue();
   744→      return;
   745→    }
   746→
   747→    // Try to talk to an adjacent NPC
   748→    const facing = this.player.facing;
   749→    const offset = DIR_OFFSET[facing];
   750→    const targetRow = this.player.row + offset.dr;
   751→    const targetCol = this.player.col + offset.dc;
   752→
   753→    // Check NPCs at target tile or current tile
   754→    let closestNpc: NPCState | null = null;
   755→    let closestDist = Infinity;
   756→
   757→    for (const npc of this.npcs) {
   758→      if (!npc.visible) continue;
   759→      const dist = Math.abs(npc.row - targetRow) + Math.abs(npc.col - targetCol);
   760→      if (dist === 0 && dist < closestDist) {
   761→        closestNpc = npc;
   762→        closestDist = dist;
   763→      }
   764→    }
   765→
   766→    // Also check NPCs on player's tile (standing on top of them)
   767→    if (!closestNpc) {
   768→      for (const npc of this.npcs) {
   769→        if (!npc.visible) continue;
   770→        const dist = Math.abs(npc.row - this.player.row) + Math.abs(npc.col - this.player.col);
   771→        if (dist <= 1 && dist < closestDist) {
   772→          closestNpc = npc;
   773→          closestDist = dist;
   774→        }
   775→      }
   776→    }
   777→
   778→    if (closestNpc) {
   779→      // Face NPC toward player
   780→      closestNpc.facing = this._oppositeDirection(this.player.facing);
   781→      this._startNpcDialogue(closestNpc);
   782→    }
   783→  }

   785→  private _startNpcDialogue(npc: NPCState): void {
   786→    // Increment interaction count
   787→    this.npcInteractionCounts[npc.id] = (this.npcInteractionCounts[npc.id] || 0) + 1;
   788→    this._saveNpcInteractions();
   789→
   790→    // Determine which dialogue to use
   791→    let dialogueId: string | null = null;
   792→    const npcId = npc.id;
   793→
   794→    // Check for first-time dialogues based on NPC ID
   795→    if (npcId === 'mang-tenyo') {
   796→      if (!saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   797→        dialogueId = 'mang-tenyo-first';
   798→      } else if (!saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   799→        dialogueId = 'mang-tenyo-repeat';
   800→      } else {
   801→        dialogueId = 'mang-tenyo-after-gossip';
   802→      }
   803→    } else if (npcId === 'vendor-1' || npcId === 'vendor-2' || npcId === 'kitchen-staff-1' || npcId === 'kitchen-staff-2') {
   804→      // Vendors share market-gossip dialogue
   805→      if (saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo') && !saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   806→        dialogueId = 'market-gossip';
   807→      } else {
   808→        // Generic vendor repeat
   809→        dialogueId = 'mang-tenyo-repeat'; // Fallback
   810→      }
   811→    } else if (npcId === 'ibara') {
   812→      // Ibarra sighting
   813→      dialogueId = 'ibarra-sighting';
   814→    }
   815→
   816→    if (dialogueId && (dialogueData.dialogues as any)[dialogueId]) {
   817→      this._startDialogue(dialogueId, npcId);
   818→    }
   819→  }
   820→
   821→  private _startDialogue(dialogueId: string, npcId: string | null): void {
   822→    const dialogueObj = (dialogueData.dialogues as any)[dialogueId];
   823→    if (!dialogueObj) return;
   824→
   825→    const lines: DialogueLine[] = dialogueObj.lines.map((l: any) => ({
   826→      speaker: l.speaker,
   827→      text: l.text,
   828→      translation: l.translation,
   829→      isTimeTransition: l.isTimeTransition,
   830→      isIbarraSighting: l.isIbarraSighting,
   831→    }));
   832→
   833→    this.dialogue = {
   834→      active: true,
   835→      dialogueId,
   836→      lines,
   837→      currentLineIndex: 0,
   838→      npcId,
   839→      triggered: true,
   840→    };
   841→
   842→    // Emit dialogue:start event
   843→    const firstLine = lines[0];
   844→    gameEvents.emit('dialogue:start', {
   845→      dialogueId,
   846→      line: firstLine,
   847→      lineIndex: 0,
   848→      totalLines: lines.length,
   849→    });
   850→
   851→    // Complete any objectives this dialogue unlocks
   852→    if (dialogueObj.unlocksObjective) {
   853→      saveManager.completeObjective(dialogueObj.unlocksObjective);
   854→      saveManager.saveProgress();
   855→    }
   856→  }
   857→
   858→  advanceDialogue(): void {
   859→    if (!this.dialogue.active) return;
   860→
   861→    const nextIndex = this.dialogue.currentLineIndex + 1;
   862→
   863→    if (nextIndex >= this.dialogue.lines.length) {
   864→      // End dialogue
   865→      const wasTimeTransition = this.dialogue.lines.some(l => l.isTimeTransition);
   866→
   867→      this.dialogue = {
   868→        active: false,
   869→        dialogueId: null,
   870→        lines: [],
   871→        currentLineIndex: 0,
   872→        npcId: null,
   873→        triggered: false,
   874→      };
   875→
   876→      gameEvents.emit('dialogue:end');
   877→
   878→      // Handle time transition
   879→      if (wasTimeTransition) {
   880→        gameEvents.emit('time:transition', 'morning');
   881→        saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   882→        soundManager.play('time-transition');
   883→      }
   884→
   885→      return;
   886→    }
   887→
   888→    // Advance to next line
   889→    this.dialogue.currentLineIndex = nextIndex;
   890→    const line = this.dialogue.lines[nextIndex];
   891→
   892→    gameEvents.emit('dialogue:line', {
   893→      dialogueId: this.dialogue.dialogueId,
   894→      line,
   895→      lineIndex: nextIndex,
   896→      totalLines: this.dialogue.lines.length,
   897→    });
   898→
   899→    // Handle time transition in dialogue line
   900→    if (line.isTimeTransition) {
   901→      gameEvents.emit('time:transition', 'morning');
   902→      saveManager.updateGameState({ timeOfDay: 'morning' });
   903→    }
   904→  }

   906→  private _checkTriggerZones(now: number): void {
   907→    for (const zone of this.triggerZones) {
   908→      if (zone.triggered) continue;
   909→
   910→      // Check objective requirement
   911→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   912→
   913→      // Check if player is in the zone
   914→      const inZone = this.player.row >= zone.row && this.player.row < zone.row + zone.height &&
   915→                     this.player.col >= zone.col && this.player.col < zone.col + zone.width;
   916→
   917→      // Or within radius
   918→      const centerRow = zone.row + (zone.height - 1) / 2;
   919→      const centerCol = zone.col + (zone.width - 1) / 2;
   920→      const dist = Math.sqrt(
   921→        Math.pow(this.player.row - centerRow, 2) +
   922→        Math.pow(this.player.col - centerCol, 2)
   923→      );
   924→      const inRadius = zone.radius > 0 && dist <= zone.radius;
   925→
   926→      if (inZone || inRadius) {
   927→        zone.triggered = true;
   928→
   929→        if (zone.dialogueId) {
   930→          this._startDialogue(zone.dialogueId, zone.npcId || null);
   931→        }
   932→
   933→        gameEvents.emit('interaction:trigger', { zoneId: zone.id, type: zone.type });
   934→
   935→        // If this triggers a time transition, handle it
   936→        if (zone.triggersTimeTransition) {
   937→          // Time transition will be handled by the dialogue lines
   938→        }
   939→      }
   940→    }
   941→  }

   965→  private _render(): void {
   966→    if (!this.offCtx || !this.ctx) return;
   967→
   968→    const ctx = this.offCtx;
   969→
   970→    // Clear
   971→    ctx.fillStyle = '#1a1a2e';
   972→    ctx.fillRect(0, 0, INTERNAL_WIDTH, INTERNAL_HEIGHT);
   973→
   974→    const camX = Math.round(this.camera.x);
   975→    const camY = Math.round(this.camera.y);
   976→
   977→    // Calculate visible tile range
   978→    const startCol = Math.floor(camX / TILE_SIZE);
   979→    const startRow = Math.floor(camY / TILE_SIZE);
   980→    const endCol = Math.ceil((camX + INTERNAL_WIDTH) / TILE_SIZE);
   981→    const endRow = Math.ceil((camY + INTERNAL_HEIGHT) / TILE_SIZE);
   982→
   983→    // ── LAYER 1: Ground tiles ──
   984→    this._renderGroundTiles(ctx, camX, camY, startRow, startRow, endRow, endCol);
   985→
   986→    // ── LAYER 2: Building sprites ──
   987→    this._renderBuildings(ctx, camX, camY);
   988→
   989→    // ── LAYER 3: Props (below characters) ──
   990→    this._renderProps(ctx, camX, camY);
   991→
   992→    // ── LAYER 4: Y-sorted characters (NPCs + Player) ──
   993→    this._renderCharacters(ctx, camX, camY);
   994→
   995→    // ── LAYER 5: Overlay (shadows) ──
   996→    this._renderOverlay(ctx, camX, camY, startRow, endRow, endCol);
   997→
   998→    // ── Copy to display canvas ──
   999→    this.ctx.imageSmoothingEnabled = false;
  1000→    this.ctx.drawImage(this.offscreen!, 0, 0);
  1001→  }

  1003→  private _renderGroundTiles(
  1004→    ctx: CanvasRenderingContext2D, camX: number, camY: number,
  1005→    _startRow: number, startRow: number, endRow: number, endCol: number
  1006→  ): void {
  1007→    for (let row = startRow; row <= endRow && row < this.mapHeight; row++) {
  1008→      if (row < 0) continue;
  1009→      for (let col = 0; col <= endCol && col < this.mapWidth; col++) {
  1010→        if (col < 0) continue;
  1011→
  1012→        const tileType = this.groundLayer[row]?.[col] || 0;
  1013→        if (tileType === 0) continue; // Empty tile
  1014→
  1015→        const screenX = col * TILE_SIZE - camX;
  1016→        const screenY = row * TILE_SIZE - camY;
  1017→
  1018→        // Render tile image from assetManager
  1019→        const tileAsset = assetManager.getTile(tileType);
  1020→        if (tileAsset) {
  1021→          ctx.drawImage(tileAsset.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1022→        } else {
  1023→          // Fallback: colored rectangle
  1024→          const fallbackColors: Record<number, string> = {
  1025→            1: '#C4A76C', 2: '#8FBC8F', 3: '#D2B48C', 4: '#A0522D',
  1026→            5: '#696969', 6: '#8B4513', 7: '#6B8E6B', 8: '#555555',
  1027→            9: '#B8A070', 10: '#4A90A4', 11: '#7CB342', 12: '#00000033',
  1028→          };
  1029→          ctx.fillStyle = fallbackColors[tileType] || '#FF00FF';
  1030→          ctx.fillRect(screenX, screenY, TILE_SIZE, TILE_SIZE);
  1031→        }
  1032→
  1033→        // Also check buildings layer (render building tile type on ground)
  1034→        const buildingType = this.buildingsLayer[row]?.[col] || 0;
  1035→        if (buildingType > 0 && buildingType !== 5 && buildingType !== 6) {
  1036→          const buildingTile = assetManager.getTile(buildingType);
  1037→          if (buildingTile) {
  1038→            ctx.drawImage(buildingTile.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1039→          }
  1040→        }
  1041→      }
  1042→    }
  1043→  }

  1084→  private _renderCharacters(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {
  1085→    // Collect all renderable characters with their Y position for sorting
  1086→    interface Renderable {
  1087→      type: 'player' | 'npc';
  1088→      y: number;
  1089→      data: any;
  1090→    }
  1091→
  1092→    const renderables: Renderable[] = [];
  1093→
  1094→    // Add player
  1095→    renderables.push({
  1096→      type: 'player',
  1097→      y: this.player.pixelY + TILE_SIZE, // Bottom of sprite for Y-sorting
  1098→      data: this.player,
  1099→    });
  1100→
  1101→    // Add visible NPCs
  1102→    for (const npc of this.npcs) {
  1103→      if (!npc.visible) continue;
  1104→      renderables.push({
  1105→        type: 'npc',
  1106→        y: npc.pixelY + TILE_SIZE,
  1107→        data: npc,
  1108→      });
  1109→    }
  1110→
  1111→    // Sort by Y position (top to bottom)
  1112→    renderables.sort((a, b) => a.y - b.y);
  1113→
  1114→    // Render in sorted order
  1115→    for (const r of renderables) {
  1116→      if (r.type === 'player') {
  1117→        this._renderPlayer(ctx, camX, camY);
  1118→      } else {
  1119→        this._renderNPC(ctx, r.data as NPCState, camX, camY);
  1120→      }
  1121→    }
  1122→  }

  1124→  private _renderPlayer(ctx: CanvasRenderingContext2D, camX: number, camY: number): void {
  1125→    const screenX = this.player.pixelX - camX;
  1126→    const screenY = this.player.pixelY - camY;
  1127→
  1128→    // Try to get player sprite from spriteLoader
  1129→    const facing = DIR_TO_FACING[this.player.facing];
  1130→    const isMoving = this.player.isSliding;
  1131→    const animName = isMoving ? 'Walk' : 'Breathing_Idle';
  1132→
  1133→    const frameCount = spriteLoader.getAnimationFrameCount('student', animName);
  1134→    const frameIndex = frameCount > 0 ? this.player.animFrame % frameCount : 0;
  1135→
  1136→    const frame = spriteLoader.getFrame('student', animName, facing, frameIndex);
  1137→
  1138→    if (frame) {
  1139→      // Draw the sprite frame scaled to CHAR_RENDER_SIZE
  1140→      const drawSize = CHAR_RENDER_SIZE;
  1141→      const offsetX = (TILE_SIZE - drawSize) / 2;
  1142→      const offsetY = TILE_SIZE - drawSize; // Character bottom aligns with tile bottom
  1143→
  1144→      ctx.drawImage(
  1145→        frame,
  1146→        0, 0, frame.width, frame.height,
  1147→        screenX + offsetX, screenY + offsetY, drawSize, drawSize
  1148→      );
  1149→    } else {
  1150→      // Fallback: draw a colored character
  1151→      this._drawPlaceholderCharacter(ctx, screenX, screenY, '#4A90D9', 'P');
  1152→    }
  1153→  }

  1155→  private _renderNPC(ctx: CanvasRenderingContext2D, npc: NPCState, camX: number, camY: number): void {
  1156→    const screenX = npc.pixelX - camX;
  1157→    const screenY = npc.pixelY - camY;
  1158→
  1159→    // Try to render using sprite sheet from assetManager
  1160→    if (npc.spriteSheet) {
  1161→      const spriteSet = assetManager.getNpcSheet(npc.spriteSheet);
  1162→      if (spriteSet) {
  1163→        const facing = DIR_TO_FACING[npc.facing];
  1164→        const frames = spriteSet.idle; // NPCs use idle animation
  1165→        const dirFrames = frames[facing] || frames['south'];
  1166→
  1167→        if (dirFrames && dirFrames.length > 0) {
  1168→          const frameIndex = npc.animFrame % dirFrames.length;
  1169→          const frame = dirFrames[frameIndex];
  1170→
  1171→          // Scale NPC sprite to fit character render area
  1172→          const drawWidth = CHAR_RENDER_SIZE;
  1173→          const drawHeight = CHAR_RENDER_SIZE + 8; // NPCs can be slightly taller
  1174→          const offsetX = (TILE_SIZE - drawWidth) / 2;
  1175→          const offsetY = TILE_SIZE - drawHeight;
  1176→
  1177→          ctx.drawImage(
  1178→            frame.canvas,
  1179→            0, 0, frame.width, frame.height,
  1180→            screenX + offsetX, screenY + offsetY, drawWidth, drawHeight
  1181→          );
  1182→          return;
  1183→        }
  1184→      }
  1185→    }
  1186→
  1187→    // Try spriteLoader as fallback (for ibara)
  1188→    const charKey = npc.characterKey;
  1189→    const facing = DIR_TO_FACING[npc.facing];
  1190→    const frame = spriteLoader.getFrame(charKey, 'Breathing_Idle', facing, npc.animFrame % 4);
  1191→
  1192→    if (frame) {
  1193→      const drawSize = CHAR_RENDER_SIZE;
  1194→      const offsetX = (TILE_SIZE - drawSize) / 2;
  1195→      const offsetY = TILE_SIZE - drawSize;
  1196→
  1197→      ctx.drawImage(
  1198→        frame,
  1199→        0, 0, frame.width, frame.height,
  1200→        screenX + offsetX, screenY + offsetY, drawSize, drawSize
  1201→      );
  1202→    } else {
  1203→      // Final fallback: colored placeholder
  1204→      const npcColors: Record<string, string> = {
  1205→        'mang-tenyo': '#8B4513',
  1206→        'vendor-1': '#D2691E',
  1207→        'vendor-2': '#CD853F',
  1208→        'ibara': '#1E3A5F',
  1209→        'kitchen-staff-1': '#D2691E',
  1210→        'kitchen-staff-2': '#CD853F',
  1211→      };
  1212→      const color = npcColors[npc.id] || '#FF6B6B';
  1213→      this._drawPlaceholderCharacter(ctx, screenX, screenY, color, npc.displayName.substring(0, 2));
  1214→    }
  1215→  }

  1233→    // Eyes

  1245→  private _renderOverlay(
  1246→    ctx: CanvasRenderingContext2D, camX: number, camY: number,
  1247→    startRow: number, endRow: number, endCol: number
  1248→  ): void {
  1249→    for (let row = startRow; row <= endRow && row < this.mapHeight; row++) {
  1250→      if (row < 0) continue;
  1251→      for (let col = 0; col <= endCol && col < this.mapWidth; col++) {
  1252→        if (col < 0) continue;
  1253→
  1254→        const overlayType = this.overlayLayer[row]?.[col] || 0;
  1255→        if (overlayType === 0) continue;
  1256→
  1257→        const screenX = col * TILE_SIZE - camX;
  1258→        const screenY = row * TILE_SIZE - camY;
  1259→
  1260→        const tileAsset = assetManager.getTile(overlayType);
  1261→        if (tileAsset) {
  1262→          ctx.globalAlpha = 0.5;
  1263→          ctx.drawImage(tileAsset.canvas, 0, 0, DISPLAY_TILE_SIZE, DISPLAY_TILE_SIZE, screenX, screenY, TILE_SIZE, TILE_SIZE);
  1264→          ctx.globalAlpha = 1.0;
  1265→        }
  1266→      }
  1267→    }
  1268→  }

  1274→  private _gameLoop = (now: number): void => {
  1275→    if (!this.running) return;
  1276→
  1277→    const dt = Math.min(now - this.lastTime, 50); // Cap delta to avoid huge jumps
  1278→    this.lastTime = now;
  1279→
  1280→    // Update systems
  1281→    this._updateMovement(now, dt);
  1282→    this._updateNPCs(dt);
  1283→    this._updateCamera();
  1284→    this._updateAmbient();
  1285→
  1286→    // Render
  1287→    this._render();
  1288→
  1289→    // Schedule next frame
  1290→    this.animFrameId = requestAnimationFrame(this._gameLoop);
  1291→  };
