   102→// Deterministic hash for tile decoration patterns
   103→function tileHash(row: number, col: number, seed: number = 0): number {
   104→  let h = seed + row * 374761393 + col * 668265263;
   105→  h = (h ^ (h >> 13)) * 1274126177;
   106→  h = h ^ (h >> 16);
   107→  return (h & 0x7FFFFFFF) / 0x7FFFFFFF; // Returns 0..1
   108→}

   163→  private _computeBuildingGroups(): void {
   164→    // For each building tile type, find contiguous rectangular groups
   165→    const buildingTypes = [5, 6]; // Church and Market building types
   166→    for (const tileType of buildingTypes) {
   167→      const visited: Set<string> = new Set();
   168→      const groups: { minRow: number; minCol: number; maxRow: number; maxCol: number }[] = [];
   169→      
   170→      for (let row = 0; row < mapData.height; row++) {
   171→        for (let col = 0; col < mapData.width; col++) {
   172→          const key = `${row},${col}`;
   173→          if (visited.has(key)) continue;
   174→          if (mapData.layers.buildings[row]?.[col] !== tileType) continue;
   175→          
   176→          // Find bounding box of contiguous group
   177→          let minRow = row, maxRow = row, minCol = col, maxCol = col;
   178→          const toVisit: [number, number][] = [[row, col]];
   179→          
   180→          while (toVisit.length > 0) {
   181→            const [r, c] = toVisit.pop()!;
   182→            const k = `${r},${c}`;
   183→            if (visited.has(k)) continue;
   184→            if (mapData.layers.buildings[r]?.[c] !== tileType) continue;
   185→            visited.add(k);
   186→            minRow = Math.min(minRow, r);
   187→            maxRow = Math.max(maxRow, r);
   188→            minCol = Math.min(minCol, c);
   189→            maxCol = Math.max(maxCol, c);
   190→            toVisit.push([r - 1, c], [r + 1, c], [r, c - 1], [r, c + 1]);
   191→          }
   192→          
   193→          groups.push({ minRow, minCol, maxRow, maxCol });
   194→        }
   195→      }
   196→      
   197→      this.buildingGroups.set(tileType, groups);
   198→    }
   199→  }

   201→  async init(canvas: HTMLCanvasElement): Promise<void> {
   202→    this.canvas = canvas;
   203→    this.ctx = canvas.getContext('2d')!;
   204→    
   205→    // Set canvas size
   206→    this.resizeCanvas();
   207→    window.addEventListener('resize', () => this.resizeCanvas());
   208→
   209→    // Load all required sprites
   210→    const charsToLoad = ['student', 'ibara', 'mang-tenyo', 'kitchen-staff-1', 'kitchen-staff-2'];
   211→    for (const charId of charsToLoad) {
   212→      await spriteLoader.loadCharacter(charId);
   213→    }
   214→
   215→    // Create player entity
   216→    const spawn = mapData.playerSpawn;
   217→    this.player = {
   218→      id: 'player',
   219→      x: spawn.col * this.tileSize + this.tileSize / 2,
   220→      y: spawn.row * this.tileSize + this.tileSize / 2,
   221→      direction: 'north',
   222→      characterKey: 'student',
   223→      isMoving: false,
   224→      animationFrame: 0,
   225→      animationTimer: 0,
   226→      speed: 120,
   227→      visible: true,
   228→      isPlayer: true,
   229→      dialogueTriggered: false,
   230→    };
   231→    this.entities.set('player', this.player);
   232→
   233→    // Create NPC entities
   234→    const npcPositions = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   235→    for (const [npcId, pos] of Object.entries(npcPositions)) {
   236→      const entity: Entity = {
   237→        id: npcId,
   238→        x: pos.col * this.tileSize + this.tileSize / 2,
   239→        y: pos.row * this.tileSize + this.tileSize / 2,
   240→        direction: pos.direction as Direction,
   241→        characterKey: npcId,
   242→        isMoving: false,
   243→        animationFrame: 0,
   244→        animationTimer: 0,
   245→        speed: 0,
   246→        visible: !pos.appearsAfter || saveManager.isObjectiveCompleted(pos.appearsAfter),
   247→        isPlayer: false,
   248→        dialogueTriggered: false,
   249→        appearsAfterObjective: pos.appearsAfter,
   250→      };
   251→      this.entities.set(npcId, entity);
   252→    }
   253→
   254→    // Load saved progress
   255→    await saveManager.loadProgress();
   256→    const gameState = saveManager.getGameState();
   257→    this.chapterPhase = gameState.chapterPhase;
   258→    this.timeOfDay = gameState.timeOfDay;
   259→    
   260→    if (this.chapterPhase !== 'intro') {
   261→      this.introShown = true;
   262→    }
   263→
   264→    // Update entity visibility based on completed objectives
   265→    this._updateEntityVisibility();
   266→
   267→    // Mark already-triggered zones
   268→    this.triggerZones.forEach(zone => {
   269→      if (zone.requiresObjective && saveManager.isObjectiveCompleted(zone.requiresObjective)) {
   270→        // The prerequisite is done, but this zone itself may not have been triggered yet
   271→        // Only mark as triggered if the objective THIS zone completes is also done
   272→      }
   273→    });
   274→
   275→    // Bind input handlers
   276→    this._bindInput();
   277→
   278→    // Listen for objective completion events to spawn celebration bursts
   279→    gameEvents.on('quest:objectiveComplete', (objectiveId: unknown) => {
   280→      const id = objectiveId as string;
   281→      const loc = OBJECTIVE_LOCATIONS[id];
   282→      if (loc) {
   283→        // Spawn burst at the objective's location in world coords
   284→        this._spawnCelebrationBurst(
   285→          loc.col * this.tileSize + this.tileSize / 2,
   286→          loc.row * this.tileSize + this.tileSize / 2
   287→        );
   288→      }
   289→    });
   290→
   291→    // Start auto-save
   292→    saveManager.startAutoSave();
   293→
   294→    // Emit game ready
   295→    gameEvents.emit('game:ready');
   296→
   297→    // Initialize some particles
   298→    this._initParticles();
   299→
   300→    // Start game loop
   301→    this.lastTimestamp = performance.now();
   302→    this._gameLoop(this.lastTimestamp);
   303→  }

   305→  /**
   306→   * Spawn a golden celebration burst at the given world position.
   307→   * Used when an objective is completed.
   308→   */
   309→  private _spawnCelebrationBurst(worldX: number, worldY: number): void {
   310→    const colors = ['#FFD700', '#FFA500', '#FFEC8B', '#FFEE00', '#FFC125'];
   311→    const count = 28;
   312→    for (let i = 0; i < count; i++) {
   313→      const angle = (i / count) * Math.PI * 2 + Math.random() * 0.3;
   314→      const speed = 60 + Math.random() * 80;
   315→      this.burstParticles.push({
   316→        x: worldX,
   317→        y: worldY,
   318→        vx: Math.cos(angle) * speed,
   319→        vy: Math.sin(angle) * speed - 30, // Slight upward bias
   320→        life: 0,
   321→        maxLife: 1.0 + Math.random() * 0.6,
   322→        color: colors[Math.floor(Math.random() * colors.length)],
   323→        size: 2 + Math.random() * 3,
   324→        gravity: 80,
   325→      });
   326→    }
   327→  }
   328→
   329→  private _updateBurstParticles(dt: number): void {
   330→    for (let i = this.burstParticles.length - 1; i >= 0; i--) {
   331→      const p = this.burstParticles[i];
   332→      p.life += dt;
   333→      p.vy += p.gravity * dt;     // gravity pulls down
   334→      p.x += p.vx * dt;
   335→      p.y += p.vy * dt;
   336→      // Air resistance
   337→      p.vx *= 0.96;
   338→      p.vy *= 0.96;
   339→      if (p.life >= p.maxLife) {
   340→        this.burstParticles.splice(i, 1);
   341→      }
   342→    }
   343→  }
   344→
   345→  private _renderBurstParticles(ctx: CanvasRenderingContext2D): void {
   346→    for (const p of this.burstParticles) {
   347→      const screenX = p.x - this.cameraX;
   348→      const screenY = p.y - this.cameraY;
   349→      if (screenX < -20 || screenX > this.viewWidth + 20 || screenY < -20 || screenY > this.viewHeight + 20) continue;
   350→      const lifeRatio = p.life / p.maxLife;
   351→      const alpha = Math.max(0, 1 - lifeRatio);
   352→      // Glow
   353→      ctx.globalAlpha = alpha * 0.4;
   354→      ctx.fillStyle = p.color;
   355→      ctx.beginPath();
   356→      ctx.arc(screenX, screenY, p.size * 2.5, 0, Math.PI * 2);
   357→      ctx.fill();
   358→      // Core
   359→      ctx.globalAlpha = alpha;
   360→      ctx.fillStyle = p.color;
   361→      ctx.beginPath();
   362→      ctx.arc(screenX, screenY, p.size, 0, Math.PI * 2);
   363→      ctx.fill();
   364→    }
   365→    ctx.globalAlpha = 1;
   366→  }

   379→  private _bindInput(): void {
   380→    window.addEventListener('keydown', (e) => {
   381→      this.keys.add(e.key.toLowerCase());
   382→      // Space/Enter for NPC interaction
   383→      if (e.key === ' ' || e.key === 'Enter') {

   386→    });

   388→      this.keys.delete(e.key.toLowerCase());
   389→    });

   407→        case 'left': this.keys.add('a'); this.keys.delete('d'); this.keys.delete('w'); this.keys.delete('s'); break;
   408→        case 'right': this.keys.add('d'); this.keys.delete('a'); this.keys.delete('w'); this.keys.delete('s'); break;

   413→  triggerInteract(): void {

   417→  private _handleInteract(): void {
   418→    if (this.dialogueActive) return;
   419→    if (this.interactCooldown > 0) return;
   420→    
   421→    // Check if player is near an NPC they can talk to
   422→    if (!this.player) return;
   423→
   424→    for (const [npcId, npc] of this.entities) {
   425→      if (npc.isPlayer || !npc.visible) continue;
   426→      
   427→      const dist = this._distanceBetween(this.player, npc);
   428→      if (dist < this.tileSize * 2) {
   429→        this.interactCooldown = 500;
   430→        
   431→        // Determine which dialogue to trigger
   432→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   433→          this._startDialogue('mang-tenyo-first');
   434→          return;
   435→        }
   436→        // Mang Tenyo can be talked to again but won't repeat the quest dialogue
   437→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   438→          this._startDialogue('mang-tenyo-repeat');
   439→          return;
   440→        }
   441→      }
   442→    }
   443→  }
   444→
   445→  private _distanceBetween(a: Entity, b: Entity): number {
   446→    const dx = a.x - b.x;
   447→    const dy = a.y - b.y;
   448→    return Math.sqrt(dx * dx + dy * dy);
   449→  }

   451→  private _startDialogue(dialogueId: string): void {
   452→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   453→    if (!dialogue) {
   454→      console.warn(`Dialogue not found: ${dialogueId}`);
   455→      return;
   456→    }
   457→
   458→    this.dialogueActive = true;
   459→    this.currentDialogueId = dialogueId;
   460→    this.currentDialogueLineIndex = 0;
   461→
   462→    gameEvents.emit('dialogue:start', {
   463→      dialogueId,
   464→      line: dialogue.lines[0],
   465→      lineIndex: 0,
   466→      totalLines: dialogue.lines.length,
   467→    });
   468→  }
   469→
   470→  advanceDialogue(): void {
   471→    if (!this.currentDialogueId) return;
   472→    
   473→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];
   474→    if (!dialogue) return;
   475→
   476→    this.currentDialogueLineIndex++;
   477→
   478→    if (this.currentDialogueLineIndex >= dialogue.lines.length) {
   479→      // Dialogue complete
   480→      this._endDialogue(dialogue);
   481→      return;
   482→    }
   483→
   484→    const line = dialogue.lines[this.currentDialogueLineIndex];
   485→    
   486→    // Check for time transition
   487→    if (line.isTimeTransition) {
   488→      this.timeOfDay = 'morning';
   489→      this.chapterPhase = 'ibarra-sighting';
   490→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   491→      gameEvents.emit('time:transition', 'morning');
   492→    }
   493→
   494→    // Check for Ibarra sighting
   495→    if (line.isIbarraSighting) {
   496→      this.ibarraAppeared = true;
   497→      // Show Ibarra NPC
   498→      const ibaraEntity = this.entities.get('ibara');
   499→      if (ibaraEntity) {
   500→        ibaraEntity.visible = true;
   501→      }
   502→    }
   503→
   504→    gameEvents.emit('dialogue:line', {
   505→      dialogueId: this.currentDialogueId,
   506→      line,
   507→      lineIndex: this.currentDialogueLineIndex,
   508→      totalLines: dialogue.lines.length,
   509→    });
   510→  }

   512→  private _endDialogue(dialogue: { unlocksObjective?: string | null; id: string }): void {
   513→    this.dialogueActive = false;
   514→    this.currentDialogueId = null;
   515→    this.currentDialogueLineIndex = 0;
   516→
   517→    // Update chapter phase based on dialogue completion
   518→    if (dialogue.id === 'intro-text') {
   519→      this.chapterPhase = 'explore';
   520→      saveManager.updateGameState({ chapterPhase: 'explore' });
   521→    }
   522→
   523→    // Complete objective if dialogue unlocks one
   524→    if (dialogue.unlocksObjective) {
   525→      saveManager.completeObjective(dialogue.unlocksObjective);
   526→      
   527→      // Update entity visibility after objectives
   528→      this._updateEntityVisibility();
   529→
   530→      // Update chapter phase based on quest progress
   531→      if (dialogue.id === 'mang-tenyo-first') {
   532→        this.chapterPhase = 'explore';
   533→        saveManager.updateGameState({ chapterPhase: 'explore' });
   534→      }
   535→      if (dialogue.id === 'market-gossip') {
   536→        this.chapterPhase = 'gossip';
   537→        saveManager.updateGameState({ chapterPhase: 'gossip' });
   538→      }
   539→    }
   540→
   541→    // ===== Progressive Codex unlocks =====
   542→    // Codex entries unlock as the player meets characters and overhears gossip,
   543→    // so the Codex fills up DURING the chapter (not all at once at the end).
   544→    // This gives the player a sense of progression and discovery.
   545→    if (dialogue.id === 'mang-tenyo-first') {
   546→      // After meeting Mang Tenyo: unlock himself + he mentions Ibarra
   547→      saveManager.unlockCodexEntry('char.mang-tenyo');
   548→      saveManager.unlockCodexEntry('char.ibarra');
   549→      saveManager.addJournalEntry(
   550→        'I arrived in San Diego. A cart driver named Mang Tenyo warned me not to mention the name "Ibarra" too loudly — something happened at last night\'s reception.',
   551→        'ch1'
   552→      );
   553→    } else if (dialogue.id === 'market-gossip') {
   554→      // After overhearing gossip: unlock the people mentioned
   555→      saveManager.unlockCodexEntry('char.tiago');
   556→      saveManager.unlockCodexEntry('char.damaso');
   557→      saveManager.unlockCodexEntry('place.san-diego');
   558→      saveManager.addJournalEntry(
   559→        'At the market, Aling Nena and Mang Andres gossiped about Ibarra\'s homecoming reception. Padre Dámaso publicly insulted Ibarra — and he was once a close friend of Ibarra\'s late father. Capitán Tiago hosted the disastrous dinner.',
   560→        'ch1'
   561→      );
   562→    } else if (dialogue.id === 'ibarra-sighting') {

   635→  private _gameLoop(timestamp: number): void {
   636→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   637→    this.lastTimestamp = timestamp;
   638→    this.gameTime += deltaTime;
   639→
   640→    this._update(deltaTime);
   641→    this._render();
   642→
   643→    this.animationFrameId = requestAnimationFrame(this._gameLoop.bind(this));
   644→  }

   646→  private _update(dt: number): void {
   647→    if (!this.player) return;
   648→    
   649→    // Update interact cooldown
   650→    if (this.interactCooldown > 0) {
   651→      this.interactCooldown -= dt * 1000;
   652→    }
   653→
   654→    // Don't allow movement during dialogue
   655→    if (this.dialogueActive) return;
   656→
   657→    // Handle intro state
   658→    if (this.chapterPhase === 'intro' && !this.introShown) {
   659→      this.introShown = true;
   660→      this._startDialogue('intro-text');
   661→      return;
   662→    }
   663→
   664→    // Player movement
   665→    let moveX = 0;
   666→    let moveY = 0;
   667→
   668→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   669→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   670→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   671→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   672→
   673→    // Normalize diagonal movement
   674→    if (moveX !== 0 && moveY !== 0) {
   675→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   676→      moveX /= len;
   677→      moveY /= len;
   678→    }
   679→
   680→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   681→
   682→    if (this.player.isMoving) {
   683→      // Determine direction from movement vector
   684→      this.player.direction = this._vectorToDirection(moveX, moveY);
   685→      
   686→      const newX = this.player.x + moveX * this.player.speed * dt;
   687→      const newY = this.player.y + moveY * this.player.speed * dt;
   688→
   689→      // Collision check
   690→      if (!this._isBlocked(newX, newY)) {
   691→        this.player.x = newX;
   692→        this.player.y = newY;
   693→      }
   694→
   695→      // Update animation
   696→      this.player.animationTimer += dt;
   697→      const animConfig = characterData.characters.student.animations.Walk;
   698→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   699→        this.player.animationTimer = 0;
   700→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   701→      }
   702→    } else {
   703→      // Idle animation
   704→      this.player.animationTimer += dt;
   705→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   706→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   707→        this.player.animationTimer = 0;
   708→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   709→      }
   710→    }
   711→
   712→    // Update camera to follow player
   713→    this.cameraX = this.player.x - this.viewWidth / 2;
   714→    this.cameraY = this.player.y - this.viewHeight / 2;
   715→
   716→    // Clamp camera to map bounds
   717→    const mapPixelWidth = mapData.width * this.tileSize;
   718→    const mapPixelHeight = mapData.height * this.tileSize;
   719→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   720→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   721→
   722→    // NPC idle animations
   723→    for (const [npcId, npc] of this.entities) {
   724→      if (npc.isPlayer || !npc.visible) continue;
   725→      npc.animationTimer += dt;
   726→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   727→      if (animConfig) {
   728→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   729→          npc.animationTimer = 0;
   730→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   731→        }
   732→      }
   733→    }
   734→
   735→    // Check trigger zones
   736→    this._checkTriggerZones();
   737→
   738→    // Update particles
   739→    this._updateParticles(dt);
   740→
   741→    // Update burst particles (celebration effect)
   742→    this._updateBurstParticles(dt);
   743→
   744→    // Update game state in SaveManager
   745→    saveManager.updateGameState({
   746→      playerPosition: {
   747→        row: Math.floor(this.player.y / this.tileSize),
   748→        col: Math.floor(this.player.x / this.tileSize),
   749→      },
   750→      playerDirection: this.player.direction,
   751→      currentDialogue: this.currentDialogueId,
   752→      timeOfDay: this.timeOfDay,
   753→      chapterPhase: this.chapterPhase,
   754→    });
   755→  }
   756→
   757→  private _vectorToDirection(x: number, y: number): Direction {
   758→    // Map movement vector to 8-direction name
   759→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   760→    
   761→    if (angle >= -22.5 && angle < 22.5) return 'east';
   762→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   763→    if (angle >= 67.5 && angle < 112.5) return 'south';
   764→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   765→    if (angle >= 157.5 || angle < -157.5) return 'west';
   766→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   767→    if (angle >= -112.5 && angle < -67.5) return 'north';
   768→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   769→    return 'south';
   770→  }

   772→  private _isBlocked(x: number, y: number): boolean {
   773→    // Check collision map
   774→    const col = Math.floor(x / this.tileSize);
   775→    const row = Math.floor(y / this.tileSize);
   776→    
   777→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   778→    
   779→    // Check player sprite footprint (check a small area around center)
   780→    const checks = [
   781→      { x: x - 16, y: y - 16 },
   782→      { x: x + 16, y: y - 16 },
   783→      { x: x - 16, y: y + 16 },
   784→      { x: x + 16, y: y + 16 },
   785→    ];
   786→    
   787→    for (const check of checks) {
   788→      const cCol = Math.floor(check.x / this.tileSize);
   789→      const cRow = Math.floor(check.y / this.tileSize);
   790→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   791→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   792→    }
   793→    
   794→    return false;
   795→  }

   797→  private _checkTriggerZones(): void {
   798→    if (!this.player || this.dialogueActive) return;
   799→    
   800→    const playerRow = Math.floor(this.player.y / this.tileSize);
   801→    const playerCol = Math.floor(this.player.x / this.tileSize);
   802→
   803→    for (const zone of this.triggerZones) {
   804→      if (zone.triggered) continue;
   805→
   806→      // Check prerequisite objective
   807→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   808→
   809→      // Check if player is in the zone
   810→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   811→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   812→
   813→      // Or within radius of NPC
   814→      let nearNpc = false;
   815→      if (zone.radius > 0 && zone.npcId) {
   816→        const npc = this.entities.get(zone.npcId);
   817→        if (npc && npc.visible) {
   818→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   819→        }
   820→      }
   821→
   822→      if (inZone || nearNpc) {
   823→        zone.triggered = true;
   824→
   825→        switch (zone.type) {
   826→          case 'npc-interact':
   827→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   828→            break;
   829→          case 'auto-trigger':
   830→            // Generic auto-trigger: if zone has a dialogueId, start that dialogue
   831→            if (zone.dialogueId) {
   832→              // Special handling for Ibarra sighting (time transition)
   833→              if (zone.id === 'ibarra-sighting') {
   834→                this.timeOfDay = 'morning';
   835→                this.chapterPhase = 'ibarra-sighting';
   836→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   837→                this._updateEntityVisibility();
   838→              }
   839→              this._startDialogue(zone.dialogueId);
   840→            } else {
   841→              // Legacy hardcoded triggers as fallback
   842→              if (zone.id === 'market-gossip') {
   843→                this._startDialogue('market-gossip');
   844→              } else if (zone.id === 'ibarra-sighting') {
   845→                this.timeOfDay = 'morning';
   846→                this.chapterPhase = 'ibarra-sighting';
   847→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   848→                this._updateEntityVisibility();
   849→                this._startDialogue('ibarra-sighting');
   850→              }
   851→            }
   852→            break;
   853→        }
   854→      }
   855→    }
   856→  }

   860→  private _initParticles(): void {

   862→    for (let i = 0; i < 15; i++) {

   867→  private _spawnParticle(): void {
   868→    if (!this.player) return;
   869→    // Spawn particle near the camera view area
   870→    const cx = this.cameraX + this.viewWidth / 2;
   871→    const cy = this.cameraY + this.viewHeight / 2;
   872→    this.particles.push({
   873→      x: cx + (Math.random() - 0.5) * this.viewWidth * 0.8,
   874→      y: cy + (Math.random() - 0.5) * this.viewHeight * 0.8,
   875→      vx: (Math.random() - 0.5) * 8, // Slow drift
   876→      vy: -Math.random() * 4 - 1, // Slight upward drift
   877→      alpha: 0,
   878→      size: Math.random() * 2.5 + 1,
   879→      life: 0,
   880→      maxLife: Math.random() * 6 + 4, // 4-10 seconds
   881→    });
   882→  }
   883→
   884→  private _updateParticles(dt: number): void {
   885→    // Spawn new particles periodically
   886→    this.particleSpawnTimer += dt;
   887→    if (this.particleSpawnTimer > 0.3 && this.particles.length < 25) {
   888→      this.particleSpawnTimer = 0;
   889→      this._spawnParticle();
   890→    }
   891→
   892→    // Update existing particles
   893→    for (let i = this.particles.length - 1; i >= 0; i--) {
   894→      const p = this.particles[i];
   895→      p.life += dt;
   896→      p.x += p.vx * dt;
   897→      p.y += p.vy * dt;
   898→      // Gentle oscillation
   899→      p.x += Math.sin(this.gameTime * 0.5 + i) * 0.3;
   900→
   901→      // Fade in then fade out
   902→      const lifeRatio = p.life / p.maxLife;
   903→      if (lifeRatio < 0.2) {
   904→        p.alpha = lifeRatio / 0.2 * 0.4; // Fade in
   905→      } else if (lifeRatio > 0.7) {
   906→        p.alpha = (1 - lifeRatio) / 0.3 * 0.4; // Fade out
   907→      } else {
   908→        p.alpha = 0.4; // Visible
   909→      }
   910→
   911→      // Remove dead particles
   912→      if (p.life >= p.maxLife) {
   913→        this.particles.splice(i, 1);
   914→      }
   915→    }
   916→  }

   975→  /**
   976→   * Render a golden arrow at the screen edge pointing toward the next incomplete objective.
   977→   * Only shown when no dialogue is active and the player isn't standing in the objective zone.
   978→   */
   979→  private _renderWaypointArrow(ctx: CanvasRenderingContext2D): void {
   980→    if (!this.player || this.dialogueActive) return;
   981→    if (this.chapterPhase === 'intro' || this.chapterPhase === 'complete') return;
   982→
   983→    // Find the next incomplete objective
   984→    const objectivesOrder = ['obj.ch1.follow_tenyo', 'obj.ch1.overhear_gossip', 'obj.ch1.see_ibarra'];
   985→    const nextObjId = objectivesOrder.find(id => !saveManager.isObjectiveCompleted(id));
   986→    if (!nextObjId) return;
   987→
   988→    const loc = OBJECTIVE_LOCATIONS[nextObjId];
   989→    if (!loc) return;
   990→
   991→    const targetX = loc.col * this.tileSize + this.tileSize / 2;
   992→    const targetY = loc.row * this.tileSize + this.tileSize / 2;
   993→
   994→    // Vector from player to target
   995→    const dx = targetX - this.player.x;
   996→    const dy = targetY - this.player.y;
   997→    const dist = Math.sqrt(dx * dx + dy * dy);
   998→
   999→    // Don't show arrow if very close
  1000→    if (dist < this.tileSize * 2.5) return;
  1001→
  1002→    // Compute target position on screen (may be off-screen)
  1003→    const screenTargetX = targetX - this.cameraX;
  1004→    const screenTargetY = targetY - this.cameraY;
  1005→
  1006→    const cx = this.viewWidth / 2;
  1007→    const cy = this.viewHeight / 2;
  1008→
  1009→    // If the target is on screen, don't show the edge arrow (the indicator is enough)
  1010→    const onScreen = screenTargetX > 40 && screenTargetX < this.viewWidth - 40 &&
  1011→                     screenTargetY > 80 && screenTargetY < this.viewHeight - 80;
  1012→    if (onScreen) return;
  1013→
  1014→    // Direction from center of screen to target
  1015→    const angle = Math.atan2(dy, dx);
  1016→
  1017→    // Place arrow at a fixed radius from screen center, clamped to a margin inside the screen
  1018→    const margin = 80;
  1019→    const maxRadiusX = this.viewWidth / 2 - margin;
  1020→    const maxRadiusY = this.viewHeight / 2 - margin;
  1021→    // Project the direction onto the screen-edge ellipse
  1022→    // Solve for t such that (t*cos(angle), t*sin(angle)) lies on the ellipse (maxRX, maxRY)
  1023→    const cosA = Math.cos(angle);
  1024→    const sinA = Math.sin(angle);
  1025→    const t = 1 / Math.sqrt((cosA * cosA) / (maxRadiusX * maxRadiusX) + (sinA * sinA) / (maxRadiusY * maxRadiusY));
  1026→    const arrowX = cx + cosA * t;
  1027→    const arrowY = cy + sinA * t;
  1028→
  1029→    // Pulsing effect
  1030→    const pulse = Math.sin(this.gameTime * 4) * 0.15 + 0.85;
  1031→
  1032→    ctx.save();
  1033→    ctx.translate(arrowX, arrowY);
  1034→    ctx.rotate(angle);
  1035→
  1036→    // Outer glow
  1037→    ctx.fillStyle = `rgba(255,215,0,${0.15 * pulse})`;
  1038→    ctx.beginPath();
  1039→    ctx.arc(0, 0, 28, 0, Math.PI * 2);
  1040→    ctx.fill();
  1041→
  1042→    // Arrow shadow
  1043→    ctx.fillStyle = 'rgba(0,0,0,0.5)';
  1044→    ctx.beginPath();
  1045→    ctx.moveTo(18, 0);
  1046→    ctx.lineTo(-8, -12);
  1047→    ctx.lineTo(-4, 0);
  1048→    ctx.lineTo(-8, 12);
  1049→    ctx.closePath();
  1050→    ctx.fill();
  1051→
  1052→    // Arrow body
  1053→    ctx.fillStyle = `rgba(255,215,0,${pulse})`;
  1054→    ctx.strokeStyle = 'rgba(180,120,0,1)';
  1055→    ctx.lineWidth = 1.5;
  1056→    ctx.beginPath();
  1057→    ctx.moveTo(15, 0);
  1058→    ctx.lineTo(-10, -10);
  1059→    ctx.lineTo(-6, 0);
  1060→    ctx.lineTo(-10, 10);
  1061→    ctx.closePath();
  1062→    ctx.fill();
  1063→    ctx.stroke();
  1064→
  1065→    ctx.restore();
  1066→
  1067→    // Distance label below arrow
  1068→    const distTiles = Math.round(dist / this.tileSize);
  1069→    ctx.font = 'bold 10px "Geist", sans-serif';
  1070→    ctx.fillStyle = 'rgba(0,0,0,0.6)';
  1071→    ctx.fillRect(arrowX - 22, arrowY + 14, 44, 14);
  1072→    ctx.fillStyle = '#FFD700';
  1073→    ctx.textAlign = 'center';
  1074→    ctx.fillText(`${distTiles}m →`, arrowX, arrowY + 24);
  1075→  }

  1079→  private _renderSkyGradient(ctx: CanvasRenderingContext2D): void {
  1080→    // Create a gradient for the sky/background
  1081→    const gradient = ctx.createLinearGradient(0, 0, 0, this.viewHeight);
  1082→    
  1083→    if (this.timeOfDay === 'afternoon') {
  1084→      // Warm golden afternoon: deep sky blue at top, warm golden at bottom
  1085→      gradient.addColorStop(0, '#6BB3D9');    // Deep afternoon sky
  1086→      gradient.addColorStop(0.4, '#8FC8E8');  // Mid sky
  1087→      gradient.addColorStop(0.7, '#C4E0E8');  // Lighter
  1088→      gradient.addColorStop(1, '#E8D4A0');    // Warm golden horizon
  1089→    } else {
  1090→      // Cool blue-white morning: pale sky, slightly warm at horizon
  1091→      gradient.addColorStop(0, '#B8D4E8');    // Cool morning sky

  1103→  private _renderMap(ctx: CanvasRenderingContext2D): void {
  1104→    // Only render visible tiles (camera culling)
  1105→    const startCol = Math.floor(this.cameraX / this.tileSize);
  1106→    const startRow = Math.floor(this.cameraY / this.tileSize);
  1107→    const endCol = Math.min(mapData.width, startCol + Math.ceil(this.viewWidth / this.tileSize) + 1);
  1108→    const endRow = Math.min(mapData.height, startRow + Math.ceil(this.viewHeight / this.tileSize) + 1);
  1109→
  1110→    // Draw ground layer with detailed patterns
  1111→    for (let row = startRow; row < endRow; row++) {
  1112→      for (let col = startCol; col < endCol; col++) {
  1113→        if (row < 0 || col < 0) continue;
  1114→        const tileType = mapData.layers.ground[row]?.[col];
  1115→        if (tileType && tileType !== 0) {
  1116→          const x = col * this.tileSize - this.cameraX;
  1117→          const y = row * this.tileSize - this.cameraY;
  1118→          
  1119→          switch (tileType) {
  1120→            case 1: // Dirt/stone (ochre/tan) - building interior floor
  1121→              this._renderDirtFloor(ctx, x, y, row, col);
  1122→              break;
  1123→            case 2: // Grass (green) - outskirts/wild areas
  1124→              this._renderGrass(ctx, x, y, row, col);
  1125→              break;
  1126→            case 3: // Plaza (tan/dirt) - walkable area
  1127→              this._renderPlazaCobblestone(ctx, x, y, row, col);
  1128→              break;
  1129→            case 4: // Market stalls (brown)
  1130→              this._renderMarketStalls(ctx, x, y, row, col);
  1131→              break;
  1132→            default:
  1133→              const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
  1134→              ctx.fillStyle = color;
  1135→              ctx.fillRect(x, y, this.tileSize, this.tileSize);
  1136→          }
  1137→        }
  1138→      }
  1139→    }
  1140→
  1141→    // Draw buildings layer (flat base color - details rendered separately)
  1142→    for (let row = startRow; row < endRow; row++) {
  1143→      for (let col = startCol; col < endCol; col++) {
  1144→        if (row < 0 || col < 0) continue;
  1145→        const tileType = mapData.layers.buildings[row]?.[col];
  1146→        if (tileType && tileType !== 0) {
  1147→          const x = col * this.tileSize - this.cameraX;
  1148→          const y = row * this.tileSize - this.cameraY;
  1149→
  1150→          // Base building fill
  1151→          const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
  1152→          ctx.fillStyle = color;
  1153→          ctx.fillRect(x, y, this.tileSize, this.tileSize);
  1154→        }
  1155→      }
  1156→    }
  1157→  }

  1688→  private _renderEntity(ctx: CanvasRenderingContext2D, entity: Entity): void {
  1689→    const screenX = entity.x - this.cameraX;
  1690→    const screenY = entity.y - this.cameraY;
  1691→
  1692→    // Don't render if off screen
  1693→    if (screenX < -150 || screenX > this.viewWidth + 150 || 
  1694→        screenY < -150 || screenY > this.viewHeight + 150) return;
  1695→
  1696→    const anim = entity.isMoving ? 'Walk' : 'Breathing_Idle';
  1697→    const dir = entity.direction;
  1698→    const frame = entity.animationFrame;
  1699→
  1700→    const sprite = spriteLoader.getFrame(entity.characterKey, anim, dir, frame);
  1701→    
  1702→    if (sprite) {
  1703→      // Center the sprite on the entity position
  1704→      const spriteW = sprite.width || characterData.characters[entity.characterKey as keyof typeof characterData.characters]?.size?.width || 72;
  1705→      const spriteH = sprite.height || characterData.characters[entity.characterKey as keyof typeof characterData.characters]?.size?.height || 72;
  1706→      
  1707→      // Scale sprites up for visibility (3x scale for better visibility)
  1708→      const scale = 3;
  1709→      const drawW = spriteW * scale;
  1710→      const drawH = spriteH * scale;
  1711→      
  1712→      // Entity shadow (semi-transparent dark oval on the ground)
  1713→      ctx.fillStyle = 'rgba(0,0,0,0.18)';
  1714→      ctx.beginPath();
  1715→      ctx.ellipse(screenX, screenY + drawH * 0.25, drawW * 0.35, 6, 0, 0, Math.PI * 2);
  1716→      ctx.fill();
  1717→      
  1718→      // Player character subtle highlight/outline
  1719→      if (entity.isPlayer) {
  1720→        // Soft glow around player
  1721→        ctx.fillStyle = 'rgba(255,220,100,0.08)';
  1722→        ctx.beginPath();
  1723→        ctx.ellipse(screenX, screenY, drawW * 0.6, drawH * 0.5, 0, 0, Math.PI * 2);
  1724→        ctx.fill();
  1725→        
  1726→        // Subtle outline stroke
  1727→        ctx.save();
  1728→        ctx.globalCompositeOperation = 'source-over';
  1729→        // Draw sprite with a slight offset outline for border effect
  1730→        ctx.strokeStyle = 'rgba(255,200,80,0.3)';
  1731→        ctx.lineWidth = 2;

  1734→        ctx.stroke();
  1735→        ctx.restore();
  1736→      }
  1737→      
  1738→      // Draw the sprite
  1739→      ctx.drawImage(sprite, screenX - drawW / 2, screenY - drawH / 2, drawW, drawH);
  1740→    }
  1741→
  1742→    // Entity name label (not for player)
  1743→    if (!entity.isPlayer) {
  1744→      const displayName = characterData.characters[entity.characterKey as keyof typeof characterData.characters]?.displayName || entity.characterKey;
  1745→      
  1746→      // Name background panel

  1748→      const nameWidth = ctx.measureText(displayName).width;
  1749→      const nameX = screenX;
  1750→      const nameY = screenY - 75;

  1755→      
  1756→      // Name text

  1758→      ctx.textAlign = 'center';
  1759→      ctx.fillText(displayName, nameX, nameY);
  1760→    }
  1761→  }
