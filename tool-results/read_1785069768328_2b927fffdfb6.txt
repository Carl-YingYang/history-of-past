     1→// GameEngine - Core game loop, Canvas rendering, input handling, and scene management
     2→// Pokémon-style top-down 2D RPG engine for Project Noor

     4→import { gameEvents } from './eventBus';

    11→// Direction constants
    12→const DIRECTIONS = ['south', 'south-east', 'east', 'north-east', 'north', 'north-west', 'west', 'south-west'] as const;
    13→type Direction = typeof DIRECTIONS[number];
    14→
    15→interface Entity {
    16→  id: string;
    17→  x: number;
    18→  y: number;
    19→  direction: Direction;
    20→  characterKey: string;

    22→  animationFrame: number;
    23→  animationTimer: number;
    24→  speed: number;
    25→  visible: boolean;
    26→  isPlayer: boolean;
    27→  dialogueTriggered: boolean;
    28→  appearsAfterObjective?: string;
    29→}

    31→interface TriggerZone {
    32→  id: string;
    33→  type: string;
    34→  npcId?: string;
    35→  row: number;
    36→  col: number;
    37→  width: number;
    38→  height: number;
    39→  radius: number;

    41→  triggersTimeTransition?: boolean;

    45→class GameEngine {
    46→  private canvas: HTMLCanvasElement | null = null;
    47→  private ctx: CanvasRenderingContext2D | null = null;
    48→  private animationFrameId: number | null = null;
    49→  private lastTimestamp: number = 0;
    50→  private tileSize: number = 48;
    51→  private cameraX: number = 0;
    52→  private cameraY: number = 0;
    53→
    54→  // Entities
    55→  private entities: Map<string, Entity> = new Map();
    56→  private player: Entity | null = null;
    57→
    58→  // Input state
    59→  private keys: Set<string> = new Set();
    60→
    61→  // Game state
    62→  private chapterPhase: string = 'intro';
    63→  private introShown: boolean = false;
    64→  private triggerZones: TriggerZone[] = [];
    65→  private dialogueActive: boolean = false;
    66→  private currentDialogueId: string | null = null;
    67→  private currentDialogueLineIndex: number = 0;
    68→  private pendingDialogue: string | null = null;
    69→  private timeOfDay: 'afternoon' | 'morning' = 'afternoon';
    70→  private ibarraAppeared: boolean = false;
    71→
    72→  // Canvas dimensions
    73→  private viewWidth: number = 960;
    74→  private viewHeight: number = 768;
    75→
    76→  // NPC interaction
    77→  private interactCooldown: number = 0;
    78→
    79→  constructor() {
    80→    // Initialize trigger zones from map data
    81→    this.triggerZones = mapData.triggerZones.map(z => ({ ...z, triggered: false }));
    82→  }
    83→
    84→  async init(canvas: HTMLCanvasElement): Promise<void> {
    85→    this.canvas = canvas;
    86→    this.ctx = canvas.getContext('2d')!;
    87→    
    88→    // Set canvas size
    89→    this.resizeCanvas();
    90→    window.addEventListener('resize', () => this.resizeCanvas());
    91→
    92→    // Load all required sprites
    93→    const charsToLoad = ['student', 'ibara', 'mang-tenyo', 'kitchen-staff-1', 'kitchen-staff-2'];
    94→    for (const charId of charsToLoad) {
    95→      await spriteLoader.loadCharacter(charId);
    96→    }
    97→
    98→    // Create player entity
    99→    const spawn = mapData.playerSpawn;
   100→    this.player = {
   101→      id: 'player',
   102→      x: spawn.col * this.tileSize + this.tileSize / 2,
   103→      y: spawn.row * this.tileSize + this.tileSize / 2,
   104→      direction: 'north',
   105→      characterKey: 'student',
   106→      isMoving: false,
   107→      animationFrame: 0,
   108→      animationTimer: 0,
   109→      speed: 120,
   110→      visible: true,
   111→      isPlayer: true,
   112→      dialogueTriggered: false,
   113→    };
   114→    this.entities.set('player', this.player);
   115→
   116→    // Create NPC entities
   117→    const npcPositions = mapData.npcPositions as Record<string, { row: number; col: number; direction: string; appearsAfter?: string }>;
   118→    for (const [npcId, pos] of Object.entries(npcPositions)) {
   119→      const entity: Entity = {
   120→        id: npcId,
   121→        x: pos.col * this.tileSize + this.tileSize / 2,
   122→        y: pos.row * this.tileSize + this.tileSize / 2,
   123→        direction: pos.direction as Direction,
   124→        characterKey: npcId,
   125→        isMoving: false,
   126→        animationFrame: 0,
   127→        animationTimer: 0,
   128→        speed: 0,
   129→        visible: !pos.appearsAfter || saveManager.isObjectiveCompleted(pos.appearsAfter),
   130→        isPlayer: false,
   131→        dialogueTriggered: false,
   132→        appearsAfterObjective: pos.appearsAfter,
   133→      };
   134→      this.entities.set(npcId, entity);
   135→    }
   136→
   137→    // Load saved progress
   138→    await saveManager.loadProgress();
   139→    const gameState = saveManager.getGameState();
   140→    this.chapterPhase = gameState.chapterPhase;
   141→    this.timeOfDay = gameState.timeOfDay;
   142→    
   143→    if (this.chapterPhase !== 'intro') {
   144→      this.introShown = true;
   145→    }
   146→
   147→    // Update entity visibility based on completed objectives
   148→    this._updateEntityVisibility();
   149→
   150→    // Mark already-triggered zones
   151→    this.triggerZones.forEach(zone => {
   152→      if (zone.requiresObjective && saveManager.isObjectiveCompleted(zone.requiresObjective)) {
   153→        // The prerequisite is done, but this zone itself may not have been triggered yet
   154→        // Only mark as triggered if the objective THIS zone completes is also done
   155→      }
   156→    });
   157→
   158→    // Bind input handlers
   159→    this._bindInput();
   160→
   161→    // Start auto-save
   162→    saveManager.startAutoSave();
   163→
   164→    // Emit game ready
   165→    gameEvents.emit('game:ready');
   166→
   167→    // Start game loop
   168→    this.lastTimestamp = performance.now();
   169→    this._gameLoop(this.lastTimestamp);
   170→  }

   172→  private resizeCanvas(): void {
   173→    if (!this.canvas) return;
   174→    const container = this.canvas.parentElement;
   175→    if (container) {
   176→      this.viewWidth = container.clientWidth;
   177→      this.viewHeight = container.clientHeight;
   178→    }
   179→    this.canvas.width = this.viewWidth;
   180→    this.canvas.height = this.viewHeight;
   181→  }
   182→
   183→  private _bindInput(): void {
   184→    window.addEventListener('keydown', (e) => {
   185→      this.keys.add(e.key.toLowerCase());
   186→      // Space/Enter for NPC interaction
   187→      if (e.key === ' ' || e.key === 'Enter') {
   188→        this._handleInteract();
   189→      }
   190→    });
   191→    window.addEventListener('keyup', (e) => {
   192→      this.keys.delete(e.key.toLowerCase());
   193→    });
   194→  }
   195→
   196→  private _handleInteract(): void {
   197→    if (this.dialogueActive) return;
   198→    if (this.interactCooldown > 0) return;
   199→    
   200→    // Check if player is near an NPC they can talk to
   201→    if (!this.player) return;
   202→
   203→    for (const [npcId, npc] of this.entities) {
   204→      if (npc.isPlayer || !npc.visible) continue;
   205→      
   206→      const dist = this._distanceBetween(this.player, npc);
   207→      if (dist < this.tileSize * 2) {
   208→        this.interactCooldown = 500;
   209→        
   210→        // Determine which dialogue to trigger
   211→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   212→          this._startDialogue('mang-tenyo-first');
   213→          return;
   214→        }
   215→        // Mang Tenyo can be talked to again but won't repeat the quest dialogue
   216→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   217→          this._startDialogue('mang-tenyo-repeat');
   218→          return;
   219→        }
   220→      }
   221→    }
   222→  }
   223→
   224→  private _distanceBetween(a: Entity, b: Entity): number {
   225→    const dx = a.x - b.x;
   226→    const dy = a.y - b.y;
   227→    return Math.sqrt(dx * dx + dy * dy);
   228→  }
   229→
   230→  private _startDialogue(dialogueId: string): void {
   231→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   232→    if (!dialogue) {
   233→      console.warn(`Dialogue not found: ${dialogueId}`);
   234→      return;
   235→    }
   236→
   237→    this.dialogueActive = true;
   238→    this.currentDialogueId = dialogueId;
   239→    this.currentDialogueLineIndex = 0;
   240→
   241→    gameEvents.emit('dialogue:start', {
   242→      dialogueId,
   243→      line: dialogue.lines[0],
   244→      lineIndex: 0,
   245→      totalLines: dialogue.lines.length,
   246→    });
   247→  }
   248→
   249→  advanceDialogue(): void {
   250→    if (!this.currentDialogueId) return;
   251→    
   252→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];
   253→    if (!dialogue) return;
   254→
   255→    this.currentDialogueLineIndex++;
   256→
   257→    if (this.currentDialogueLineIndex >= dialogue.lines.length) {
   258→      // Dialogue complete
   259→      this._endDialogue(dialogue);
   260→      return;
   261→    }
   262→
   263→    const line = dialogue.lines[this.currentDialogueLineIndex];
   264→    
   265→    // Check for time transition
   266→    if (line.isTimeTransition) {
   267→      this.timeOfDay = 'morning';
   268→      this.chapterPhase = 'ibarra-sighting';
   269→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   270→      gameEvents.emit('time:transition', 'morning');
   271→    }
   272→
   273→    // Check for Ibarra sighting
   274→    if (line.isIbarraSighting) {
   275→      this.ibarraAppeared = true;
   276→      // Show Ibarra NPC
   277→      const ibaraEntity = this.entities.get('ibara');
   278→      if (ibaraEntity) {
   279→        ibaraEntity.visible = true;
   280→      }
   281→    }
   282→
   283→    gameEvents.emit('dialogue:line', {
   284→      dialogueId: this.currentDialogueId,
   285→      line,
   286→      lineIndex: this.currentDialogueLineIndex,
   287→      totalLines: dialogue.lines.length,
   288→    });
   289→  }
   290→
   291→  private _endDialogue(dialogue: { unlocksObjective?: string | null; id: string }): void {
   292→    this.dialogueActive = false;
   293→    this.currentDialogueId = null;
   294→    this.currentDialogueLineIndex = 0;
   295→
   296→    // Complete objective if dialogue unlocks one
   297→    if (dialogue.unlocksObjective) {
   298→      saveManager.completeObjective(dialogue.unlocksObjective);
   299→      
   300→      // Update entity visibility after objectives
   301→      this._updateEntityVisibility();
   302→    }
   303→
   304→    gameEvents.emit('dialogue:end', dialogue.id);
   305→
   306→    // Check if this was the gossip dialogue - enable Ibarra sighting zone
   307→    if (dialogue.id === 'market-gossip') {
   308→      // The market gossip zone should now be triggered, check for ibarra sighting zone
   309→      this.triggerZones.forEach(z => {
   310→        if (z.id === 'market-gossip') z.triggered = true;
   311→      });
   312→    }
   313→
   314→    // Check if chapter is complete after Ibarra sighting dialogue
   315→    if (dialogue.id === 'ibarra-sighting') {
   316→      this._triggerChapterComplete();
   317→    }
   318→  }
   319→
   320→  private _updateEntityVisibility(): void {
   321→    for (const [npcId, entity] of this.entities) {
   322→      if (entity.appearsAfterObjective) {
   323→        entity.visible = saveManager.isObjectiveCompleted(entity.appearsAfterObjective);
   324→      }
   325→    }
   326→    
   327→    // Special: Ibarra appears after gossip objective AND time transition
   328→    const ibara = this.entities.get('ibara');
   329→    if (ibara && this.timeOfDay === 'morning' && saveManager.isObjectiveCompleted('obj.ch1.overhear_gossip')) {
   330→      ibara.visible = true;
   331→    }
   332→  }
   333→
   334→  private _triggerChapterComplete(): void {
   335→    // Run the end-of-chapter loop (Section 8)
   336→    // 1. Quiz is triggered via event
   337→    gameEvents.emit('quiz:start', 'ch1');
   338→    // The quiz component handles the rest (XP, codex, journal, medal, save)
   339→  }
   340→
   341→  completeChapterLoop(): void {
   342→    // Called after quiz is completed
   343→    // 2. Award XP
   344→    saveManager.awardXp(60);
   345→    // 3. Unlock codex entries
   346→    saveManager.unlockCodexEntry('char.ibarra');
   347→    saveManager.unlockCodexEntry('char.tiago');
   348→    saveManager.unlockCodexEntry('char.damaso');
   349→    // 4. Journal entry
   350→    saveManager.addJournalEntry(
   351→      "Mang Tenyo warned me not to mention the Ibarra name loudly. Something happened at last night's reception. I overheard the kitchen staff talking about Padre Dámaso's rudeness to Ibarra. Then I saw Ibarra himself crossing the plaza — a young man carrying ideas from Europe that the friars fear.",
   352→      'ch1'
   353→    );
   354→    // 5. Chapter medal
   355→    saveManager.awardChapterMedal('ch1', 'Listener');
   356→    // 6. Complete quest
   357→    saveManager.completeQuest('mq.ch1.arrival');
   358→    saveManager.completeChapter('ch1');
   359→    // 7. Save progress
   360→    saveManager.saveProgress();
   361→    
   362→    this.chapterPhase = 'complete';
   363→    gameEvents.emit('chapter:complete', 'ch1');
   364→  }
   365→
   366→  private _gameLoop(timestamp: number): void {
   367→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   368→    this.lastTimestamp = timestamp;
   369→
   370→    this._update(deltaTime);
   371→    this._render();
   372→
   373→    this.animationFrameId = requestAnimationFrame(this._gameLoop.bind(this));
   374→  }
   375→
   376→  private _update(dt: number): void {
   377→    if (!this.player) return;
   378→    
   379→    // Update interact cooldown
   380→    if (this.interactCooldown > 0) {
   381→      this.interactCooldown -= dt * 1000;
   382→    }
   383→
   384→    // Don't allow movement during dialogue
   385→    if (this.dialogueActive) return;
   386→
   387→    // Handle intro state
   388→    if (this.chapterPhase === 'intro' && !this.introShown) {
   389→      this.introShown = true;
   390→      this._startDialogue('intro-text');
   391→      return;
   392→    }
   393→
   394→    // Player movement
   395→    let moveX = 0;
   396→    let moveY = 0;
   397→
   398→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   399→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   400→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   401→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   402→
   403→    // Normalize diagonal movement
   404→    if (moveX !== 0 && moveY !== 0) {
   405→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   406→      moveX /= len;
   407→      moveY /= len;
   408→    }
   409→
   410→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   411→
   412→    if (this.player.isMoving) {
   413→      // Determine direction from movement vector
   414→      this.player.direction = this._vectorToDirection(moveX, moveY);
   415→      
   416→      const newX = this.player.x + moveX * this.player.speed * dt;
   417→      const newY = this.player.y + moveY * this.player.speed * dt;
   418→
   419→      // Collision check
   420→      if (!this._isBlocked(newX, newY)) {
   421→        this.player.x = newX;
   422→        this.player.y = newY;
   423→      }
   424→
   425→      // Update animation
   426→      this.player.animationTimer += dt;
   427→      const animConfig = characterData.characters.student.animations.Walk;
   428→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   429→        this.player.animationTimer = 0;
   430→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   431→      }
   432→    } else {
   433→      // Idle animation
   434→      this.player.animationTimer += dt;
   435→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   436→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   437→        this.player.animationTimer = 0;
   438→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   439→      }
   440→    }
   441→
   442→    // Update camera to follow player
   443→    this.cameraX = this.player.x - this.viewWidth / 2;
   444→    this.cameraY = this.player.y - this.viewHeight / 2;
   445→
   446→    // Clamp camera to map bounds
   447→    const mapPixelWidth = mapData.width * this.tileSize;
   448→    const mapPixelHeight = mapData.height * this.tileSize;
   449→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   450→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   451→
   452→    // NPC idle animations
   453→    for (const [npcId, npc] of this.entities) {
   454→      if (npc.isPlayer || !npc.visible) continue;
   455→      npc.animationTimer += dt;
   456→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   457→      if (animConfig) {
   458→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   459→          npc.animationTimer = 0;
   460→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   461→        }
   462→      }
   463→    }
   464→
   465→    // Check trigger zones
   466→    this._checkTriggerZones();
   467→
   468→    // Update game state in SaveManager
   469→    saveManager.updateGameState({
   470→      playerPosition: {
   471→        row: Math.floor(this.player.y / this.tileSize),
   472→        col: Math.floor(this.player.x / this.tileSize),
   473→      },
   474→      playerDirection: this.player.direction,
   475→      currentDialogue: this.currentDialogueId,
   476→      timeOfDay: this.timeOfDay,
   477→      chapterPhase: this.chapterPhase,
   478→    });
   479→  }
   480→
   481→  private _vectorToDirection(x: number, y: number): Direction {
   482→    // Map movement vector to 8-direction name
   483→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   484→    // Adjust for south-first (our south is positive Y)
   485→    // Angle: south=90, south-east=45ish... need to remap
   486→    
   487→    if (angle >= -22.5 && angle < 22.5) return 'east';
   488→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   489→    if (angle >= 67.5 && angle < 112.5) return 'south';
   490→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   491→    if (angle >= 157.5 || angle < -157.5) return 'west';
   492→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   493→    if (angle >= -112.5 && angle < -67.5) return 'north';
   494→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   495→    return 'south';
   496→  }
   497→
   498→  private _isBlocked(x: number, y: number): boolean {
   499→    // Check collision map
   500→    const col = Math.floor(x / this.tileSize);
   501→    const row = Math.floor(y / this.tileSize);
   502→    
   503→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   504→    
   505→    // Check player sprite footprint (check a small area around center)
   506→    const checks = [
   507→      { x: x - 16, y: y - 16 },
   508→      { x: x + 16, y: y - 16 },
   509→      { x: x - 16, y: y + 16 },
   510→      { x: x + 16, y: y + 16 },
   511→    ];
   512→    
   513→    for (const check of checks) {
   514→      const cCol = Math.floor(check.x / this.tileSize);
   515→      const cRow = Math.floor(check.y / this.tileSize);
   516→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   517→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   518→    }
   519→    
   520→    return false;
   521→  }
   522→
   523→  private _checkTriggerZones(): void {
   524→    if (!this.player || this.dialogueActive) return;
   525→    
   526→    const playerRow = Math.floor(this.player.y / this.tileSize);
   527→    const playerCol = Math.floor(this.player.x / this.tileSize);
   528→
   529→    for (const zone of this.triggerZones) {
   530→      if (zone.triggered) continue;
   531→
   532→      // Check prerequisite objective
   533→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   534→
   535→      // Check if player is in the zone
   536→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   537→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   538→
   539→      // Or within radius of NPC
   540→      let nearNpc = false;
   541→      if (zone.radius > 0 && zone.npcId) {
   542→        const npc = this.entities.get(zone.npcId);
   543→        if (npc && npc.visible) {
   544→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   545→        }
   546→      }
   547→
   548→      if (inZone || nearNpc) {
   549→        zone.triggered = true;
   550→
   551→        switch (zone.type) {
   552→          case 'npc-interact':
   553→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   554→            break;
   555→          case 'auto-trigger':
   556→            if (zone.id === 'market-gossip') {
   557→              this._startDialogue('market-gossip');
   558→            } else if (zone.id === 'ibarra-sighting') {
   559→              // Time transition first
   560→              this.timeOfDay = 'morning';
   561→              this.chapterPhase = 'ibarra-sighting';
   562→              saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   563→              this._updateEntityVisibility();
   564→              this._startDialogue('ibarra-sighting');
   565→            }
   566→            break;
   567→        }
   568→      }
   569→    }
   570→  }
   571→
   572→  private _render(): void {
   573→    if (!this.ctx || !this.canvas) return;
   574→    const ctx = this.ctx;
   575→
   576→    // Clear canvas
   577→    ctx.clearRect(0, 0, this.viewWidth, this.viewHeight);
   578→
   579→    // Draw sky/background based on time of day
   580→    if (this.timeOfDay === 'afternoon') {
   581→      ctx.fillStyle = '#87CEEB'; // Light blue afternoon sky
   582→    } else {
   583→      ctx.fillStyle = '#E8D5B7'; // Warm morning light
   584→    }
   585→    ctx.fillRect(0, 0, this.viewWidth, this.viewHeight);
   586→
   587→    // Render map layers
   588→    this._renderMap(ctx);
   589→
   590→    // Render building labels
   591→    this._renderBuildingLabels(ctx);
   592→
   593→    // Render entities (sorted by Y for depth)
   594→    const sortedEntities = [...this.entities.values()]
   595→      .filter(e => e.visible)
   596→      .sort((a, b) => a.y - b.y);
   597→
   598→    for (const entity of sortedEntities) {
   599→      this._renderEntity(ctx, entity);
   600→    }
   601→
   602→    // Render NPC interaction indicators
   603→    this._renderInteractionIndicators(ctx);
   604→
   605→    // Render trigger zone hints (subtle visual cues)
   606→    this._renderTriggerHints(ctx);
   607→  }
   608→
   609→  private _renderMap(ctx: CanvasRenderingContext2D): void {
   610→    // Only render visible tiles (camera culling)
   611→    const startCol = Math.floor(this.cameraX / this.tileSize);
   612→    const startRow = Math.floor(this.cameraY / this.tileSize);
   613→    const endCol = Math.min(mapData.width, startCol + Math.ceil(this.viewWidth / this.tileSize) + 1);
   614→    const endRow = Math.min(mapData.height, startRow + Math.ceil(this.viewHeight / this.tileSize) + 1);
   615→
   616→    // Draw ground layer
   617→    for (let row = startRow; row < endRow; row++) {
   618→      for (let col = startCol; col < endCol; col++) {
   619→        if (row < 0 || col < 0) continue;
   620→        const tileType = mapData.layers.ground[row]?.[col];
   621→        if (tileType && tileType !== 0) {
   622→          const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
   623→          const x = col * this.tileSize - this.cameraX;
   624→          const y = row * this.tileSize - this.cameraY;
   625→          
   626→          ctx.fillStyle = color;
   627→          ctx.fillRect(x, y, this.tileSize, this.tileSize);
   628→          
   629→          // Add subtle grid lines for ground tiles
   630→          ctx.strokeStyle = 'rgba(0,0,0,0.08)';
   631→          ctx.lineWidth = 0.5;
   632→          ctx.strokeRect(x, y, this.tileSize, this.tileSize);
   633→        }
   634→      }
   635→    }
   636→
   637→    // Draw buildings layer
   638→    for (let row = startRow; row < endRow; row++) {
   639→      for (let col = startCol; col < endCol; col++) {
   640→        if (row < 0 || col < 0) continue;
   641→        const tileType = mapData.layers.buildings[row]?.[col];
   642→        if (tileType && tileType !== 0) {
   643→          const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
   644→          const x = col * this.tileSize - this.cameraX;
   645→          const y = row * this.tileSize - this.cameraY;
   646→
   647→          ctx.fillStyle = color;
   648→          ctx.fillRect(x, y, this.tileSize, this.tileSize);
   649→          
   650→          // Building shadow
   651→          ctx.fillStyle = 'rgba(0,0,0,0.15)';
   652→          ctx.fillRect(x, y + this.tileSize - 8, this.tileSize, 8);
   653→        }
   654→      }
   655→    }
   656→
   657→    // Draw grass/outskirts border (rows 0-1 and 13-15)
   658→    ctx.fillStyle = '#6B8E23';
   659→    for (let row = startRow; row < endRow; row++) {
   660→      for (let col = startCol; col < endCol; col++) {
   661→        if (row < 0 || col < 0) continue;
   662→        const tileType = mapData.layers.ground[row]?.[col];
   663→        if (tileType === 2) {
   664→          const x = col * this.tileSize - this.cameraX;
   665→          const y = row * this.tileSize - this.cameraY;
   666→          // Grass pattern
   667→          ctx.fillStyle = '#6B8E23';
   668→          ctx.fillRect(x, y, this.tileSize, this.tileSize);
   669→          // Grass blades
   670→          ctx.fillStyle = '#7CFC00';
   671→          for (let i = 0; i < 5; i++) {
   672→            const bx = x + (i * this.tileSize / 5) + 4;
   673→            const by = y + this.tileSize / 2;
   674→            ctx.fillRect(bx, by - 6, 2, 6);
   675→          }
   676→        }
   677→      }
   678→    }
   679→  }
   680→
   681→  private _renderBuildingLabels(ctx: CanvasRenderingContext2D): void {
   682→    for (const label of mapData.buildingLabels) {
   683→      const centerX = (label.col + label.width / 2) * this.tileSize - this.cameraX;
   684→      const topY = label.row * this.tileSize - this.cameraY - 12;
   685→
   686→      ctx.font = 'bold 14px "Geist", sans-serif';
   687→      ctx.fillStyle = '#FFFFFF';
   688→      ctx.strokeStyle = '#000000';
   689→      ctx.lineWidth = 3;
   690→      ctx.textAlign = 'center';
   691→
   692→      // Draw text with outline
   693→      ctx.strokeText(label.label, centerX, topY);
   694→      ctx.fillText(label.label, centerX, topY);
   695→
   696→      if (label.sublabel) {
   697→        ctx.font = '10px "Geist", sans-serif';
   698→        ctx.strokeText(label.sublabel, centerX, topY + 14);
   699→        ctx.fillText(label.sublabel, centerX, topY + 14);
   700→      }
   701→    }
   702→  }
   703→
   704→  private _renderEntity(ctx: CanvasRenderingContext2D, entity: Entity): void {
   705→    const screenX = entity.x - this.cameraX;
   706→    const screenY = entity.y - this.cameraY;
   707→
   708→    // Don't render if off screen
   709→    if (screenX < -100 || screenX > this.viewWidth + 100 || 
   710→        screenY < -100 || screenY > this.viewHeight + 100) return;
   711→
   712→    const anim = entity.isMoving ? 'Walk' : 'Breathing_Idle';
   713→    const dir = entity.direction;
   714→    const frame = entity.animationFrame;
   715→
   716→    const sprite = spriteLoader.getFrame(entity.characterKey, anim, dir, frame);
   717→    
   718→    if (sprite) {
   719→      // Center the sprite on the entity position
   720→      const spriteW = sprite.width || characterData.characters[entity.characterKey as keyof typeof characterData.characters]?.size?.width || 72;
   721→      const spriteH = sprite.height || characterData.characters[entity.characterKey as keyof typeof characterData.characters]?.size?.height || 72;
   722→      
   723→      // Scale sprites up for visibility (2x scale)
   724→      const scale = 2;
   725→      const drawW = spriteW * scale;
   726→      const drawH = spriteH * scale;
   727→      
   728→      ctx.drawImage(sprite, screenX - drawW / 2, screenY - drawH / 2, drawW, drawH);
   729→    }
   730→
   731→    // Entity name label
   732→    if (!entity.isPlayer) {
   733→      const displayName = characterData.characters[entity.characterKey as keyof typeof characterData.characters]?.displayName || entity.characterKey;
   734→      ctx.font = '10px "Geist", sans-serif';
   735→      ctx.fillStyle = '#FFFFFF';
   736→      ctx.strokeStyle = '#000000';
   737→      ctx.lineWidth = 2;
   738→      ctx.textAlign = 'center';
   739→      ctx.strokeText(displayName, screenX, screenY - 50);
   740→      ctx.fillText(displayName, screenX, screenY - 50);
   741→    }
   742→  }
   743→
   744→  private _renderInteractionIndicators(ctx: CanvasRenderingContext2D): void {
   745→    if (!this.player) return;
   746→    
   747→    for (const [npcId, npc] of this.entities) {
   748→      if (npc.isPlayer || !npc.visible) continue;
   749→      
   750→      const dist = this._distanceBetween(this.player, npc);
   751→      if (dist < this.tileSize * 2 && !this.dialogueActive) {
   752→        const screenX = npc.x - this.cameraX;
   753→        const screenY = npc.y - this.cameraY;
   754→        
   755→        // Draw "!" indicator
   756→        ctx.font = 'bold 20px "Geist", sans-serif';
   757→        ctx.fillStyle = '#FFD700';
   758→        ctx.strokeStyle = '#000000';
   759→        ctx.lineWidth = 2;
   760→        ctx.textAlign = 'center';
   761→        const bounceY = Math.sin(Date.now() / 300) * 4;
   762→        ctx.strokeText('!', screenX, screenY - 60 + bounceY);
   763→        ctx.fillText('!', screenX, screenY - 60 + bounceY);
   764→        
   765→        // Draw "Press Space" hint
   766→        ctx.font = '8px "Geist", sans-serif';
   767→        ctx.fillStyle = '#FFFFFF';
   768→        ctx.strokeStyle = '#000000';
   769→        ctx.lineWidth = 2;
   770→        ctx.strokeText('[Space]', screenX, screenY - 45 + bounceY);
   771→        ctx.fillText('[Space]', screenX, screenY - 45 + bounceY);
   772→      }
   773→    }
   774→  }
   775→
   776→  private _renderTriggerHints(ctx: CanvasRenderingContext2D): void {
   777→    // Subtle visual hint for the gossip area
   778→    const gossipZone = this.triggerZones.find(z => z.id === 'market-gossip');
   779→    if (gossipZone && !gossipZone.triggered && saveManager.isObjectiveCompleted(gossipZone.requiresObjective || '')) {
   780→      const x = gossipZone.col * this.tileSize - this.cameraX;
   781→      const y = gossipZone.row * this.tileSize - this.cameraY;
   782→      ctx.fillStyle = 'rgba(255, 215, 0, 0.1)';
   783→      ctx.fillRect(x, y, gossipZone.width * this.tileSize, gossipZone.height * this.tileSize);
   784→    }
   785→  }

   794→  }
