   102→   102→// Deterministic hash for tile decoration patterns
   103→   103→function tileHash(row: number, col: number, seed: number = 0): number {
   104→   104→  let h = seed + row * 374761393 + col * 668265263;
   105→   105→  h = (h ^ (h >> 13)) * 1274126177;
   106→   106→  h = h ^ (h >> 16);
   107→   107→  return (h & 0x7FFFFFFF) / 0x7FFFFFFF; // Returns 0..1
   108→   108→}

   163→   163→  private _computeBuildingGroups(): void {
   164→   164→    // For each building tile type, find contiguous rectangular groups
   165→   165→    const buildingTypes = [5, 6]; // Church and Market building types
   166→   166→    for (const tileType of buildingTypes) {
   167→   167→      const visited: Set<string> = new Set();
   168→   168→      const groups: { minRow: number; minCol: number; maxRow: number; maxCol: number }[] = [];
   169→   169→      
   170→   170→      for (let row = 0; row < mapData.height; row++) {
   171→   171→        for (let col = 0; col < mapData.width; col++) {
   172→   172→          const key = `${row},${col}`;
   173→   173→          if (visited.has(key)) continue;
   174→   174→          if (mapData.layers.buildings[row]?.[col] !== tileType) continue;
   175→   175→          
   176→   176→          // Find bounding box of contiguous group
   177→   177→          let minRow = row, maxRow = row, minCol = col, maxCol = col;
   178→   178→          const toVisit: [number, number][] = [[row, col]];
   179→   179→          
   180→   180→          while (toVisit.length > 0) {
   181→   181→            const [r, c] = toVisit.pop()!;
   182→   182→            const k = `${r},${c}`;
   183→   183→            if (visited.has(k)) continue;
   184→   184→            if (mapData.layers.buildings[r]?.[c] !== tileType) continue;
   185→   185→            visited.add(k);
   186→   186→            minRow = Math.min(minRow, r);
   187→   187→            maxRow = Math.max(maxRow, r);
   188→   188→            minCol = Math.min(minCol, c);
   189→   189→            maxCol = Math.max(maxCol, c);
   190→   190→            toVisit.push([r - 1, c], [r + 1, c], [r, c - 1], [r, c + 1]);
   191→   191→          }
   192→   192→          
   193→   193→          groups.push({ minRow, minCol, maxRow, maxCol });
   194→   194→        }
   195→   195→      }
   196→   196→      
   197→   197→      this.buildingGroups.set(tileType, groups);
   198→   198→    }
   199→   199→  }

   305→   305→  /**
   306→   306→   * Spawn a golden celebration burst at the given world position.
   307→   307→   * Used when an objective is completed.
   308→   308→   */
   309→   309→  private _spawnCelebrationBurst(worldX: number, worldY: number): void {
   310→   310→    const colors = ['#FFD700', '#FFA500', '#FFEC8B', '#FFEE00', '#FFC125'];
   311→   311→    const count = 28;
   312→   312→    for (let i = 0; i < count; i++) {
   313→   313→      const angle = (i / count) * Math.PI * 2 + Math.random() * 0.3;
   314→   314→      const speed = 60 + Math.random() * 80;
   315→   315→      this.burstParticles.push({
   316→   316→        x: worldX,
   317→   317→        y: worldY,
   318→   318→        vx: Math.cos(angle) * speed,
   319→   319→        vy: Math.sin(angle) * speed - 30, // Slight upward bias
   320→   320→        life: 0,
   321→   321→        maxLife: 1.0 + Math.random() * 0.6,
   322→   322→        color: colors[Math.floor(Math.random() * colors.length)],
   323→   323→        size: 2 + Math.random() * 3,
   324→   324→        gravity: 80,
   325→   325→      });
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  private _updateBurstParticles(dt: number): void {
   330→   330→    for (let i = this.burstParticles.length - 1; i >= 0; i--) {
   331→   331→      const p = this.burstParticles[i];
   332→   332→      p.life += dt;
   333→   333→      p.vy += p.gravity * dt;     // gravity pulls down
   334→   334→      p.x += p.vx * dt;
   335→   335→      p.y += p.vy * dt;
   336→   336→      // Air resistance
   337→   337→      p.vx *= 0.96;
   338→   338→      p.vy *= 0.96;
   339→   339→      if (p.life >= p.maxLife) {
   340→   340→        this.burstParticles.splice(i, 1);
   341→   341→      }
   342→   342→    }
   343→   343→  }
   344→   344→
   345→   345→  private _renderBurstParticles(ctx: CanvasRenderingContext2D): void {
   346→   346→    for (const p of this.burstParticles) {
   347→   347→      const screenX = p.x - this.cameraX;
   348→   348→      const screenY = p.y - this.cameraY;
   349→   349→      if (screenX < -20 || screenX > this.viewWidth + 20 || screenY < -20 || screenY > this.viewHeight + 20) continue;
   350→   350→      const lifeRatio = p.life / p.maxLife;
   351→   351→      const alpha = Math.max(0, 1 - lifeRatio);
   352→   352→      // Glow
   353→   353→      ctx.globalAlpha = alpha * 0.4;
   354→   354→      ctx.fillStyle = p.color;
   355→   355→      ctx.beginPath();
   356→   356→      ctx.arc(screenX, screenY, p.size * 2.5, 0, Math.PI * 2);
   357→   357→      ctx.fill();
   358→   358→      // Core
   359→   359→      ctx.globalAlpha = alpha;
   360→   360→      ctx.fillStyle = p.color;
   361→   361→      ctx.beginPath();
   362→   362→      ctx.arc(screenX, screenY, p.size, 0, Math.PI * 2);
   363→   363→      ctx.fill();
   364→   364→    }
   365→   365→    ctx.globalAlpha = 1;
   366→   366→  }

   417→   417→  private _handleInteract(): void {
   418→   418→    if (this.dialogueActive) return;
   419→   419→    if (this.interactCooldown > 0) return;
   420→   420→    
   421→   421→    // Check if player is near an NPC they can talk to
   422→   422→    if (!this.player) return;
   423→   423→
   424→   424→    for (const [npcId, npc] of this.entities) {
   425→   425→      if (npc.isPlayer || !npc.visible) continue;
   426→   426→      
   427→   427→      const dist = this._distanceBetween(this.player, npc);
   428→   428→      if (dist < this.tileSize * 2) {
   429→   429→        this.interactCooldown = 500;
   430→   430→        
   431→   431→        // Determine which dialogue to trigger
   432→   432→        if (npcId === 'mang-tenyo' && !saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   433→   433→          this._startDialogue('mang-tenyo-first');
   434→   434→          return;
   435→   435→        }
   436→   436→        // Mang Tenyo can be talked to again but won't repeat the quest dialogue
   437→   437→        if (npcId === 'mang-tenyo' && saveManager.isObjectiveCompleted('obj.ch1.follow_tenyo')) {
   438→   438→          this._startDialogue('mang-tenyo-repeat');
   439→   439→          return;
   440→   440→        }
   441→   441→      }
   442→   442→    }
   443→   443→  }

   445→   445→  private _distanceBetween(a: Entity, b: Entity): number {
   446→   446→    const dx = a.x - b.x;
   447→   447→    const dy = a.y - b.y;
   448→   448→    return Math.sqrt(dx * dx + dy * dy);
   449→   449→  }

   452→   452→    const dialogue = dialogueData.dialogues[dialogueId as keyof typeof dialogueData.dialogues];
   453→   453→    if (!dialogue) {
   454→   454→      console.warn(`Dialogue not found: ${dialogueId}`);
   455→   455→      return;
   456→   456→    }
   457→   457→
   458→   458→    this.dialogueActive = true;
   459→   459→    this.currentDialogueId = dialogueId;
   460→   460→    this.currentDialogueLineIndex = 0;
   461→   461→
   462→   462→    gameEvents.emit('dialogue:start', {
   463→   463→      dialogueId,
   464→   464→      line: dialogue.lines[0],
   465→   465→      lineIndex: 0,
   466→   466→      totalLines: dialogue.lines.length,
   467→   467→    });

   470→   470→  advanceDialogue(): void {
   471→   471→    if (!this.currentDialogueId) return;
   472→   472→    
   473→   473→    const dialogue = dialogueData.dialogues[this.currentDialogueId as keyof typeof dialogueData.dialogues];

   479→   479→      // Dialogue complete
   480→   480→      this._endDialogue(dialogue);
   481→   481→      return;
   482→   482→    }
   483→   483→
   484→   484→    const line = dialogue.lines[this.currentDialogueLineIndex];
   485→   485→    
   486→   486→    // Check for time transition
   487→   487→    if (line.isTimeTransition) {
   488→   488→      this.timeOfDay = 'morning';
   489→   489→      this.chapterPhase = 'ibarra-sighting';
   490→   490→      saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   491→   491→      gameEvents.emit('time:transition', 'morning');
   492→   492→    }
   493→   493→
   494→   494→    // Check for Ibarra sighting
   495→   495→    if (line.isIbarraSighting) {
   496→   496→      this.ibarraAppeared = true;
   497→   497→      // Show Ibarra NPC
   498→   498→      const ibaraEntity = this.entities.get('ibara');
   499→   499→      if (ibaraEntity) {
   500→   500→        ibaraEntity.visible = true;
   501→   501→      }
   502→   502→    }
   503→   503→
   504→   504→    gameEvents.emit('dialogue:line', {
   505→   505→      dialogueId: this.currentDialogueId,
   506→   506→      line,
   507→   507→      lineIndex: this.currentDialogueLineIndex,
   508→   508→      totalLines: dialogue.lines.length,
   509→   509→    });
   510→   510→  }

   635→   635→  private _gameLoop(timestamp: number): void {
   636→   636→    const deltaTime = (timestamp - this.lastTimestamp) / 1000;
   637→   637→    this.lastTimestamp = timestamp;
   638→   638→    this.gameTime += deltaTime;
   639→   639→
   640→   640→    this._update(deltaTime);
   641→   641→    this._render();

   643→   643→    this.animationFrameId = requestAnimationFrame(this._gameLoop.bind(this));

   646→   646→  private _update(dt: number): void {
   647→   647→    if (!this.player) return;
   648→   648→    
   649→   649→    // Update interact cooldown
   650→   650→    if (this.interactCooldown > 0) {
   651→   651→      this.interactCooldown -= dt * 1000;
   652→   652→    }
   653→   653→
   654→   654→    // Don't allow movement during dialogue
   655→   655→    if (this.dialogueActive) return;
   656→   656→
   657→   657→    // Handle intro state
   658→   658→    if (this.chapterPhase === 'intro' && !this.introShown) {
   659→   659→      this.introShown = true;
   660→   660→      this._startDialogue('intro-text');
   661→   661→      return;
   662→   662→    }
   663→   663→
   664→   664→    // Player movement
   665→   665→    let moveX = 0;
   666→   666→    let moveY = 0;
   667→   667→
   668→   668→    if (this.keys.has('w') || this.keys.has('arrowup')) moveY = -1;
   669→   669→    if (this.keys.has('s') || this.keys.has('arrowdown')) moveY = 1;
   670→   670→    if (this.keys.has('a') || this.keys.has('arrowleft')) moveX = -1;
   671→   671→    if (this.keys.has('d') || this.keys.has('arrowright')) moveX = 1;
   672→   672→
   673→   673→    // Normalize diagonal movement
   674→   674→    if (moveX !== 0 && moveY !== 0) {
   675→   675→      const len = Math.sqrt(moveX * moveX + moveY * moveY);
   676→   676→      moveX /= len;
   677→   677→      moveY /= len;
   678→   678→    }
   679→   679→
   680→   680→    this.player.isMoving = moveX !== 0 || moveY !== 0;
   681→   681→
   682→   682→    if (this.player.isMoving) {
   683→   683→      // Determine direction from movement vector
   684→   684→      this.player.direction = this._vectorToDirection(moveX, moveY);
   685→   685→      
   686→   686→      const newX = this.player.x + moveX * this.player.speed * dt;
   687→   687→      const newY = this.player.y + moveY * this.player.speed * dt;
   688→   688→
   689→   689→      // Collision check
   690→   690→      if (!this._isBlocked(newX, newY)) {
   691→   691→        this.player.x = newX;
   692→   692→        this.player.y = newY;
   693→   693→      }
   694→   694→
   695→   695→      // Update animation
   696→   696→      this.player.animationTimer += dt;
   697→   697→      const animConfig = characterData.characters.student.animations.Walk;
   698→   698→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   699→   699→        this.player.animationTimer = 0;
   700→   700→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   701→   701→      }
   702→   702→    } else {
   703→   703→      // Idle animation
   704→   704→      this.player.animationTimer += dt;
   705→   705→      const animConfig = characterData.characters.student.animations.Breathing_Idle;
   706→   706→      if (this.player.animationTimer >= 1 / animConfig.frameRate) {
   707→   707→        this.player.animationTimer = 0;
   708→   708→        this.player.animationFrame = (this.player.animationFrame + 1) % animConfig.frames;
   709→   709→      }
   710→   710→    }
   711→   711→
   712→   712→    // Update camera to follow player
   713→   713→    this.cameraX = this.player.x - this.viewWidth / 2;
   714→   714→    this.cameraY = this.player.y - this.viewHeight / 2;
   715→   715→
   716→   716→    // Clamp camera to map bounds
   717→   717→    const mapPixelWidth = mapData.width * this.tileSize;
   718→   718→    const mapPixelHeight = mapData.height * this.tileSize;
   719→   719→    this.cameraX = Math.max(0, Math.min(this.cameraX, mapPixelWidth - this.viewWidth));
   720→   720→    this.cameraY = Math.max(0, Math.min(this.cameraY, mapPixelHeight - this.viewHeight));
   721→   721→
   722→   722→    // NPC idle animations
   723→   723→    for (const [npcId, npc] of this.entities) {
   724→   724→      if (npc.isPlayer || !npc.visible) continue;
   725→   725→      npc.animationTimer += dt;
   726→   726→      const animConfig = characterData.characters[npc.characterKey as keyof typeof characterData.characters]?.animations?.Breathing_Idle;
   727→   727→      if (animConfig) {
   728→   728→        if (npc.animationTimer >= 1 / animConfig.frameRate) {
   729→   729→          npc.animationTimer = 0;
   730→   730→          npc.animationFrame = (npc.animationFrame + 1) % animConfig.frames;
   731→   731→        }
   732→   732→      }
   733→   733→    }
   734→   734→
   735→   735→    // Check trigger zones
   736→   736→    this._checkTriggerZones();
   737→   737→
   738→   738→    // Update particles
   739→   739→    this._updateParticles(dt);
   740→   740→
   741→   741→    // Update burst particles (celebration effect)
   742→   742→    this._updateBurstParticles(dt);
   743→   743→
   744→   744→    // Update game state in SaveManager
   745→   745→    saveManager.updateGameState({
   746→   746→      playerPosition: {
   747→   747→        row: Math.floor(this.player.y / this.tileSize),
   748→   748→        col: Math.floor(this.player.x / this.tileSize),
   749→   749→      },
   750→   750→      playerDirection: this.player.direction,
   751→   751→      currentDialogue: this.currentDialogueId,
   752→   752→      timeOfDay: this.timeOfDay,
   753→   753→      chapterPhase: this.chapterPhase,
   754→   754→    });
   755→   755→  }
   756→   756→
   757→   757→  private _vectorToDirection(x: number, y: number): Direction {
   758→   758→    // Map movement vector to 8-direction name
   759→   759→    const angle = Math.atan2(y, x) * (180 / Math.PI);
   760→   760→    
   761→   761→    if (angle >= -22.5 && angle < 22.5) return 'east';
   762→   762→    if (angle >= 22.5 && angle < 67.5) return 'south-east';
   763→   763→    if (angle >= 67.5 && angle < 112.5) return 'south';
   764→   764→    if (angle >= 112.5 && angle < 157.5) return 'south-west';
   765→   765→    if (angle >= 157.5 || angle < -157.5) return 'west';
   766→   766→    if (angle >= -157.5 && angle < -112.5) return 'north-west';
   767→   767→    if (angle >= -112.5 && angle < -67.5) return 'north';
   768→   768→    if (angle >= -67.5 && angle < -22.5) return 'north-east';
   769→   769→    return 'south';
   770→   770→  }
   771→   771→
   772→   772→  private _isBlocked(x: number, y: number): boolean {
   773→   773→    // Check collision map
   774→   774→    const col = Math.floor(x / this.tileSize);
   775→   775→    const row = Math.floor(y / this.tileSize);
   776→   776→    
   777→   777→    if (row < 0 || row >= mapData.height || col < 0 || col >= mapData.width) return true;
   778→   778→    
   779→   779→    // Check player sprite footprint (check a small area around center)
   780→   780→    const checks = [
   781→   781→      { x: x - 16, y: y - 16 },
   782→   782→      { x: x + 16, y: y - 16 },
   783→   783→      { x: x - 16, y: y + 16 },
   784→   784→      { x: x + 16, y: y + 16 },
   785→   785→    ];
   786→   786→    
   787→   787→    for (const check of checks) {
   788→   788→      const cCol = Math.floor(check.x / this.tileSize);
   789→   789→      const cRow = Math.floor(check.y / this.tileSize);
   790→   790→      if (cRow < 0 || cRow >= mapData.height || cCol < 0 || cCol >= mapData.width) return true;
   791→   791→      if (mapData.layers.collision[cRow][cCol] === 1) return true;
   792→   792→    }
   793→   793→    
   794→   794→    return false;
   795→   795→  }
   796→   796→
   797→   797→  private _checkTriggerZones(): void {
   798→   798→    if (!this.player || this.dialogueActive) return;
   799→   799→    
   800→   800→    const playerRow = Math.floor(this.player.y / this.tileSize);
   801→   801→    const playerCol = Math.floor(this.player.x / this.tileSize);
   802→   802→
   803→   803→    for (const zone of this.triggerZones) {
   804→   804→      if (zone.triggered) continue;
   805→   805→
   806→   806→      // Check prerequisite objective
   807→   807→      if (zone.requiresObjective && !saveManager.isObjectiveCompleted(zone.requiresObjective)) continue;
   808→   808→
   809→   809→      // Check if player is in the zone
   810→   810→      const inZone = playerCol >= zone.col && playerCol < zone.col + zone.width &&
   811→   811→                     playerRow >= zone.row && playerRow < zone.row + zone.height;
   812→   812→
   813→   813→      // Or within radius of NPC
   814→   814→      let nearNpc = false;
   815→   815→      if (zone.radius > 0 && zone.npcId) {
   816→   816→        const npc = this.entities.get(zone.npcId);
   817→   817→        if (npc && npc.visible) {
   818→   818→          nearNpc = this._distanceBetween(this.player, npc) < this.tileSize * zone.radius;
   819→   819→        }
   820→   820→      }
   821→   821→
   822→   822→      if (inZone || nearNpc) {
   823→   823→        zone.triggered = true;
   824→   824→
   825→   825→        switch (zone.type) {
   826→   826→          case 'npc-interact':
   827→   827→            // NPC interaction zones just highlight the NPC, actual dialogue starts on interact
   828→   828→            break;
   829→   829→          case 'auto-trigger':
   830→   830→            // Generic auto-trigger: if zone has a dialogueId, start that dialogue
   831→   831→            if (zone.dialogueId) {
   832→   832→              // Special handling for Ibarra sighting (time transition)
   833→   833→              if (zone.id === 'ibarra-sighting') {
   834→   834→                this.timeOfDay = 'morning';
   835→   835→                this.chapterPhase = 'ibarra-sighting';
   836→   836→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   837→   837→                this._updateEntityVisibility();
   838→   838→              }
   839→   839→              this._startDialogue(zone.dialogueId);
   840→   840→            } else {
   841→   841→              // Legacy hardcoded triggers as fallback
   842→   842→              if (zone.id === 'market-gossip') {
   843→   843→                this._startDialogue('market-gossip');
   844→   844→              } else if (zone.id === 'ibarra-sighting') {
   845→   845→                this.timeOfDay = 'morning';
   846→   846→                this.chapterPhase = 'ibarra-sighting';
   847→   847→                saveManager.updateGameState({ timeOfDay: 'morning', chapterPhase: 'ibarra-sighting' });
   848→   848→                this._updateEntityVisibility();
   849→   849→                this._startDialogue('ibarra-sighting');
   850→   850→              }
   851→   851→            }
   852→   852→            break;
   853→   853→        }
   854→   854→      }
   855→   855→    }
   856→   856→  }

   860→   860→  private _initParticles(): void {
   861→   861→    // Seed some initial particles
   862→   862→    for (let i = 0; i < 15; i++) {
   863→   863→      this._spawnParticle();
   864→   864→    }

   866→   866→
   867→   867→  private _spawnParticle(): void {
   868→   868→    if (!this.player) return;
   869→   869→    // Spawn particle near the camera view area
   870→   870→    const cx = this.cameraX + this.viewWidth / 2;
   871→   871→    const cy = this.cameraY + this.viewHeight / 2;
   872→   872→    this.particles.push({
   873→   873→      x: cx + (Math.random() - 0.5) * this.viewWidth * 0.8,
   874→   874→      y: cy + (Math.random() - 0.5) * this.viewHeight * 0.8,
   875→   875→      vx: (Math.random() - 0.5) * 8, // Slow drift
   876→   876→      vy: -Math.random() * 4 - 1, // Slight upward drift
   877→   877→      alpha: 0,
   878→   878→      size: Math.random() * 2.5 + 1,
   879→   879→      life: 0,
   880→   880→      maxLife: Math.random() * 6 + 4, // 4-10 seconds
   881→   881→    });
   882→   882→  }
   883→   883→
   884→   884→  private _updateParticles(dt: number): void {
   885→   885→    // Spawn new particles periodically
   886→   886→    this.particleSpawnTimer += dt;
   887→   887→    if (this.particleSpawnTimer > 0.3 && this.particles.length < 25) {
   888→   888→      this.particleSpawnTimer = 0;
   889→   889→      this._spawnParticle();
   890→   890→    }
   891→   891→
   892→   892→    // Update existing particles
   893→   893→    for (let i = this.particles.length - 1; i >= 0; i--) {
   894→   894→      const p = this.particles[i];
   895→   895→      p.life += dt;
   896→   896→      p.x += p.vx * dt;
   897→   897→      p.y += p.vy * dt;
   898→   898→      // Gentle oscillation
   899→   899→      p.x += Math.sin(this.gameTime * 0.5 + i) * 0.3;
   900→   900→
   901→   901→      // Fade in then fade out
   902→   902→      const lifeRatio = p.life / p.maxLife;
   903→   903→      if (lifeRatio < 0.2) {
   904→   904→        p.alpha = lifeRatio / 0.2 * 0.4; // Fade in
   905→   905→      } else if (lifeRatio > 0.7) {
   906→   906→        p.alpha = (1 - lifeRatio) / 0.3 * 0.4; // Fade out
   907→   907→      } else {
   908→   908→        p.alpha = 0.4; // Visible
   909→   909→      }
   910→   910→
   911→   911→      // Remove dead particles

   975→   975→  /**
   976→   976→   * Render a golden arrow at the screen edge pointing toward the next incomplete objective.
   977→   977→   * Only shown when no dialogue is active and the player isn't standing in the objective zone.
   978→   978→   */
   979→   979→  private _renderWaypointArrow(ctx: CanvasRenderingContext2D): void {
   980→   980→    if (!this.player || this.dialogueActive) return;
   981→   981→    if (this.chapterPhase === 'intro' || this.chapterPhase === 'complete') return;
   982→   982→
   983→   983→    // Find the next incomplete objective
   984→   984→    const objectivesOrder = ['obj.ch1.follow_tenyo', 'obj.ch1.overhear_gossip', 'obj.ch1.see_ibarra'];
   985→   985→    const nextObjId = objectivesOrder.find(id => !saveManager.isObjectiveCompleted(id));
   986→   986→    if (!nextObjId) return;
   987→   987→
   988→   988→    const loc = OBJECTIVE_LOCATIONS[nextObjId];
   989→   989→    if (!loc) return;
   990→   990→
   991→   991→    const targetX = loc.col * this.tileSize + this.tileSize / 2;
   992→   992→    const targetY = loc.row * this.tileSize + this.tileSize / 2;
   993→   993→
   994→   994→    // Vector from player to target
   995→   995→    const dx = targetX - this.player.x;
   996→   996→    const dy = targetY - this.player.y;
   997→   997→    const dist = Math.sqrt(dx * dx + dy * dy);
   998→   998→
   999→   999→    // Don't show arrow if very close
  1000→  1000→    if (dist < this.tileSize * 2.5) return;
  1001→  1001→
  1002→  1002→    // Compute target position on screen (may be off-screen)
  1003→  1003→    const screenTargetX = targetX - this.cameraX;
  1004→  1004→    const screenTargetY = targetY - this.cameraY;
  1005→  1005→
  1006→  1006→    const cx = this.viewWidth / 2;
  1007→  1007→    const cy = this.viewHeight / 2;
  1008→  1008→
  1009→  1009→    // If the target is on screen, don't show the edge arrow (the indicator is enough)
  1010→  1010→    const onScreen = screenTargetX > 40 && screenTargetX < this.viewWidth - 40 &&
  1011→  1011→                     screenTargetY > 80 && screenTargetY < this.viewHeight - 80;
  1012→  1012→    if (onScreen) return;
  1013→  1013→
  1014→  1014→    // Direction from center of screen to target
  1015→  1015→    const angle = Math.atan2(dy, dx);
  1016→  1016→
  1017→  1017→    // Place arrow at a fixed radius from screen center, clamped to a margin inside the screen
  1018→  1018→    const margin = 80;
  1019→  1019→    const maxRadiusX = this.viewWidth / 2 - margin;
  1020→  1020→    const maxRadiusY = this.viewHeight / 2 - margin;
  1021→  1021→    // Project the direction onto the screen-edge ellipse
  1022→  1022→    // Solve for t such that (t*cos(angle), t*sin(angle)) lies on the ellipse (maxRX, maxRY)
  1023→  1023→    const cosA = Math.cos(angle);
  1024→  1024→    const sinA = Math.sin(angle);
  1025→  1025→    const t = 1 / Math.sqrt((cosA * cosA) / (maxRadiusX * maxRadiusX) + (sinA * sinA) / (maxRadiusY * maxRadiusY));
  1026→  1026→    const arrowX = cx + cosA * t;
  1027→  1027→    const arrowY = cy + sinA * t;
  1028→  1028→
  1029→  1029→    // Pulsing effect
  1030→  1030→    const pulse = Math.sin(this.gameTime * 4) * 0.15 + 0.85;
  1031→  1031→
  1032→  1032→    ctx.save();
  1033→  1033→    ctx.translate(arrowX, arrowY);
  1034→  1034→    ctx.rotate(angle);
  1035→  1035→
  1036→  1036→    // Outer glow
  1037→  1037→    ctx.fillStyle = `rgba(255,215,0,${0.15 * pulse})`;
  1038→  1038→    ctx.beginPath();
  1039→  1039→    ctx.arc(0, 0, 28, 0, Math.PI * 2);
  1040→  1040→    ctx.fill();
  1041→  1041→
  1042→  1042→    // Arrow shadow
  1043→  1043→    ctx.fillStyle = 'rgba(0,0,0,0.5)';
  1044→  1044→    ctx.beginPath();
  1045→  1045→    ctx.moveTo(18, 0);
  1046→  1046→    ctx.lineTo(-8, -12);
  1047→  1047→    ctx.lineTo(-4, 0);
  1048→  1048→    ctx.lineTo(-8, 12);
  1049→  1049→    ctx.closePath();
  1050→  1050→    ctx.fill();
  1051→  1051→
  1052→  1052→    // Arrow body
  1053→  1053→    ctx.fillStyle = `rgba(255,215,0,${pulse})`;
  1054→  1054→    ctx.strokeStyle = 'rgba(180,120,0,1)';
  1055→  1055→    ctx.lineWidth = 1.5;
  1056→  1056→    ctx.beginPath();
  1057→  1057→    ctx.moveTo(15, 0);
  1058→  1058→    ctx.lineTo(-10, -10);
  1059→  1059→    ctx.lineTo(-6, 0);
  1060→  1060→    ctx.lineTo(-10, 10);
  1061→  1061→    ctx.closePath();
  1062→  1062→    ctx.fill();
  1063→  1063→    ctx.stroke();
  1064→  1064→
  1065→  1065→    ctx.restore();
  1066→  1066→
  1067→  1067→    // Distance label below arrow
  1068→  1068→    const distTiles = Math.round(dist / this.tileSize);
  1069→  1069→    ctx.font = 'bold 10px "Geist", sans-serif';
  1070→  1070→    ctx.fillStyle = 'rgba(0,0,0,0.6)';
  1071→  1071→    ctx.fillRect(arrowX - 22, arrowY + 14, 44, 14);
  1072→  1072→    ctx.fillStyle = '#FFD700';
  1073→  1073→    ctx.textAlign = 'center';
  1074→  1074→    ctx.fillText(`${distTiles}m →`, arrowX, arrowY + 24);
  1075→  1075→  }

  1103→  1103→  private _renderMap(ctx: CanvasRenderingContext2D): void {

  1115→  1115→        if (tileType && tileType !== 0) {
  1116→  1116→          const x = col * this.tileSize - this.cameraX;

  1122→  1122→              break;

  1124→  1124→              this._renderGrass(ctx, x, y, row, col);
  1125→  1125→              break;
  1126→  1126→            case 3: // Plaza (tan/dirt) - walkable area

  1131→  1131→              break;
  1132→  1132→            default:

  1147→  1147→          const x = col * this.tileSize - this.cameraX;
  1148→  1148→          const y = row * this.tileSize - this.cameraY;
  1149→  1149→
  1150→  1150→          // Base building fill
  1151→  1151→          const color = mapData.tileColors[String(tileType) as keyof typeof mapData.tileColors];
  1152→  1152→          ctx.fillStyle = color;
  1153→  1153→          ctx.fillRect(x, y, this.tileSize, this.tileSize);

  1155→  1155→      }
  1156→  1156→    }
  1157→  1157→  }

  1688→  1688→  private _renderEntity(ctx: CanvasRenderingContext2D, entity: Entity): void {

  1690→  1690→    const screenY = entity.y - this.cameraY;
  1691→  1691→
  1692→  1692→    // Don't render if off screen

  1694→  1694→        screenY < -150 || screenY > this.viewHeight + 150) return;

  1710→  1710→      const drawH = spriteH * scale;

  1713→  1713→      ctx.fillStyle = 'rgba(0,0,0,0.18)';
  1714→  1714→      ctx.beginPath();
  1715→  1715→      ctx.ellipse(screenX, screenY + drawH * 0.25, drawW * 0.35, 6, 0, 0, Math.PI * 2);
  1716→  1716→      ctx.fill();

  1719→  1719→      if (entity.isPlayer) {
  1720→  1720→        // Soft glow around player
  1721→  1721→        ctx.fillStyle = 'rgba(255,220,100,0.08)';
  1722→  1722→        ctx.beginPath();
  1723→  1723→        ctx.ellipse(screenX, screenY, drawW * 0.6, drawH * 0.5, 0, 0, Math.PI * 2);
  1724→  1724→        ctx.fill();

  1726→  1726→        // Subtle outline stroke

  1736→  1736→      }

  1738→  1738→      // Draw the sprite

  1740→  1740→    }

  1743→  1743→    if (!entity.isPlayer) {

  1745→  1745→

  1750→  1750→      const nameY = screenY - 75;

  1758→  1758→      ctx.textAlign = 'center';

  1760→  1760→    }
